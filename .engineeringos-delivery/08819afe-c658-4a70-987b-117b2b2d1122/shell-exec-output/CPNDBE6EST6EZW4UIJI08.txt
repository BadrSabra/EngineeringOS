  1254  const attemptedModels: string[] = [];

  1284        continue;
  1285      }
  1286      if (err instanceof GroqClientError) {

  1322/**
  1323 * Streaming completion via OpenRouter.
  1324 *
  1325 * OR-003: same context trim and conservative maxTokens as `openrouterCompleteRaw`.
  1326 * OR-005: retries the stream once on transient errors.
  1327 */
  1328export async function* openrouterCompleteStream(
  1329  messages: RawMessage[],
  1330  opts: Omit<OpenAICompatibleStreamOptions, "baseUrl" | "providerName" | "extraHeaders">,
  1331): AsyncGenerator<string> {

  1333  const fullOpts: OpenAICompatibleStreamOptions = {
  1334    maxTokens: OPENROUTER_DEFAULT_MAX_TOKENS,

  1337    providerName: "OpenRouter",
  1338    extraHeaders: OPENROUTER_EXTRA_HEADERS,
  1339  };

  1342  const initialModel = opts.model;
  1343  if (initialModel && capability && !isCatalogFreeModelForCapability(initialModel, { capability })) {
  1344    throw new GroqClientError(
  1345      "INVALID_CONFIG",
  1346      `OpenRouter stream requires a currently-free model matching capability="${capability}"`,
  1347      { context: { providerName: "OpenRouter", providerModel: initialModel } },
  1348    );
  1349  }
  1350  const resolved = initialModel
  1351    ? buildFallbackChainFromId(initialModel, opts.taskType, capability ? { capability } : {})
  1352    : resolveFallbackChain({
  1353        capability: capability ?? "chat",
  1354        quality: opts.quality ?? "fast",
  1355      }).map((model) => model.id);
  1356  const maxFallbackModels =
  1357    Number.isInteger(opts.maxFallbackModels) && opts.maxFallbackModels! > 0
  1358      ? opts.maxFallbackModels
  1359      : undefined;
  1360  const chain = (maxFallbackModels ? resolved.slice(0, maxFallbackModels) : resolved);
  1361  const attemptedModels: string[] = [];
  1362  let lastError: GroqClientError | undefined;
  1363
  1364  for (let index = 0; index < chain.length; index++) {
  1365    const model = chain[index]!;
  1366    attemptedModels.push(model);
  1367    let emitted = false;
  1368    let transientRetried = false;
  1369    while (true) {
  1370      try {
  1371        for await (const delta of oacCompleteStream(trimmed, { ...fullOpts, model })) {
  1372          emitted = true;
  1373          yield delta;
  1374        }
  1375        return;
  1376      } catch (err) {

  1378        // Once bytes have been emitted, retrying would duplicate or splice the
  1379        // answer. The caller must terminalize the partial stream instead.
  1380        if (emitted || !providerError) {

  1382            "NETWORK_ERROR",
  1383            err instanceof Error ? err.message : "OpenRouter stream disconnected",
  1384            { cause: err, context: { providerName: "OpenRouter", providerModel: model } },
  1385          );
  1386          throw new GroqClientError(terminalError.code, terminalError.message, {
  1387            cause: terminalError,
  1388            context: {
  1389              ...terminalError.toProviderContext(),
  1390              providerModel: terminalError.providerModel ?? model,
  1391              providerAttemptedModels: [...attemptedModels],
  1392            },
  1393          });
  1394        }
  1395        if (
  1396          !transientRetried &&
  1397          opts.retryTransient !== false &&
  1398          isTransientError(providerError) &&
  1399          providerError.retryAfterMs === undefined
  1400        ) {
  1401          transientRetried = true;
  1402          await sleep(1500);
  1403          continue;
  1404        }
  1405        const canAdvance = isModelUnavailableError(providerError) ||
  1406          (opts.retryTransient === false && isTransientError(providerError));
  1407        if (!canAdvance || index >= chain.length - 1) {
  1408          throw new GroqClientError(providerError.code, providerError.message, {
  1409            cause: providerError,
  1410            context: {
  1411              ...providerError.toProviderContext(),
  1412              providerModel: providerError.providerModel ?? model,
  1413              providerAttemptedModels: [...attemptedModels],
  1414            },
  1415          });
  1416        }
  1417        lastError = providerError;
  1418        console.warn(JSON.stringify({
  1419          scope: "openrouter-fallback",
  1420          code: "STREAM_MODEL_FALLBACK",
  1421          reason: providerError.code,
  1422          failedModel: model,
  1423          nextModel: chain[index + 1],
  1424          attempt: index + 1,
  1425        }));
  1426        break;
  1427      }
  1428    }
  1429  }
  1430  if (lastError) {
  1431    throw new GroqClientError(lastError.code, lastError.message, {
  1432      cause: lastError,
  1433      context: { ...lastError.toProviderContext(), providerAttemptedModels: attemptedModels },
  1434    });
  1435  }
  1436  throw new GroqClientError("MODEL_NOT_FOUND", "No OpenRouter free model satisfies the stream capability", {
  1437    context: { providerAttemptedModels: attemptedModels },
  1438  });
