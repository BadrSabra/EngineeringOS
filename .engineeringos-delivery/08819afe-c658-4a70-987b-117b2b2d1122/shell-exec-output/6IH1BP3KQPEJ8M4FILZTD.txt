     1/**
     2 * Base agent pipeline for structured single-shot LLM tasks.
     3 *
     4 * Consolidates the common flow shared by task-agent, scan-analyst, and
     5 * code-reviewer:
     6 *   1. build messages
     7 *   2. select a quality-aware execution plan
     8 *   3. call the configured provider with bounded retries
     9 *   4. parse JSON into the declared schema
    10 *   5. attach a parse-failure marker when validation degrades
    11 */
    12import { GroqClientError, type AgentErrorCode } from "../errors.js";
    13import { agentComplete, type AgentCompleteOpts } from "../agent-complete.js";
    14import { parseAgentResponse } from "../parsing.js";
    15import { assessStructuredOutput, type QualityProfile } from "../quality-engine.js";
    16import { resolveExecutionDecision } from "../model-selection/decision-engine.js";
    17import { decideRetry } from "../quality/retry-controller.js";
    18import type { Message } from "../groq-client.js";
    19import type { ZodType, ZodTypeDef } from "zod";
    20
    21export type AgentRunResult<T> = T & {
    22  _parseError?: { code: AgentErrorCode; message: string; raw: string };
    23};
    24
    25export abstract class BaseAgent<TInput, TOutput> {
    26  protected abstract readonly scope: string;
    27  protected abstract readonly schema: ZodType<TOutput, ZodTypeDef, any>;
    28  protected abstract buildMessages(input: TInput): Message[];
    29  protected abstract fallbackOutput(raw: string): TOutput;
    30
    31  protected buildQualityProfile(_input: TInput): QualityProfile | undefined {
    32    switch (this.scope) {
    33      case "task-agent":
    34        return "task_execution";
    35      case "scan-analyst":
    36        return "analysis";
    37      case "code-reviewer":
    38        return "code_review";
    39      case "workflow-orchestrator":
    40        return "workflow";
    41      default:
    42        return undefined;
    43    }
    44  }
    45
    46  protected buildCompleteOpts(_input: TInput): AgentCompleteOpts {
    47    return {};
    48  }
    49
    50  protected async complete(messages: Message[], opts: AgentCompleteOpts): Promise<{ content: string }> {
    51    try {
    52      return await agentComplete(messages, opts);
    53    } catch (err) {
    54      if (err instanceof GroqClientError && err.code === "NON_200") {
    55        console.warn(JSON.stringify({ scope: this.scope, code: "MODEL_RETRY", originalError: err.code }));
    56        return await agentComplete(messages, opts);
    57      }
    58      throw err;
    59    }
    60  }
    61
    62  async run(input: TInput, opts?: AgentCompleteOpts): Promise<AgentRunResult<TOutput>> {
    63    const messages = this.buildMessages(input);
    64    const executionPlan = resolveExecutionDecision(this.scope, {
    65      qualityProfile: opts?.qualityProfile ?? this.buildQualityProfile(input),
    66    });
    67    const qualityProfile = opts?.qualityProfile ?? this.buildQualityProfile(input) ?? executionPlan.qualityProfile;
    68    const parseResponse = (raw: string) => parseAgentResponse(raw, this.schema, (fallbackRaw) => this.fallbackOutput(fallbackRaw));
    69
    70    opts?.onProgress?.("Calling AI model…");
    71    const response = await this.complete(messages, {
    72      ...this.buildCompleteOpts(input),
    73      ...(opts ?? {}),
    74      qualityProfile,
    75      qualityHints: opts?.qualityHints ?? executionPlan.strictHints,
    76    });
    77
    78    let parsed = parseResponse(response.content);
    79    let parseError = parsed.ok ? undefined : { code: parsed.code, message: parsed.message, raw: parsed.raw };
    80    const initialAssessment = parsed.ok && qualityProfile ? assessStructuredOutput(qualityProfile, parsed.data) : undefined;
    81    let attempt = 1;
    82    let assessment = initialAssessment;
    83    while (attempt < executionPlan.retryLimit) {
    84      const retryDecision = decideRetry({
    85        attempt,
    86        limit: executionPlan.retryLimit,
    87        parseError,
    88        assessment,
    89      });
    90      if (!retryDecision.shouldRetry) break;
    91      console.warn(
    92        JSON.stringify({
    93          scope: this.scope,
    94          code: "MODEL_RETRY",
    95          reason: retryDecision.reason,
    96          qualityProfile,
    97        }),
    98      );
    99
   100      opts?.onProgress?.("Retrying — improving output quality…");
   101      const retryResponse = await this.complete(messages, {
   102        ...this.buildCompleteOpts(input),
   103        ...(opts ?? {}),
   104        qualityProfile,
   105        qualityHints: executionPlan.relaxedHints,
   106      });
   107
   108      parsed = parseResponse(retryResponse.content);
   109      parseError = parsed.ok ? undefined : { code: parsed.code, message: parsed.message, raw: parsed.raw };
   110      assessment = parsed.ok && qualityProfile ? assessStructuredOutput(qualityProfile, parsed.data) : undefined;
   111      attempt += 1;
   112    }
   113
   114    if (!parsed.ok) {
   115      console.warn(JSON.stringify({ scope: this.scope, code: parsed.code, message: parsed.message }));
   116      return { ...parsed.data, _parseError: parseError };
   117    }
   118
   119    assessment = qualityProfile ? assessStructuredOutput(qualityProfile, parsed.data) : undefined;
   120    if (assessment && assessment.decision !== "accept") {
   121      console.warn(
   122        JSON.stringify({
   123          scope: this.scope,
   124          code: "QUALITY_REVIEW_LOW",
   125          qualityProfile,
   126          score: assessment.score,
   127          threshold: assessment.threshold,
   128          reasons: assessment.reasons,
   129        }),
   130      );
   131    }
   132
   133    // parsed.data is TOutput, which satisfies AgentRunResult<TOutput> (= T & { _parseError? })
   134    // because _parseError is optional. Cast is required because TypeScript cannot
   135    // prove T satisfies T & { _parseError?: … } for an unconstrained generic.
   136    return parsed.data as AgentRunResult<TOutput>;
   137  }
   138}

     1/**
     2 * Provider-agnostic completion helper for single-shot agents.
     3 *
     4 * This layer still delegates to the provider-specific transport clients, but
     5 * the provider configuration itself now flows through the registry helpers so
     6 * selection, defaults, and validation share one source of truth.
     7 */
     8import { complete, completeRaw } from "./groq-client.js";
     9import { deepseekCompleteRaw } from "./deepseek-client.js";

    24export type AgentCompleteOpts = {
    25  /** Caller-owned cancellation signal for the complete request and retries. */
    26  signal?: AbortSignal;
    27  /** Per-user API key. Required for DeepSeek and OpenRouter; falls back to GROQ_API_KEY env for Groq. */
    28  apiKey?: string;
    29  /** Which AI provider to use. Defaults to the quality-aware best available provider. */
    30  provider?: ProviderId;
    31  /**
    32   * Optional quality profile used to bias provider selection toward the
    33   * strongest available runtime for this task type.
    34   */
    35  qualityProfile?: QualityProfile;
    36  /** Optional explicit capability hints override. */
    37  qualityHints?: ProviderCapabilityHints;
    38  /**
    39   * Optional progress callback. Called at key execution milestones (model call,
    40   * retry) so callers can stream status updates to the user in real-time.
    41   * The callback may be async; errors are silently swallowed to keep the agent
    42   * pipeline stable.
    43   */
    44  onProgress?: (msg: string) => void | Promise<void>;
    45  /**
    46   * Maximum number of OpenRouter models to try for this request. Structured
    47   * agents use this to keep a provider outage bounded instead of walking the
    48   * entire free catalog.
    49   */
    50  maxFallbackModels?: number;
    51  /** For OpenRouter, advance to another model instead of retrying transient errors. */
    52  retryTransient?: boolean;
    53};
    54
    55function requireApiKey(provider: ProviderConfig, apiKey?: string): string {
    56  if (provider.providerId === "groq") {
    57    return apiKey ?? "";
    58  }
    59  if (!apiKey) {
    60    throw new GroqClientError(
    61      "INVALID_CONFIG",
    62      `${provider.label} API key is required but was not provided`,
    63    );
    64  }
    65  return apiKey;
    66}
    67
    68function assertContent(provider: ProviderConfig, content: string | null | undefined): string {
    69  if (content) return content;
    70  throw new GroqClientError("EMPTY_RESPONSE", `${provider.label} returned no content for this request`);
    71}
    72
    73function inferExecutionScope(
    74  qualityProfile?: QualityProfile,
    75  qualityHints?: ProviderCapabilityHints,
    76): QualityProfile {
    77  if (qualityProfile) return qualityProfile;
    78  if (qualityHints?.requireTools) return "tool_chat";
    79  if (
    80    qualityHints?.requireThinking ||
    81    qualityHints?.requireReasoning ||
    82    (qualityHints?.minimumContext ?? 0) >= 16_000
    83  ) {
    84    return "analysis";
    85  }
    86  return "chat";
    87}

    89async function probeProviderKey(provider: ProviderConfig, apiKey: string): Promise<void> {
    90  const testMessages: Message[] = [{ role: "user", content: "hi" }];
    91  const probePlan = resolveExecutionDecision(
    92    provider.providerId === "openrouter" ? "task_execution" : "chat",
    93  );
    94  const probeModelDecision = resolveExecutionModel(provider.providerId, probePlan);
    95
    96  switch (provider.providerId) {
    97    case "deepseek":
    98      await deepseekCompleteRaw(testMessages, { apiKey, maxTokens: 1, temperature: 0, model: probeModelDecision.model });
    99      return;
   100    case "openrouter":
   101      // Use the same decision engine as the main call path so the validation
   102      // probe exercises the same model-selection rules.
   103      await openrouterCompleteWithFallback(testMessages, {
   104        apiKey,
   105        maxTokens: 1,
   106        temperature: 0,
   107        model: probeModelDecision.model,
   108        capability: probeModelDecision.capability,
   109        quality: probeModelDecision.quality,
   110        requireTools: probePlan.strictHints.requireTools ?? false,
   111      });
   112      return;
   113    case "gemini":
   114      await geminiCompleteRaw(testMessages, { apiKey, maxTokens: 1, temperature: 0, model: probeModelDecision.model });
   115      return;
   116    default:
   117      // groq — uses high-level client with 10s timeout for the probe
   118      await completeRaw(testMessages, { apiKey, maxTokens: 1, timeoutMs: 10_000, model: probeModelDecision.model });
   119      return;
   120  }
   121}

   123/**
   124 * Validates an API key by sending a minimal 1-token probe to the provider.
   125 *
   126 * Returns `{ valid: true }` on success.
   127 * Returns `{ valid: false, reason }` only on AUTH_ERROR — the key is definitively
   128 * rejected by the provider.
   129 * On any transient error (NETWORK_ERROR, TIMEOUT, SERVER_ERROR, NON_200) returns
   130 * `{ valid: true }` so a connectivity blip never prevents the user from saving a
   131 * key that might be perfectly valid.
   132 */
   133export async function validateProviderKey(
   134  provider: ProviderId,
   135  apiKey: string,
   136): Promise<{ valid: boolean; reason?: string }> {
   137  const config = loadProvider(provider);
   138  try {
   139    await probeProviderKey(config, apiKey);
   140    return { valid: true };
   141  } catch (err) {
   142    if (err instanceof GroqClientError && err.code === "AUTH_ERROR") {
   143      return { valid: false, reason: err.message };
   144    }
   145    // Transient error — allow the save; the next real request has provider fallback.
   146    return { valid: true };
   147  }
   148}

   150/**
   151 * Send a single-shot chat completion to the configured provider.
   152 *
   153 * Always returns `{ content: string }` — content is guaranteed non-empty.
   154 * Throws `GroqClientError` on any provider error so callers handle the same
   155 * error union regardless of which provider is active.
   156 */
   157export async function agentComplete(
   158  messages: Message[],
   159  opts: AgentCompleteOpts,
   160): Promise<{ content: string }> {
   161  const qualityHints = opts.qualityHints ?? (opts.qualityProfile ? buildQualityHints(opts.qualityProfile) : undefined);
   162  const executionScope = inferExecutionScope(opts.qualityProfile, qualityHints);
   163  const executionPlan = resolveExecutionDecision(executionScope, {
   164    hasTools: !!qualityHints?.requireTools,
   165    requireTools: !!qualityHints?.requireTools,
   166    qualityProfile: opts.qualityProfile ?? executionScope,
   167  });
   168  const providerDecision = resolveExecutionProvider(executionPlan, opts.provider);
   169  const providerId = providerDecision.providerId;
   170  const provider = loadProvider(providerId);
   171  const apiKey = requireApiKey(provider, opts.apiKey);
   172
   173  switch (provider.providerId) {
   174    case "deepseek": {
   175      const modelDecision = resolveExecutionModel(providerId, executionPlan);
   176      const result = await deepseekCompleteRaw(messages, {
   177        model: modelDecision.model,
   178        apiKey,
   179        responseFormat: qualityHints?.requireJsonMode ? { type: "json_object" } : undefined,
   180        signal: opts.signal,
   181      });
   182      return { content: assertContent(provider, result.content) };
   183    }
   184
   185    case "openrouter": {
   186      // Refresh before resolving the first model, not only inside the
   187      // transport. This keeps the initial model and its bounded fallback
   188      // chain aligned with the currently usable free catalog.
   189      await refreshDynamicCatalog(apiKey);
   190      const modelDecision = resolveExecutionModel(providerId, executionPlan);
   191      const result = await openrouterCompleteWithFallback(messages, {
   192        model: modelDecision.model,
   193        apiKey,
   194        capability: modelDecision.capability,
   195        quality: modelDecision.quality,
   196        requireTools: qualityHints?.requireTools ?? false,
   197        maxFallbackModels: opts.maxFallbackModels,
   198        retryTransient: opts.retryTransient,
   199        responseFormat: qualityHints?.requireJsonMode ? { type: "json_object" } : undefined,
   200        signal: opts.signal,
   201      });
   202      return { content: assertContent(provider, result.content) };
   203    }
   204
   205    case "gemini": {
   206      const modelDecision = resolveExecutionModel(providerId, executionPlan);
   207      const result = await geminiCompleteRaw(messages, {
   208        model: modelDecision.model,
   209        apiKey,
   210        responseFormat: qualityHints?.requireJsonMode ? { type: "json_object" } : undefined,
   211        signal: opts.signal,
   212      });
   213      return { content: assertContent(provider, result.content) };
   214    }
   215
   216    default: {
   217      const modelDecision = resolveExecutionModel(providerId, executionPlan);
   218      const result = await complete(messages, {
   219        model: modelDecision.model,
   220        apiKey: apiKey || undefined,
   221        responseFormat: qualityHints?.requireJsonMode ? { type: "json_object" } : undefined,
   222        signal: opts.signal,
   223      });
   224      return { content: assertContent(provider, result.content) };
   225    }
   226  }
