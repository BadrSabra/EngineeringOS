   509    ? (() => {
   510        const secs = Math.floor((Date.now() - new Date(metric.lastSuccessAt).getTime()) / 1000);
   511        if (secs < 60)   return `${secs}s ago`;
   512        if (secs < 3600) return `${Math.floor(secs / 60)}m ago`;
   513        return `${Math.floor(secs / 3600)}h ago`;
   514      })()
   515    : null;

   518    missing_credentials: 'Not configured',

   522    catalog_stale: 'Model catalog stale',
   523    quota_exhausted: 'Quota exhausted',
   524    rate_limited: 'Rate limited',
   525    circuit_open: 'Circuit open',
   526    provider_outage: 'Provider unavailable',
   527    degraded: 'Degraded',
   528    healthy: 'Healthy',
   529    unknown: 'Ready',
   530  };
   531  const state = metric.availabilityState ?? (
   532    metric.circuitOpen ? 'circuit_open' :
   533      metric.requests === 0 ? 'unknown' :
   534        metric.consecutiveFailures > 0 ? 'degraded' : 'healthy'
   535  );
   536  const isActionable = !['healthy', 'unknown'].includes(state);
   537
   538  if (state === 'circuit_open' || metric.circuitOpen) {
   539    return (
   540      <div className="mt-1.5 space-y-0.5 text-[10px] text-red-400">

   542        <span className="w-1.5 h-1.5 rounded-full bg-red-400 shrink-0" />
   543          <span>{stateLabels[state]}{cooldownSec != null ? ` · ${cooldownSec}s cooldown` : ''}</span>
   544        </div>
   545        {metric.operatorAction && <div className="pl-3 text-red-300/80">{metric.operatorAction}</div>}
   546      </div>
   547    );
   548  }
   549
   550  const isHealthy = metric.consecutiveFailures === 0 && metric.successRate != null && metric.successRate > 0.8;
   551  const isDegraded = metric.consecutiveFailures > 0 || (metric.successRate != null && metric.successRate < 0.8);
   552
   553  return (
   554    <div className={`mt-1.5 space-y-0.5 text-[10px] ${isActionable ? 'text-amber-300' : 'text-muted-foreground'}`}>
   555      <div className="flex items-center gap-1.5">
   556      <span className={`w-1.5 h-1.5 rounded-full shrink-0 ${isHealthy ? 'bg-green-400' : isDegraded ? 'bg-yellow-400' : 'bg-muted-foreground'}`} />
   557      <span className="flex-1 truncate">
   558        {isActionable
   559          ? stateLabels[state] ?? state.replace(/_/g, ' ')
   560          : metric.consecutiveFailures > 0
   561          ? `${metric.consecutiveFailures} consecutive fail${metric.consecutiveFailures > 1 ? 's' : ''}`
   562          : lastSuccessLabel
   563            ? `OK · ${lastSuccessLabel}`
   564            : 'Active'}
