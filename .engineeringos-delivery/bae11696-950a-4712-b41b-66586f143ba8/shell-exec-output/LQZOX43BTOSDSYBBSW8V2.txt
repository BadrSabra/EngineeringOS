     3 *
     4 * Any provider that speaks POST /chat/completions with Bearer auth can be
     5 * driven through this module by supplying a baseUrl and optional extra headers.

    15 * PR-003: 410 and 422 treated as MODEL_UNAVAILABLE (fallback-worthy).
    16 * PR-007: GroqClientError carries providerStatus/providerCode/providerMessage.
    17 * PR-008: QUOTA code for billing exhaustion; MODEL_UNAVAILABLE for 410/422.

    91/**
    92 * An atomic message group:
    93 *   - "single"     — a user, system, or assistant-without-tool-calls message.
    94 *   - "tool_group" — an assistant turn that requested tool calls plus ALL of
    95 *                    the immediately following tool-result messages that belong
    96 *                    to it.  This group is never split: it is either kept whole
    97 *                    or dropped whole.
    98 */
    99type MessageGroup =
   100  | { kind: "single";     message:  RawMessage }
   101  | { kind: "tool_group"; messages: RawMessage[] };
   102
   103/**
   104 * Partition a flat message array into atomic groups.
   105 *
   106 * Walking forward:
   107 *   - An assistant message that carries `tool_calls` opens a tool_group and
   108 *     greedily absorbs the immediately following `tool` role messages whose
   109 *     `tool_call_id` is in that assistant turn's declared IDs.
   110 *   - Everything else becomes a "single" group.
   111 *
   112 * The result lets the trimmer drop or keep complete groups so the conversation
   113 * reaching the model is always structurally valid.
   114 */
   115function groupMessages(messages: RawMessage[]): MessageGroup[] {
   116  const groups: MessageGroup[] = [];
   117  let i = 0;
   118
   119  while (i < messages.length) {
   120    const m = messages[i];
   121
   122    if (
   123      m.role === "assistant" &&
   124      Array.isArray(m.tool_calls) &&
   125      m.tool_calls.length > 0
   126    ) {
   127      // Collect the IDs this assistant turn declared.
   128      const declaredIds = new Set<string>(
   129        m.tool_calls.map((tc) => tc.id).filter((id): id is string => !!id),
   130      );
   131
   132      // Greedily absorb the immediately following tool-result messages.
   133      const groupMsgs: RawMessage[] = [m];
   134      let j = i + 1;
   135      while (j < messages.length && messages[j].role === "tool") {
   136        const toolMsg = messages[j] as { role: "tool"; tool_call_id?: string };
   137        if (toolMsg.tool_call_id && declaredIds.has(toolMsg.tool_call_id)) {
   138          groupMsgs.push(messages[j]);
   139          j++;
   140        } else {
   350      err.code === "MODEL_UNAVAILABLE" ||
   351      err.code === "EMPTY_RESPONSE")
   352  );
   353}

   355/**
   356 * PR-007 / PR-008: extract a structured error code and message from a provider
   357 * response body string. Returns { code, message } where code is the provider's
   358 * own error tag (e.g. "model_not_found") and message is the human text.
   359 */
   360function extractProviderError(body: string): { code?: string; message?: string } {
   361  try {
   362    const parsed = JSON.parse(body) as Record<string, unknown>;
   363    const error = parsed.error;
   364
   365    if (typeof error === "object" && error !== null) {
   366      const errorObj = error as Record<string, unknown>;
   367      return {
   368        code:    asString(errorObj.code) ?? asString(errorObj.type),
   369        message: asString(errorObj.message),
   370      };
   371    }
   372
   373    if (typeof error === "string") {
   374      return { message: error };
   375    }
   376
   377    const message = parsed.message;
   378    if (typeof message === "string") {
   379      return { message };
   380    }
   381  } catch {
   382    // not JSON
   383  }
   384  return {};
   385}

   389 * recoverable by trying another model in the fallback chain — for example,

   392function looksLikeRecoverableOpenRouterRequestError(body: string, providerCode?: string): boolean {
   393  const normalizedBody = body.toLowerCase();
   394  return (
   395    normalizedBody.includes("response_format") ||
   396    normalizedBody.includes("tool_choice") ||
   397    normalizedBody.includes("structured output") ||
   398    normalizedBody.includes("structured outputs") ||
   399    normalizedBody.includes("unsupported parameter") ||
   400    normalizedBody.includes("request parameter") ||
   401    normalizedBody.includes("invalid request") ||
   402    normalizedBody.includes("not supported") ||
   403    normalizedBody.includes("agent harness") ||
   404    normalizedBody.includes("requires an agent") ||
   405    normalizedBody.includes("only available through an agent") ||
   406    normalizedBody.includes("agent-only") ||
   407    normalizedBody.includes("agentic") ||
   408    (providerCode?.includes("invalid_request") ?? false) ||
   409    (providerCode?.includes("unsupported") ?? false)
   410  );

   413const MAX_RETRY_AFTER_MS = 60_000;
   414
   415function parseRetryAfterMs(headers?: Headers): number | undefined {
   416  const value = headers?.get("retry-after")?.trim();
   417  if (!value) return undefined;
   418  const seconds = Number(value);
   419  if (Number.isFinite(seconds)) {
   420    return Math.min(MAX_RETRY_AFTER_MS, Math.max(0, Math.ceil(seconds * 1000)));
   421  }
   422  const timestamp = Date.parse(value);
   423  if (!Number.isFinite(timestamp)) return undefined;
   424  return Math.min(MAX_RETRY_AFTER_MS, Math.max(0, timestamp - Date.now()));
   425}

   432  model?: string,

   435  const { code: pCode, message: pMessage } = extractProviderError(body);
   436  const providerCode = asString(pCode);
   437  const ctx = {
   438    providerStatus:  status,
   439    providerCode,
   440    providerMessage: pMessage ?? body.slice(0, 200),
   441    providerName,
   442    providerModel:   model,
   443    retryAfterMs:    status === 429 ? parseRetryAfterMs(headers) : undefined,
   444  };
   445
   446  // Log the classification decision so errors can be traced back to their root cause.
   447  console.warn(
   448    JSON.stringify({
   449      scope: "openai-compatible-client",
   450      action: "classify_status",
   451      providerName,
   452      model,
   453      providerStatus: status,
   454      providerCode,
   455      providerMessage: pMessage ?? null,
   456      bodyPreview: body.slice(0, 300),
   457    }),
   458  );
   459
   460  if (status === 401 || status === 403) {
   461    return new GroqClientError(
   462      "AUTH_ERROR",
   463      `${providerName} API authentication failed (${status}) — check your API key`,
   464      { context: ctx },
   465    );
   466  }
   467
   468  if (status === 429) {
   469    // PR-008: distinguish QUOTA (credits exhausted) from RATE_LIMITED (per-minute).
   470    const bodyLower = body.toLowerCase();

   472      bodyLower.includes("quota") ||
   473      bodyLower.includes("credits") ||
   474      bodyLower.includes("billing") ||
   475      bodyLower.includes("insufficient")
   476    ) {
   477      return new GroqClientError(
   478        "QUOTA",
   479        `${providerName} billing quota or credits exhausted (${status}) — check your account`,
   480        { context: ctx },
   481      );
   482    }
   483    return new GroqClientError(
   484      "RATE_LIMITED",
   485      `${providerName} API rate limit exceeded — wait a moment before retrying`,
   486      { context: ctx },
   487    );
   488  }
   489

   491    return new GroqClientError(

   503    normalizedBody.includes("model not found") ||
   504    normalizedBody.includes("model unavailable") ||
   505    normalizedBody.includes("unavailable for free") ||
   506    normalizedBody.includes("no endpoints") ||

   508      providerCode?.includes("invalid_model") ||
   509      providerCode?.includes("model_unavailable"));

   514  //   400 — invalid model slug (body confirms it's a model error).
   515  if (providerName === "OpenRouter" && status === 404) {
   516    return new GroqClientError(
   517      "MODEL_NOT_FOUND",
   518      `OpenRouter model not found (404) — the model has been discontinued or removed. ${body.slice(0, 200)}`,
   519      { context: ctx },
   520    );
   521  }
   522
   523  if (providerName === "OpenRouter" && status === 402) {
   524    const accountQuota =
   525      normalizedBody.includes("quota") ||
   526      normalizedBody.includes("credits exhausted") ||
   527      normalizedBody.includes("credit balance") ||
   528      normalizedBody.includes("insufficient funds") ||
   529      normalizedBody.includes("insufficient credits") ||
   530      normalizedBody.includes("balance is") ||
   531      normalizedBody.includes("account limit");

   533      return new GroqClientError(
   534        "QUOTA",
   535        `OpenRouter account quota or credits exhausted (402) — stop safely and check your account`,
   536        { context: ctx },
   537      );
   538    }

   721export async function oacCompleteRaw(
   722  messages: RawMessage[],
   723  opts: OpenAICompatibleOptions,
   724): Promise<RawGroqResponse> {
   725  const {
   726    model = FALLBACK_DEFAULT_MODEL,  // PR-001: no hardcoded string
   727    temperature = 0.2,
   728    maxTokens = opts.outputPhase ? getPhaseBudget(opts.outputPhase).maxOutputTokens : 4096,
   729    timeoutMs = DEFAULT_TIMEOUT_MS,
   730    apiKey,
   731    tools,
   732    baseUrl,
   733    providerName,
   734    extraHeaders = {},
   735    signal,
   736  } = opts;
   737
   738  const body: Record<string, unknown> = {
   739    model,
   740    messages,
   741    temperature,
   742    max_tokens: maxTokens,
   743  };
   744
   745  const isGemini = providerName === "Gemini";
   746  const hasTools = !isGemini && Array.isArray(tools) && tools.length > 0;
   747  if (hasTools) {
   748    body.tools = tools;
   749    body.tool_choice = opts.toolChoice ?? "auto";
   750  } else if (!isGemini && opts.responseFormat) {
   751    body.response_format = opts.responseFormat;
   752  }
   753
   754  const controller = new AbortController();
   755  const timer = setTimeout(() => controller.abort(), timeoutMs);
   756  const onAbort = () => controller.abort(signal?.reason);
   757  if (signal?.aborted) onAbort();
   758  else signal?.addEventListener("abort", onAbort, { once: true });
   759  const cleanup = () => {
   760    clearTimeout(timer);
   761    signal?.removeEventListener("abort", onAbort);
   762  };
   763
   764  // Capture raw payload for diagnostics before the request is sent.
   765  console.info(
   766    JSON.stringify({
   767      scope: "openai-compatible-client",
   768      action: "pre_fetch",
   769      providerName,
   770      model,
   771      baseUrl,
   772      messageCount: messages.length,
   773      hasTools: hasTools,
   774      toolCount: hasTools && Array.isArray(tools) ? tools.length : 0,
   775      responseFormatRequested: opts.responseFormat?.type ?? null,
   776      responseFormatApplied: !hasTools && opts.responseFormat?.type
   777        ? opts.responseFormat.type
   778        : null,
   779      maxTokens,
   780      temperature,
   781    }),
   782  );
   783
   784  let response: Response;
   785  try {
   786    response = await fetch(`${baseUrl}/chat/completions`, {
   787      method: "POST",
   788      headers: {
   789        "Content-Type": "application/json",
   790        Authorization: `Bearer ${apiKey}`,
   791        ...extraHeaders,
   792      },
   793      body: JSON.stringify(body),
   794      signal: controller.signal,
   795    });
   796  } catch (err) {
   797    cleanup();
   798    if (controller.signal.aborted) {
   799      throw new GroqClientError("TIMEOUT", `${providerName} request timed out`, {
   800        cause: err,
   801        context: { providerName, providerModel: model },
   802      });
   803    }
   804    throw new GroqClientError(
   805      "NETWORK_ERROR",
   806      err instanceof Error ? err.message : `Network error contacting ${providerName}`,
   807      { cause: err, context: { providerName, providerModel: model } },
   808    );
   809  }
   810  if (!response.ok) {
   811    const text = await response.text().catch(() => "");
   812    cleanup();
   813    // Log the raw response before classifying so we never lose context.
   814    console.warn(
   815      JSON.stringify({
   816        scope: "openai-compatible-client",
   817        action: "non_ok_response",
   818        providerName,
   819        model,
   820        status: response.status,
   821        bodyPreview: text.slice(0, 400),
   822      }),
   823    );
   824    throw classifyStatus(response.status, text, providerName, model, response.headers);
   825  }
   826
   827  let data: {
   828    choices: Array<{
   829      finish_reason?: string | null;
   830      message?: {
   831        content?: string | null;
   832        tool_calls?: ToolCall[];
   833        reasoning_content?: string | null;
   834        reasoning?: string | null;
   835      };
   836    }>;
   837    model: string;
   838    output_text?: string | null;
   839    usage?: {
   840      prompt_tokens?: number;
   841      completion_tokens?: number;
   842      reasoning_tokens?: number;
   843      completion_tokens_details?: { reasoning_tokens?: number };
   844    };
   845  };
   846  try {
   847    // Keep the request timer active until the response body has been fully
   848    // consumed. Fetch can resolve headers before a slow provider finishes
   849    // sending the JSON body.
   850    data = (await response.json()) as typeof data;
   851  } catch (err) {
   852    cleanup();
   853    if (controller.signal.aborted) {
   854      throw new GroqClientError("TIMEOUT", `${providerName} response body timed out`, {
   855        cause: err,
   856        context: { providerName, providerModel: model },
   857      });
   858    }
   859    throw err;
   860  }
   861  cleanup();
   862
   863  const choice = data.choices[0];
   864  const msg = choice?.message;
   865  const outputText = data.output_text ?? null;
   866  if (!msg && !outputText) {
   867    throw new GroqClientError("EMPTY_RESPONSE", `${providerName} returned an empty response`, {
   868      context: { providerName, providerModel: model },
   869    });
   870  }
   871
   872  let content = msg?.content ?? outputText;
   873  const reasoningContent = msg?.reasoning_content ?? msg?.reasoning ?? null;
   874  const reasoningTokens =
   875    data.usage?.reasoning_tokens ??
   876    data.usage?.completion_tokens_details?.reasoning_tokens;
   877  let hasCalls = Array.isArray(msg?.tool_calls) && msg.tool_calls.length > 0;
   878  let toolCalls: ToolCall[] | null = hasCalls
   879    ? normalizeToolCallArguments(msg!.tool_calls as ToolCall[], providerName, model)
   880    : null;
   881
   882  // Some free models (e.g. Ling-3.0-flash) embed tool calls as XML text in
   883  // content instead of using the standard tool_calls JSON field.
   884  if (content && content.includes("<tool_call>")) {
   885    if (!hasCalls) {
   886      const xmlCalls = parseXmlToolCalls(content);
   887      if (xmlCalls) {
   888        toolCalls = xmlCalls;
   889        hasCalls = true;
   890      }
   891    }
   892    content = stripXmlToolCalls(content) || null;
   893  }
   894
   895  if (!content && !hasCalls) {
   896    throw new GroqClientError(
   897      "EMPTY_RESPONSE",
   898      `${providerName} returned neither content nor tool calls`,
   899      { context: { providerName, providerModel: model } },
   900    );
   901  }
   902
   903  console.info(
   904    JSON.stringify({
   905      scope: "openai-compatible-client",
   906      action: "response_received",
   907      providerName,
   908      requestedModel: model,
   909      actualModel: data.model,
   910      finishReason: choice?.finish_reason ?? null,
   911      contentLength: content?.length ?? 0,
   912      outputTextLength: outputText?.length ?? 0,
   913      reasoningContentLength: reasoningContent?.length ?? 0,
   914      reasoningTokens: reasoningTokens ?? 0,
   915      hasToolCalls: hasCalls,
   916    }),
   917  );
   918
   919  return {
   920    content,
   921    toolCalls,
   922    model: data.model,
   923    usage: {
   924      promptTokens: data.usage?.prompt_tokens ?? 0,
   925      completionTokens: data.usage?.completion_tokens ?? 0,
   926    },
   927    finishReason: choice?.finish_reason ?? null,
   928    reasoningContent,
   929    outputText,
   930    reasoningTokens,
   931  };
   932}
