    46type GroqModelAvailability = {
    47  status: "available" | "unavailable" | "check_unavailable" | "invalid_credential" | "not_configured";
    48  source: "personal" | "server" | "none";
    49  checkedModels: { fast: string; powerful: string };
    50  unavailableRoles: Array<"fast" | "powerful">;
    51  checkedAt: string;
    52  reason?: string;
    53};

    55/**
    56 * Check Groq's configured defaults without returning a credential or raw
    57 * provider error. A valid key and an available model are separate concerns:
    58 * Groq can accept the key while retiring one of the configured model IDs.
    59 */
    60async function getGroqModelAvailability(
    61  apiKey: string | undefined,
    62  source: GroqModelAvailability["source"],
    63): Promise<GroqModelAvailability> {
    64  const checkedModels = PROVIDER_REGISTRY.groq.defaultModels;
    65  const checkedAt = new Date().toISOString();
    66
    67  if (!apiKey) {
    68    return {
    69      status: source === "none" ? "not_configured" : "check_unavailable",
    70      source,
    71      checkedModels,
    72      unavailableRoles: [],
    73      checkedAt,
    74      ...(source !== "none"
    75        ? { reason: "Groq model availability could not be checked. Save the Groq API key again, then retry." }
    76        : {}),
    77    };
    78  }
    79
    80  try {
    81    const validation = await validateGroqDefaultModels(apiKey, checkedModels);
    82    return {
    83      status: validation.valid ? "available" : "unavailable",
    84      source,
    85      checkedModels: validation.checkedModels,
    86      unavailableRoles: validation.missing,
    87      checkedAt,
    88      ...(validation.reason ? { reason: validation.reason } : {}),
    89    };
    90  } catch (error) {
    91    const isInvalidCredential =
    92      error instanceof GroqClientError && error.code === "AUTH_ERROR";
    93    return {
    94      status: isInvalidCredential ? "invalid_credential" : "check_unavailable",
    95      source,
    96      checkedModels,
    97      unavailableRoles: [],
    98      checkedAt,
    99      reason: isInvalidCredential
   100        ? "Groq rejected this credential while checking the configured models. Replace the Groq API key."
   101        : "Groq model availability could not be confirmed. Retry shortly; if this persists, check Groq status.",
   102    };
   103  }
   104}

   106/** Shared GET handler — return key status (never the key itself). */
   107async function handleGetKey(req: Request, res: Response, provider: ProviderId) {
   108  const [row] = await db
   109    .select({
   110      encryptedApiKey: aiProviderCredentialsTable.encryptedApiKey,
   111      last4: aiProviderCredentialsTable.last4,
   112      updatedAt: aiProviderCredentialsTable.updatedAt,
   113    })
   114    .from(aiProviderCredentialsTable)
   115    .where(
   116      and(
   117        eq(aiProviderCredentialsTable.ownerId, req.userId),
   118        eq(aiProviderCredentialsTable.provider, provider),
   119      ),
   120    )
   121    .limit(1);
   122
   123  if (!row) {
   124    if (provider !== "groq") {
   125      return res.json({ configured: false, last4: null, updatedAt: null });
   126    }
   127    const modelAvailability = await getGroqModelAvailability(
   128      process.env.GROQ_API_KEY,
   129      process.env.GROQ_API_KEY ? "server" : "none",
   130    );
   131    return res.json({
   132      configured: false,
   133      last4: null,
   134      updatedAt: null,
   135      modelAvailability,
   136    });
   137  }
   138
   139  if (provider !== "groq") {
   140    return res.json({ configured: true, last4: row.last4, updatedAt: row.updatedAt });
   141  }
   142
   143  let apiKey: string | undefined;
   144  let source: GroqModelAvailability["source"] = "personal";
   145  try {
   146    apiKey = decryptApiKey(row.encryptedApiKey);
   147  } catch (error) {
   148    logger.error({ err: error, provider }, "Failed to decrypt stored provider key for readiness check");
   149    apiKey = process.env.GROQ_API_KEY;
   150    source = apiKey ? "server" : "none";
   151  }
   152  const modelAvailability = await getGroqModelAvailability(
   153    apiKey,
   154    source,
   155  );
   156  return res.json({
   157    configured: true,
   158    last4: row.last4,
   159    updatedAt: row.updatedAt,
   160    modelAvailability,
   161  });
   162}
