    21function isAllowedApplicationOrigin(origin: string, req: Request): boolean {
    22  let parsed: URL;
    23  try {
    24    parsed = new URL(origin);
    25  } catch {
    26    return false;
    27  }
    28  if (!["http:", "https:"].includes(parsed.protocol) || parsed.username || parsed.password) {
    29    return false;
    30  }
    31
    32  const normalizedOrigin = parsed.origin;
    33  const requestOrigin = `${req.protocol}://${req.get("host")}`;
    34  if (normalizedOrigin === requestOrigin) return true;
    35  if (config.applicationOrigins.includes(normalizedOrigin)) return true;
    36
    37  // The development dashboard may run on a local port or a Replit preview
    38  // domain. These are intentionally not accepted in production.
    39  if (!config.isProduction) {
    40    const hostname = parsed.hostname.toLowerCase();
    41    if (
    42      hostname === "localhost" ||
    43      hostname === "127.0.0.1" ||
    44      hostname === "::1" ||
    45      hostname.endsWith(".replit.dev") ||
    46      hostname.endsWith(".repl.co")
    47    ) {
    48      return true;
    49    }
    50  }
    51  return false;
    52}

    54function rejectCrossOriginMutations(
    55  req: Request,
    56  res: Response,
    57  next: NextFunction,
    58): void {
    59  if (!["POST", "PUT", "PATCH", "DELETE"].includes(req.method)) {
    60    next();
    61    return;
    62  }
    63
    64  const origin = req.get("origin");
    65  const fetchSite = req.get("sec-fetch-site");
    66  const hasCookies = Boolean(req.get("cookie"));
    67  const csrfCookie = req.get("cookie")?.match(/(?:^|;\s*)csrf_token=([^;]+)/)?.[1];
    68  const csrfHeader = req.get("x-csrf-token");
    69  if (
    70    (origin && !isAllowedApplicationOrigin(origin, req)) ||
    71    (!origin && fetchSite === "cross-site")
    72  ) {
    73    res.status(403).json({
    74      error: "Cross-origin state-changing requests are not allowed",
    75      code: "cross_origin_request",
    76    });
    77    return;
    78  }
    79  // Cookie-authenticated requests without browser provenance headers are
    80  // accepted only with a matching double-submit CSRF token. Same-origin
    81  // browser requests retain their normal Fetch Metadata path.
    82  if (hasCookies && !origin && fetchSite !== "same-origin"
    83    && (!csrfCookie || !csrfHeader || csrfCookie !== csrfHeader)) {
    84    res.status(403).json({
    85      error: "CSRF validation failed",
    86      code: "csrf_failed",
    87    });
    88    return;
    89  }
