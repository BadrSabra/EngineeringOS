    32type StructuredTask = "analyze" | "review";
    33type StructuredAuditTraceEntry = {
    34  stage: string;
    35  status: "started" | "completed" | "failed" | "incomplete";
    36  provider?: string;
    37};
    38type StructuredAuditMetadata = {
    39  operationId: string;
    40  projectId: string;
    41  projectRevision: string;
    42  rootAvailable: boolean;
    43  incomplete: boolean;
    44  operationalTrace: StructuredAuditTraceEntry[];
    45};
    46type StructuredTaskEvent =
    47  | ({ type: "task_started"; task: StructuredTask; projectId: string }
    48    | { type: "stage"; stage: string }
    49    | { type: "task_progress"; task: StructuredTask; message: string; provider?: string }
    50    | { type: "task_done"; task: StructuredTask; result: Record<string, unknown> }
    51   | { type: "error"; code: string; message: string; hint?: string; retryable?: boolean; failureKind?: "PROVIDER_FORMAT" | "RATE_LIMIT" | "CONFIGURATION" | "PROVIDER_FAILURE" | "TRANSPORT"; outcome?: "FAILED" | "INTERRUPTED" })
    52    & Partial<StructuredAuditMetadata>;
    53
    54function auditEnvelope(metadata: StructuredAuditMetadata): Record<string, unknown> {
    55  return {
    56    // Correlation identifiers are server diagnostics. Do not expose opaque
    57    // UUIDs in a model-output error response where they can be persisted or
    58    // echoed back to the user.
    59    rootAvailable: metadata.rootAvailable,
    60    incomplete: metadata.incomplete,
    61    operationalTrace: metadata.operationalTrace,
    62  };
    63}
    64
    65function recordTrace(
    66  metadata: StructuredAuditMetadata,
    67  stage: string,
    68  status: StructuredAuditTraceEntry["status"],
    69  provider?: string,
    70): void {
    71  metadata.operationalTrace.push({
    72    stage,
    73    status,
    74    ...(provider ? { provider: redactUserFacingText(provider) } : {}),
    75  });
    76}

    78async function createAuditMetadata(
    79  projectId: string,
    80  project: typeof projectsTable.$inferSelect,
    81): Promise<StructuredAuditMetadata> {
    82  const root = await resolveRootPath(project.rootPath, projectId);
    83  return {
    84    operationId: randomUUID(),
    85    projectId,
    86    projectRevision: project.updatedAt.toISOString(),
    87    rootAvailable: Boolean(root.validRootPath),
    88    incomplete: !root.validRootPath,
    89    operationalTrace: [],
    90  };
    91}

    93function beginTaskStream(
    94  res: import("express").Response,
    95  task: StructuredTask,
    96  projectId: string,
    97  metadata: StructuredAuditMetadata,
    98) {
    99  res.status(200).set({
   100    "Content-Type": "text/event-stream",
   101    "Cache-Control": "no-cache, no-transform",
   102    Connection: "keep-alive",
   103    "X-Accel-Buffering": "no",
   104  });
   105  res.flushHeaders();
   106
   107  let closed = false;
   108  const heartbeat = setInterval(() => {
   109    if (!closed && !res.writableEnded) res.write(": heartbeat\n\n");
   110  }, 15_000);
   130  metadata.operationalTrace.push({ stage: "task", status: "started" });
   131  emit({ type: "task_started", task, projectId });
   132  return { emit, close };
   133}

   135function emitTaskFailure(
   136  emit: (event: StructuredTaskEvent) => void,
   137  close: () => void,
   138  err: unknown,
   139  metadata: StructuredAuditMetadata,
   140) {
   141  const candidate = err as { code?: unknown; message?: unknown };
   142  const code = typeof candidate.code === "string" ? candidate.code : "task_failed";
   143  const failureKind =
   144    code === "model_output_invalid" || code === "INVALID_MODEL_OUTPUT" ? "PROVIDER_FORMAT" :
   145    code === "EMPTY_RESPONSE" ? "PROVIDER_FORMAT" :
   146    code === "RATE_LIMITED" ? "RATE_LIMIT" :
   147    code === "INVALID_CONFIG" || code === "AUTH_ERROR" || code === "MODEL_NOT_FOUND" || code === "PLAN_RESTRICTED" ? "CONFIGURATION" :
   148    code === "TIMEOUT" || code === "NETWORK_ERROR" || code === "NON_200" || code === "SERVER_ERROR" ? "PROVIDER_FAILURE" :
   149    "PROVIDER_FAILURE";
   150  const message =
   151    failureKind === "RATE_LIMIT" ? "The AI provider is rate-limited. Please wait before retrying." :
   152    failureKind === "CONFIGURATION" ? "The AI provider configuration needs attention before this can run." :
   153    failureKind === "PROVIDER_FORMAT" ? "The AI returned an unexpected response format." :
   154    "The AI provider could not complete this run.";
   155  metadata.incomplete = true;
   156  metadata.operationalTrace.push({ stage: "failed", status: "failed" });
   157  emit({
   158    type: "error",
   159    code,
   160    message,
   161    hint: failureKind === "RATE_LIMIT"
   162      ? "Wait a moment and retry the analysis."
   163      : failureKind === "CONFIGURATION"
   164        ? "Check the configured provider key or choose another provider."
   165        : "You can retry this task without sending another prompt.",
   166    retryable: failureKind !== "CONFIGURATION",
   167    failureKind,
   168    outcome: "FAILED",
   169  });
   170  close();
   171}

   175router.post("/ai/projects/:projectId/analyze", requireProjectAccess, async (req, res) => {
   176  const projectId = req.params.projectId as string;
   177  const project = req.project;
   178  if (!project) return res.status(500).json({ error: "Project context unavailable" });
   179  const metadata = await createAuditMetadata(projectId, project);
   180  recordTrace(metadata, "analyze", "started");
   181
   182  logger.info({ projectId }, "AI scan analysis requested");
   183
   184  const providerResolved = await requireProvider(req.userId, res, {
   185    qualityProfile: "analysis",
   186  });
   187  if (!providerResolved) return;
   188  const { provider, apiKey } = providerResolved;
   189
   190  const projectContext = await buildProjectContext(projectId, {
   191    sections: ["tasks", "metrics", "graphEntities", "graphRelationships", "events"],
   192  });
   193
   194  const rlAnalyze = await checkProjectRateLimitDb(projectId);
   195  if (!rlAnalyze.allowed) {
   196    return res.status(429).json({
   197      error: "The AI provider is temporarily rate-limited.",
   198      code: "RATE_LIMITED",
   199      hint: `Wait ${rlAnalyze.retryAfterSec}s, then retry the analysis.`,
   200      retryable: true,
   201      failureKind: "RATE_LIMIT",
   202    });
   203  }
   204
   205  let result: Awaited<ReturnType<typeof analyzeScan>>;
   206  let effectiveProvider = provider;
   207  try {
   208    ({ result, effectiveProvider } = await runAgentWithFallback(

   379router.post("/ai/projects/:projectId/analyze/stream", requireProjectAccess, async (req, res) => {
   380  const projectId = req.params.projectId as string;
   381  const project = req.project;
   382  if (!project) return res.status(500).json({ error: "Project context unavailable" });
   383  const providerResolved = await requireProvider(req.userId, res, { qualityProfile: "analysis" });
   384  if (!providerResolved) return;
   385  const metadata = await createAuditMetadata(projectId, project);
   386  recordTrace(metadata, "analyze", "started");
   387
   388  const rlAnalyze = await checkProjectRateLimitDb(projectId);
   389  if (!rlAnalyze.allowed) {
   390    return res.status(429).json({
   391      error: `LLM rate limit exceeded — max ${LLM_RATE_LIMIT} calls per minute per project. Retry in ${rlAnalyze.retryAfterSec}s.`,
   392    });
   393  }
   394
   395  const { emit, close } = beginTaskStream(res, "analyze", projectId, metadata);
   396  try {
   397    recordTrace(metadata, "building-context", "started");
   398    emit({ type: "stage", stage: "building-context" });
   399    const projectContext = await buildProjectContext(projectId, {
   400      sections: ["tasks", "metrics", "graphEntities", "graphRelationships", "events"],
   401    });
   402
   403    emit({ type: "stage", stage: "calling-model" });
   404    recordTrace(metadata, "calling-model", "started", providerResolved.provider);
   405    const { provider, apiKey } = providerResolved;
   406    let effectiveProvider = provider;
   407    const { result } = await runAgentWithFallback(
   408      req.userId,
   409      { provider, apiKey },
   410      (opts) => analyzeScan(projectContext, {
   411        ...opts,
   412        onProgress: (message) => emit({
   413          type: "task_progress",
   414          task: "analyze",
   415          message,
   416          provider: opts.provider,
   417        }),
   418      }),
   419      { qualityProfile: "analysis" },
   420    ).then((output) => {
   421      effectiveProvider = output.effectiveProvider;
   422      return output;
   423    });
   424
   425    if (result._parseError) {
   426      metadata.incomplete = true;
   427      recordTrace(metadata, "calling-model", "incomplete", effectiveProvider);
   428      emit({
   429        type: "error",
   430        code: "model_output_invalid",
   431        message: "The AI model returned an unexpected response.",
   432        hint: "The response could not be verified as a structured result. You can retry this task.",
   433        retryable: true,
   434        failureKind: "PROVIDER_FORMAT",
   435        outcome: "FAILED",
   436      });
   437      close();
   438      return;
   439    }
   440
   441    emit({ type: "stage", stage: "persisting-result" });
   442    recordTrace(metadata, "persisting-result", "completed", effectiveProvider);
   443    invalidateContextCache(projectId);
   444    await db.transaction(async (tx) => {
   445      await tx.insert(auditLogsTable).values({
   446        id: randomUUID(),
   447        entityType: "project",
   448        entityId: projectId,
   449        action: "ai_analyzed",
   450        projectId,
   451        actor: req.userId,
   452        stateBefore: {},
   453        stateAfter: { summary: result.summary, overallAssessment: result.overallAssessment },
   454      });
   455      await tx.insert(eventsTable).values({
   456        id: randomUUID(),
   457        type: "AiScanAnalysisCompleted",
   458        projectId,
   459        severity: "info",
   460        message: `AI scan analysis completed: ${result.summary}`,
   461      });
   462    });
   463
   464    emit({ type: "stage", stage: "completed" });
   465    recordTrace(metadata, "analyze", "completed", effectiveProvider);
   466    emit({
   467      type: "task_done",
   468      task: "analyze",
   469      result: {
   470        ...redactUserFacingValue(result) as Record<string, unknown>,
   471        ...auditEnvelope(metadata),
   472      },
   473    });
   474    close();
   475    logger.info({ projectId, provider: effectiveProvider }, "AI scan analysis stream completed");
   476  } catch (err) {
   477    logger.error({ err, projectId }, "AI scan analysis stream failed");
   478    emitTaskFailure(emit, close, err, metadata);
   479  }
   480  return;
   481});
