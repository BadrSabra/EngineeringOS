     1/**
     2 * Unified error model for the ai-orchestrator package.
     3 *
     4 * Two families:
     5 *
     6 * AgentErrorCode — non-fatal issues turning a model response into a typed
     7 * result (parsing/validation). Agents never throw these; they log the code
     8 * and fall back to a safe default so a single bad LLM response cannot crash
     9 * a request. See `parsing.ts`.
    10 *
    11 *   EMPTY_MODEL_RESPONSE      Raw text from the model was empty.
    12 *   MALFORMED_JSON            Raw text could not be parsed as JSON.
    13 *   SCHEMA_VALIDATION_FAILED  Parsed JSON did not satisfy the Zod schema.
    14 *
    15 * GroqErrorCode — failures talking to the model provider. These DO throw,
    16 * because there is no safe per-agent fallback for "we could not reach the
    17 * model at all". Routes handle them as hard failures.
    18 *
    19 * Transport
    20 *   TIMEOUT           AbortController fired before the model responded.
    21 *   NETWORK_ERROR     Transport-level failure: DNS, ECONNRESET, no HTTP status.
    22 *
    23 * HTTP — classified from the numeric status code
    24 *   AUTH_ERROR        401 or 403: API key missing, invalid, or lacks permission.
    25 *   RATE_LIMITED      429: per-minute quota exhausted; retryable after back-off.
    26 *   QUOTA             402/429-credits: billing quota / credits exhausted.
    27 *   SERVER_ERROR      5xx: provider infrastructure failure; retryable.
    28 *   NON_200           Any other non-200 status (e.g. 400 bad request).
    29 *
    30 * Model availability
    31 *   MODEL_NOT_FOUND    404: model permanently discontinued.
    32 *   PLAN_RESTRICTED    402: model requires paid tier / free-tier restriction.
    33 *   MODEL_UNAVAILABLE  422/410: model temporarily offline or being retired.
    34 *
    35 * Response
    36 *   EMPTY_RESPONSE    Response received but contained no content or tool calls.
    37 *
    38 * Configuration
    39 *   INVALID_CONFIG    GROQ_API_KEY is absent or empty at startup.
    40 */
    41
    42export type AgentErrorCode = "EMPTY_MODEL_RESPONSE" | "MALFORMED_JSON" | "SCHEMA_VALIDATION_FAILED";
    43
    44export type GroqErrorCode =
    45  // transport
    46  | "TIMEOUT"
    47  | "NETWORK_ERROR"
    48  // HTTP — split by semantics
    49  | "AUTH_ERROR"
    50  | "RATE_LIMITED"
    51  | "QUOTA"             // billing quota / credits exhausted (PR-008)
    52  | "SERVER_ERROR"
    53  | "NON_200"
    54  // model availability — distinct from each other (PR-004 / PR-008)
    55  | "MODEL_NOT_FOUND"     // 404 — model permanently discontinued
    56  | "PLAN_RESTRICTED"     // 402 — model requires paid plan / free-tier restriction
    57  | "MODEL_UNAVAILABLE"   // 422/410 — model temporarily offline (PR-008)
    58  // response
    59  | "EMPTY_RESPONSE"
    60  // configuration
    61  | "INVALID_CONFIG";

    63/** PR-007: structured fields carried alongside the error message. */
    64export type ProviderErrorContext = {
    65  /** Raw HTTP status code from the provider. */
    66  providerStatus?: number;
    67  /** Provider-specific error code extracted from the response body. */
    68  providerCode?: string;
    69  /** Provider-specific error message extracted from the response body. */
    70  providerMessage?: string;
    71  /** Provider name (e.g. "OpenRouter", "Groq"). */
    72  providerName?: string;
    73  /** Model slug that was attempted when the error occurred. */
    74  providerModel?: string;
    75  /** Bounded provider retry hint, in milliseconds, when supplied by HTTP. */
    76  retryAfterMs?: number;
    77  /** Every provider-owned fallback model attempted before this error surfaced. */
    78  providerAttemptedModels?: string[];
    79  /** Safe OpenRouter catalog diagnostics for model-resolution failures. */
    80  catalogLoaded?: boolean;
    81  catalogUsable?: boolean;
    82  catalogStatus?: "never" | "success" | "failed" | "empty";
    83  catalogError?: string;
    84};

    86/**
    87 * Provider SDKs and HTTP clients sometimes include credentials in otherwise
    88 * useful transport errors (for example, a URL query string or an echoed
    89 * Authorization header). Keep the internal error available for server logs,
    90 * but make every message/context serialization safe for user-facing output.
    91 */
    92export function redactProviderErrorText(value: string): string {
    93  return value
    94    .replace(/((?:authorization|x-api-key|api[_-]?key|access[_-]?token|token|key)\s*[=:]\s*)([^\s,;)"']+)/gi, "$1[redacted]")
    95    .replace(/(Bearer\s+)([A-Za-z0-9._~+/=-]+)/gi, "$1[redacted]")
    96    .replace(/\bsk-or-v1-[A-Za-z0-9_-]+\b/gi, "[redacted]")
    97    .replace(/\bAIza[A-Za-z0-9_-]{20,}\b/g, "[redacted]");
    98}
    99
   100function redactProviderContext(context: ProviderErrorContext): ProviderErrorContext {
   101  return {
   102    ...context,
   103    providerCode: context.providerCode ? redactProviderErrorText(context.providerCode) : context.providerCode,
   104    providerMessage: context.providerMessage ? redactProviderErrorText(context.providerMessage) : context.providerMessage,
   105    providerName: context.providerName ? redactProviderErrorText(context.providerName) : context.providerName,
   106    providerModel: context.providerModel ? redactProviderErrorText(context.providerModel) : context.providerModel,
   107    providerAttemptedModels: context.providerAttemptedModels?.map(redactProviderErrorText),
   108    catalogLoaded: context.catalogLoaded,
   109    catalogUsable: context.catalogUsable,
   110    catalogStatus: context.catalogStatus,
   111  };
   112}

   114export class GroqClientError extends Error {
   115  readonly code: GroqErrorCode;
   116
   117  // PR-007: structured provider context preserved all the way to the caller
   118  readonly providerStatus?: number;
   119  readonly providerCode?: string;
   120  readonly providerMessage?: string;
   121  readonly providerName?: string;
   122  readonly providerModel?: string;
   123  readonly retryAfterMs?: number;
   124  readonly providerAttemptedModels?: string[];
   125  readonly catalogLoaded?: boolean;
   126  readonly catalogUsable?: boolean;
   127  readonly catalogStatus?: ProviderErrorContext["catalogStatus"];
   128  readonly catalogError?: string;
   129
   130  constructor(
   131    code: GroqErrorCode,
   132    message: string,
   133    options?: { cause?: unknown; context?: ProviderErrorContext },
   134  ) {
   135    super(redactProviderErrorText(message), { cause: options?.cause });
   136    this.name = "GroqClientError";
   137    this.code = code;
   138    if (options?.context) {
   139      this.providerStatus  = options.context.providerStatus;
   140      this.providerCode    = options.context.providerCode;
   141      this.providerMessage = options.context.providerMessage;
   142      this.providerName    = options.context.providerName;
   143      this.providerModel   = options.context.providerModel;
   144      this.retryAfterMs    = options.context.retryAfterMs;
   145      this.providerAttemptedModels = options.context.providerAttemptedModels;
   146      this.catalogLoaded = options.context.catalogLoaded;
   147      this.catalogUsable = options.context.catalogUsable;
   148      this.catalogStatus = options.context.catalogStatus;
   149      this.catalogError = options.context.catalogError;
   150    }
   151  }
   152
   153  /** PR-007: serialise to a plain object suitable for SSE / JSON responses. */
   154  toProviderContext(): ProviderErrorContext {
   155    return redactProviderContext({
   156      providerStatus:  this.providerStatus,
   157      providerCode:    this.providerCode,
   158      providerMessage: this.providerMessage,
   159      providerName:    this.providerName,
   160      providerModel:   this.providerModel,
   161      retryAfterMs:    this.retryAfterMs,
   162      providerAttemptedModels: this.providerAttemptedModels,
   163      catalogLoaded: this.catalogLoaded,
   164      catalogUsable: this.catalogUsable,
   165      catalogStatus: this.catalogStatus,
   166    });
   167  }
