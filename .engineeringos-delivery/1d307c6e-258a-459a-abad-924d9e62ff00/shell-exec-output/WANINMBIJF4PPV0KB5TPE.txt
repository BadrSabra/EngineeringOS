    60  }
    61
    62  async run(input: TInput, opts?: AgentCompleteOpts): Promise<AgentRunResult<TOutput>> {
    63    const messages = this.buildMessages(input);
    64    const executionPlan = resolveExecutionDecision(this.scope, {
    65      qualityProfile: opts?.qualityProfile ?? this.buildQualityProfile(input),
    66    });
    67    const qualityProfile = opts?.qualityProfile ?? this.buildQualityProfile(input) ?? executionPlan.qualityProfile;
    68    const parseResponse = (raw: string) => parseAgentResponse(raw, this.schema, (fallbackRaw) => this.fallbackOutput(fallbackRaw));
    69
    70    opts?.onProgress?.("Calling AI model…");
    71    const response = await this.complete(messages, {
    72      ...this.buildCompleteOpts(input),
    73      ...(opts ?? {}),
    74      qualityProfile,
    75      qualityHints: opts?.qualityHints ?? executionPlan.strictHints,
    76    });
    77
    78    let parsed = parseResponse(response.content);
    79    let parseError = parsed.ok ? undefined : { code: parsed.code, message: parsed.message, raw: parsed.raw };
    80    const initialAssessment = parsed.ok && qualityProfile ? assessStructuredOutput(qualityProfile, parsed.data) : undefined;
    81    let attempt = 1;
    82    let assessment = initialAssessment;
    83    while (attempt < executionPlan.retryLimit) {
    84      const retryDecision = decideRetry({
    85        attempt,
    86        limit: executionPlan.retryLimit,
    87        parseError,
    88        assessment,
    89      });
    90      if (!retryDecision.shouldRetry) break;
    91      console.warn(
    92        JSON.stringify({
    93          scope: this.scope,
    94          code: "MODEL_RETRY",
    95          reason: retryDecision.reason,
    96          qualityProfile,
    97        }),
    98      );
    99
   100      opts?.onProgress?.("Retrying — improving output quality…");
   101      const retryResponse = await this.complete(messages, {
   102        ...this.buildCompleteOpts(input),
   103        ...(opts ?? {}),
   104        qualityProfile,
   105        qualityHints: executionPlan.relaxedHints,
   106      });
   107
   108      parsed = parseResponse(retryResponse.content);
   109      parseError = parsed.ok ? undefined : { code: parsed.code, message: parsed.message, raw: parsed.raw };
   110      assessment = parsed.ok && qualityProfile ? assessStructuredOutput(qualityProfile, parsed.data) : undefined;
   111      attempt += 1;
   112    }
   113
   114    if (!parsed.ok) {
   115      console.warn(JSON.stringify({ scope: this.scope, code: parsed.code, message: parsed.message }));
   116      return { ...parsed.data, _parseError: parseError };
   117    }
   118
   119    assessment = qualityProfile ? assessStructuredOutput(qualityProfile, parsed.data) : undefined;
   120    if (assessment && assessment.decision !== "accept") {
   121      console.warn(
   122        JSON.stringify({
   123          scope: this.scope,
   124          code: "QUALITY_REVIEW_LOW",
   125          qualityProfile,
   126          score: assessment.score,
   127          threshold: assessment.threshold,
   128          reasons: assessment.reasons,
   129        }),
   130      );
   131    }
   132
   133    // parsed.data is TOutput, which satisfies AgentRunResult<TOutput> (= T & { _parseError? })
   134    // because _parseError is optional. Cast is required because TypeScript cannot
   135    // prove T satisfies T & { _parseError?: … } for an unconstrained generic.
   136    return parsed.data as AgentRunResult<TOutput>;
   137  }
