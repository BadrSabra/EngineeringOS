     1/**
     2 * Unified error model for the ai-orchestrator package.
     3 *
     4 * Two families:
     5 *
     6 * AgentErrorCode — non-fatal issues turning a model response into a typed
     7 * result (parsing/validation). Agents never throw these; they log the code
     8 * and fall back to a safe default so a single bad LLM response cannot crash
     9 * a request. See `parsing.ts`.
    10 *
    11 *   EMPTY_MODEL_RESPONSE      Raw text from the model was empty.
    12 *   MALFORMED_JSON            Raw text could not be parsed as JSON.
    13 *   SCHEMA_VALIDATION_FAILED  Parsed JSON did not satisfy the Zod schema.
    14 *
    15 * GroqErrorCode — failures talking to the model provider. These DO throw,
    16 * because there is no safe per-agent fallback for "we could not reach the
    17 * model at all". Routes handle them as hard failures.
    18 *
    19 * Transport
    20 *   TIMEOUT           AbortController fired before the model responded.
    21 *   NETWORK_ERROR     Transport-level failure: DNS, ECONNRESET, no HTTP status.
    22 *
    23 * HTTP — classified from the numeric status code
    24 *   AUTH_ERROR        401 or 403: API key missing, invalid, or lacks permission.
    25 *   RATE_LIMITED      429: per-minute quota exhausted; retryable after back-off.
    26 *   QUOTA             402/429-credits: billing quota / credits exhausted.
    27 *   SERVER_ERROR      5xx: provider infrastructure failure; retryable.
    28 *   NON_200           Any other non-200 status (e.g. 400 bad request).
    29 *
    30 * Model availability
    31 *   MODEL_NOT_FOUND    404: model permanently discontinued.
    32 *   PLAN_RESTRICTED    402: model requires paid tier / free-tier restriction.
    33 *   MODEL_UNAVAILABLE  422/410: model temporarily offline or being retired.
    34 *
    35 * Response
    36 *   EMPTY_RESPONSE    Response received but contained no content or tool calls.
    37 *
    38 * Configuration
    39 *   INVALID_CONFIG    GROQ_API_KEY is absent or empty at startup.

    55export function redactUserFacingValue<T>(value: T): T {
    56  if (typeof value === "string") return redactUserFacingText(value) as T;
    57  if (Array.isArray(value)) return value.map((item) => redactUserFacingValue(item)) as T;
    58  if (value && typeof value === "object") {
    59    return Object.fromEntries(
    60      Object.entries(value).map(([key, item]) => [key, redactUserFacingValue(item)]),
    61    ) as T;
    62  }
    63  return value;

    83): Promise<Array<{ provider: ProviderId; apiKey: string }>> {
    84  const available: Array<{ provider: ProviderId; apiKey: string }> = [];
    85  const skipped: Array<{ provider: ProviderId; reason: string }> = [];
    86
    87  for (const provider of PROVIDER_PRIORITY) {
    88    const key = await resolveProviderKey(userId, provider);
    89    if (!key) {
    90      skipped.push({ provider, reason: "no_api_key" });
    91      continue;
    92    }
    93    if (!providerCanHandleRequest(provider, options)) {
    94      const reason = options?.requireTools ? "no_tool_support" : "capability_mismatch";
    95      logger.info(
    96        { provider, requireTools: !!options?.requireTools, reason },
    97        "Skipping provider that cannot satisfy the current request",
    98      );
    99      skipped.push({ provider, reason });
   100      continue;
   101    }
   102    // PR-07: skip providers whose circuit breaker is open (cooldown not elapsed).
   103    if (isCircuitOpen(provider)) {
   104      logger.warn(
   105        { provider, reason: "circuit_open" },
   106        "Skipping provider — circuit is open due to consecutive failures",
   107      );
   108      skipped.push({ provider, reason: "circuit_open" });
   109      continue;
   110    }
   111    available.push({ provider, apiKey: key });
   112  }
   113
   114  if (options?.qualityProfile && available.length > 1) {
   115    const orderedIds = sortProviderIdsByQuality(
   116      available.map((candidate) => candidate.provider),
   117      options.qualityProfile,
   118      { requireTools: options.requireTools },
   119    );
   120    const byProvider = new Map(available.map((candidate) => [candidate.provider, candidate.apiKey]));
   121    const reordered = orderedIds
   122      .map((provider) => ({ provider, apiKey: byProvider.get(provider) }))
   123      .filter((candidate): candidate is { provider: ProviderId; apiKey: string } => Boolean(candidate?.apiKey));
   124
   125    logger.info(
   126      {
   127        scope: "provider-selection",
   128        action: "collect_available_providers",
   129        qualityProfile: options.qualityProfile,
   130        requireTools: !!options?.requireTools,
   131        available: reordered.map((c) => c.provider),
   132        skipped,
   133        finalOrder: reordered.map((c) => c.provider),
   134      },
   135      "provider selection complete (quality-ordered)",
   136    );
   137    return reordered;
   138  }
   139
   140  logger.info(
   141    {
   142      scope: "provider-selection",
   143      action: "collect_available_providers",
   144      qualityProfile: options?.qualityProfile ?? null,
   145      requireTools: !!options?.requireTools,
   146      available: available.map((c) => c.provider),
   147      skipped,
   148    },
   149    "provider selection complete",
   150  );
   151  return available;
   152}

   156/**
   157 * Generic: resolve the saved API key for any provider.
   158 *
   159 * Resolution order:
   160 *   1. User's own saved key (decrypted from DB).
   161 *   2. Server-wide env fallback: OPENROUTER_API_KEY / DEEPSEEK_API_KEY / GROQ_API_KEY.
   162 *   3. undefined — caller MUST return 428 to the client.
   163 */
   164export async function resolveProviderKey(
   165  userId: string,
   166  provider: ProviderId,
   167): Promise<string | undefined> {
   168  const [row] = await db
   169    .select()
   170    .from(aiProviderCredentialsTable)
   171    .where(
   172      and(
   173        eq(aiProviderCredentialsTable.ownerId, userId),
   174        eq(aiProviderCredentialsTable.provider, provider),
   175      ),
   176    )
   177    .limit(1);
   178
   179  if (row) {
   180    try {
   181      return decryptApiKey(row.encryptedApiKey);
   182    } catch (err) {
   183      const label = PROVIDER_REGISTRY[provider]?.label ?? provider;
   184      logger.error(
   185        { err, ownerId: userId, provider },
   186        `Failed to decrypt stored ${label} API key — falling back to env`,
   187      );
   188    }
   189  }
   190
   191  // Server-wide env fallback — supported for all providers via dedicated env vars.
   192  const ENV_FALLBACK: Record<ProviderId, string | undefined> = {
   193    openrouter: process.env.OPENROUTER_API_KEY,
   194    gemini: process.env.GEMINI_API_KEY,
   195    deepseek: process.env.DEEPSEEK_API_KEY,
   196    groq: process.env.GROQ_API_KEY,
   197  };
   198  return ENV_FALLBACK[provider] || undefined;
   199}

   300/**
   301 * Run any single-shot agent function with automatic provider fallback.
   302 *
   303 * Accepts a `run` closure that receives `{ provider, apiKey }` and returns a
   304 * result.  On any error code in FALLBACK_TRIGGER_CODES the next provider in
   305 * PROVIDER_PRIORITY order is tried.  Use this for scan-analysis, code-review,
   306 * task-execution — any route that calls an agent but doesn't use chat streaming.
   307 */
   308export async function runAgentWithFallback<T>(
   309  userId: string,
   310  initialProvider: { provider: ProviderId; apiKey: string },
   311  run: (opts: { provider: ProviderId; apiKey: string; signal?: AbortSignal }) => Promise<T>,
   312  options?: ProviderSelectionOptions & { signal?: AbortSignal },
   313): Promise<{ result: T; effectiveProvider: ProviderId }> {
   314  const orderedProviders = await collectAvailableProviders(userId, options);
   315  if (!orderedProviders.some((candidate) => candidate.provider === initialProvider.provider)) {
   316    orderedProviders.unshift(initialProvider);
   317  }
   318
   319  let lastErr: GroqClientError | undefined;
   320
   321  for (const providerEntry of orderedProviders) {
   322    if (options?.signal?.aborted) {
   323      throw Object.assign(new Error("Execution cancelled"), { name: "AbortError" });
   324    }
   325    if (lastErr) {
   326      logger.info(
   327        { primary: initialProvider.provider, fallback: providerEntry.provider, errorCode: lastErr.code },
   328        "primary provider error; retrying agent with fallback provider",
   329      );
   330    }
   331    try {
   332      const result = await run({ ...providerEntry, signal: options?.signal });
   333      return { result, effectiveProvider: providerEntry.provider };
   334    } catch (err) {
   335      if (options?.signal?.aborted) {
   336        throw Object.assign(new Error("Execution cancelled"), { name: "AbortError", cause: err });
   337      }
   338      if (err instanceof GroqClientError && FALLBACK_TRIGGER_CODES.has(err.code)) {
   339        logger.warn(
   340          { provider: providerEntry.provider, code: err.code, message: err.message },
   341          "provider failed — trying next in chain",
   342        );
   343        lastErr = err;
   344        continue;
   345      }
   346      throw err;
   347    }
   348  }
   349
   350  throw lastErr ?? new GroqClientError("EMPTY_RESPONSE", "No AI provider returned a response");
   351}
