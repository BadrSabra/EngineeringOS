   253function emitTaskFailure(
   254  emit: (event: StructuredTaskEvent) => void,
   255  close: () => void,
   256  err: unknown,
   257  metadata: StructuredAuditMetadata,
   258  params: { projectId: string; userId: string; task: StructuredTask; sessionId?: string },
   259) {
   260  const candidate = err as { code?: unknown; message?: unknown };
   261  const code = typeof candidate.code === "string" ? candidate.code : "task_failed";
   262  const failureKind =
   263    code === "model_output_invalid" || code === "INVALID_MODEL_OUTPUT" ? "PROVIDER_FORMAT" :
   264    code === "EMPTY_RESPONSE" ? "PROVIDER_FORMAT" :
   265    code === "RATE_LIMITED" ? "RATE_LIMIT" :
   266    code === "INVALID_CONFIG" || code === "AUTH_ERROR" || code === "MODEL_NOT_FOUND" || code === "PLAN_RESTRICTED" ? "CONFIGURATION" :
   267    code === "TIMEOUT" || code === "NETWORK_ERROR" || code === "NON_200" || code === "SERVER_ERROR" ? "PROVIDER_FAILURE" :
   268    "PROVIDER_FAILURE";
   269  const message =
   270    failureKind === "RATE_LIMIT" ? "The AI provider is rate-limited. Please wait before retrying." :
   271    failureKind === "CONFIGURATION" ? "The AI provider configuration needs attention before this can run." :
   272    failureKind === "PROVIDER_FORMAT" ? "The AI returned an unexpected response format." :
   273    "The AI provider could not complete this run.";
   274  metadata.incomplete = true;
   275  metadata.operationalTrace.push({ stage: "failed", status: "failed" });
   276  void persistStructuredFailure({
   277    ...params,
   278    failureKind,
   279    retryable: failureKind !== "CONFIGURATION",
   280    errorCode: code,
   281    errorMessage: message,
   282  }).then((sessionId) => {
   283    emit({
   284      type: "error",
   285      code,
   286      message,
   287      hint: failureKind === "RATE_LIMIT"
   288        ? "Wait a moment and retry the task."
   289        : failureKind === "CONFIGURATION"
   290          ? "Update the AI setup before starting a new task."
   291          : "You can retry this task without sending another prompt.",
   292      retryable: failureKind !== "CONFIGURATION",
   293      failureKind,
   294      outcome: "FAILED",
   295      sessionId,
   296    });
   297    close();
   298  }).catch((persistError) => {
   299    logger.error({ persistError, projectId: params.projectId, task: params.task }, "structured task failure persistence failed");
   300    emit({
   301      type: "error",
   302      code,
   303      message,
   304      hint: "You can retry this task without sending another prompt.",
   305      retryable: failureKind !== "CONFIGURATION",
   306      failureKind,
   307      outcome: "FAILED",
   308    });
   309    close();
   310  });
   311}
