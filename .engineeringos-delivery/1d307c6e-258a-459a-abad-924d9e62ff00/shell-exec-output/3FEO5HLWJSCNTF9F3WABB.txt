   136export function getAiReleaseChecks(options: {
   137  enablePreview?: boolean;
   138  enableLiveProvider?: boolean;
   139} = {}): AiReleaseCheckDefinition[] {
   140  const enablePreview = options.enablePreview === true;
   141  const enableLiveProvider = options.enableLiveProvider === true;
   142  return CHECKS.map((check) => ({
   143    ...check,
   144    enabled: check.kind !== "preview" || enablePreview,
   145  })).concat(enableLiveProvider ? [{
   146    id: "live-provider-quality",
   147    kind: "benchmark" as const,
   148    command: "pnpm --filter @workspace/api-server run benchmark:witness",
   149    blocking: false,
   150    enabled: true,
   151    coverage: ["optional live-provider observation"],
   152  }] : []);
   153}

   155function failureCode(check: AiReleaseCheckDefinition, exitCode: number | null, signal: NodeJS.Signals | null): string {
   156  if (signal) return `${check.id.toUpperCase().replaceAll("-", "_")}_TERMINATED`;
   157  return `${check.id.toUpperCase().replaceAll("-", "_")}_FAILED_${exitCode ?? "UNKNOWN"}`;
   158}

   160export function evaluateAiReleaseQuality(results: readonly AiReleaseCheckResult[], options: {
   161  enableLiveProvider?: boolean;
   162  enablePreview?: boolean;
   163  generatedAt?: string;
   164} = {}): AiReleaseQualityDecision {
   165  const blockers = results
   166    .filter((result) => result.status === "failed" && result.blocking)
   167    .map((result) => result.failureCode ?? `${result.id.toUpperCase().replaceAll("-", "_")}_FAILED`);
   168  const failed = results.filter((result) => result.status === "failed");
   169  return {
   170    kind: "ai-release-quality-decision",
   171    version: AI_RELEASE_QUALITY_GATE_VERSION,
   172    generatedAt: options.generatedAt ?? new Date().toISOString(),
   173    status: blockers.length === 0 ? "passed" : "blocked",
   174    liveProviderChecks: options.enableLiveProvider ? "enabled" : "disabled",
   175    previewChecks: options.enablePreview ? "enabled" : "disabled",
   176    summary: {
   177      totalCases: results.length,
   178      passedCases: results.filter((result) => result.status === "passed").length,
   179      failedCases: failed.length,
   180      skippedCases: results.filter((result) => result.status === "skipped").length,
   181      blockingFailures: blockers.length,
   182      informationalFailures: failed.filter((result) => !result.blocking).length,
   183    },
   184    blockers: [...new Set(blockers)],
   185    checks: [...results],
   186  };
   187}

   189async function runCommand(check: AiReleaseCheckDefinition, cwd: string): Promise<AiReleaseCheckResult> {
   190  const started = Date.now();
   191  return await new Promise((resolve) => {
   192    // Output is deliberately drained but never included in the report. This
   193    // keeps prompts, model output, source snippets, and credentials out of
   194    // persisted release artifacts.
   195    const child = spawn("sh", ["-c", check.command], {
   196      cwd,
   197      env: { ...process.env, RUN_CONTROLLED_RELEASE_VALIDATION: "1" },
   198      stdio: ["ignore", "ignore", "ignore"],
   199    });
   200    child.once("error", () => resolve({
   201      ...check,
   202      status: "failed",
   203      failureCode: failureCode(check, null, null),
   204      durationMs: Date.now() - started,
   205    }));
   206    child.once("exit", (exitCode, signal) => resolve({
   207      ...check,
   208      status: exitCode === 0 ? "passed" : "failed",
   209      ...(exitCode === 0 ? {} : { failureCode: failureCode(check, exitCode, signal) }),
   210      durationMs: Date.now() - started,
   211    }));
   212  });
   213}

   215export async function runAiReleaseQualityGate(options: {
   216  cwd?: string;
   217  enablePreview?: boolean;
   218  enableLiveProvider?: boolean;
   219  generatedAt?: string;
   220} = {}): Promise<AiReleaseQualityDecision> {
   221  const enablePreview = options.enablePreview === true;
   222  const enableLiveProvider = options.enableLiveProvider === true;
   223  const checks = getAiReleaseChecks({ enablePreview, enableLiveProvider });
   224  const results: AiReleaseCheckResult[] = [];
   225  for (const check of checks) {
   226    if (!check.enabled) {
   227      results.push({ ...check, status: "skipped", durationMs: 0 });
   228      continue;
   229    }
   230    results.push(await runCommand(check, options.cwd ?? path.resolve(process.cwd(), "../..")));
   231  }
   232  return evaluateAiReleaseQuality(results, { enableLiveProvider, enablePreview, generatedAt: options.generatedAt });
   233}

   235export async function writeAiReleaseQualityDecision(filePath: string, decision: AiReleaseQualityDecision): Promise<void> {
   236  const temporaryPath = `${filePath}.tmp-${process.pid}`;
   237  await fs.mkdir(path.dirname(filePath), { recursive: true });
   238  await fs.writeFile(temporaryPath, `${JSON.stringify(decision, null, 2)}\n`, "utf8");
   239  await fs.rename(temporaryPath, filePath);
   240}
   241
   242if (import.meta.url === `file://${process.argv[1]}`) {
   243  const workspaceRoot = path.resolve(process.cwd(), "../..");
   244  const decision = await runAiReleaseQualityGate({
   245    enablePreview: process.env.AI_RELEASE_ENABLE_PREVIEW === "true",
   246    enableLiveProvider: process.env.AI_RELEASE_ENABLE_LIVE_PROVIDER === "true",
   247    cwd: workspaceRoot,
   248  });
   249  const outputPath = path.resolve(
   250    process.env.AI_RELEASE_QUALITY_REPORT_PATH ??
   251      path.join(workspaceRoot, "lib/ai-orchestrator/benchmark-results/ai-release-quality-decision.json"),
   252  );
   253  await writeAiReleaseQualityDecision(outputPath, decision);
   254  console.log(JSON.stringify({ ...decision, outputPath }, null, 2));
   255  if (decision.status === "blocked") process.exitCode = 1;
   256}
