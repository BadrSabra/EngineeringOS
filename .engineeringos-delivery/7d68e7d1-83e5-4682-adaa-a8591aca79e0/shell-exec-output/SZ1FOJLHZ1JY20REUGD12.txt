   123/**
   124 * Validates an API key by sending a minimal 1-token probe to the provider.
   125 *
   126 * Returns `{ valid: true }` on success.
   127 * Returns `{ valid: false, reason }` only on AUTH_ERROR — the key is definitively
   128 * rejected by the provider.
   129 * On any transient error (NETWORK_ERROR, TIMEOUT, SERVER_ERROR, NON_200) returns
   130 * `{ valid: true }` so a connectivity blip never prevents the user from saving a
   131 * key that might be perfectly valid.
   132 */
   133export async function validateProviderKey(
   134  provider: ProviderId,
   135  apiKey: string,
   136): Promise<{ valid: boolean; reason?: string }> {
   137  const config = loadProvider(provider);
   138  try {
   139    await probeProviderKey(config, apiKey);
   140    return { valid: true };
   141  } catch (err) {
   142    if (err instanceof GroqClientError && err.code === "AUTH_ERROR") {
   143      return { valid: false, reason: err.message };
   144    }
   145    // Transient error — allow the save; the next real request has provider fallback.
   146    return { valid: true };
   147  }
   148}
