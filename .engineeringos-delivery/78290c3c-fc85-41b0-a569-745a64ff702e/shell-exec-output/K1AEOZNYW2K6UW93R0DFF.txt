   177function beginTaskStream(
   178  res: import("express").Response,
   179  task: StructuredTask,
   180  projectId: string,
   181  metadata: StructuredAuditMetadata,
   182) {
   183  res.status(200).set({
   184    "Content-Type": "text/event-stream",
   185    "Cache-Control": "no-cache, no-transform",
   186    Connection: "keep-alive",
   187    "X-Accel-Buffering": "no",
   188  });
   189  res.flushHeaders();
   190
   191  let closed = false;
   192  const heartbeat = setInterval(() => {
   193    if (!closed && !res.writableEnded) res.write(": heartbeat\n\n");
   194  }, 15_000);
   195
   196  const emit = (event: StructuredTaskEvent) => {
   197    if (!closed && !res.writableEnded) {
   198      res.write(`data: ${JSON.stringify({ ...event, ...auditEnvelope(metadata) })}\n\n`);
   199    }
   200  };
   201
   202  const close = () => {
   203    if (closed) return;
   204    closed = true;
   205    clearInterval(heartbeat);
   206    if (!res.writableEnded) res.end();
   207  };
   208
   209  res.once("close", () => {
   210    closed = true;
   211    clearInterval(heartbeat);
   212  });
   213
   214  metadata.operationalTrace.push({ stage: "task", status: "started" });
   215  emit({ type: "task_started", task, projectId });
   216  return { emit, close };
   217}

   219function emitTaskFailure(
   220  emit: (event: StructuredTaskEvent) => void,
   221  close: () => void,
   222  err: unknown,
   223  metadata: StructuredAuditMetadata,
   224  params: { projectId: string; userId: string; task: StructuredTask; sessionId?: string },
   225) {
   226  const candidate = err as { code?: unknown; message?: unknown };
   227  const code = typeof candidate.code === "string" ? candidate.code : "task_failed";
   228  const failureKind =
   229    code === "model_output_invalid" || code === "INVALID_MODEL_OUTPUT" ? "PROVIDER_FORMAT" :
   230    code === "EMPTY_RESPONSE" ? "PROVIDER_FORMAT" :
   231    code === "RATE_LIMITED" ? "RATE_LIMIT" :
   232    code === "INVALID_CONFIG" || code === "AUTH_ERROR" || code === "MODEL_NOT_FOUND" || code === "PLAN_RESTRICTED" ? "CONFIGURATION" :
   233    code === "TIMEOUT" || code === "NETWORK_ERROR" || code === "NON_200" || code === "SERVER_ERROR" ? "PROVIDER_FAILURE" :
   234    "PROVIDER_FAILURE";
   235  const message =
   236    failureKind === "RATE_LIMIT" ? "The AI provider is rate-limited. Please wait before retrying." :
   237    failureKind === "CONFIGURATION" ? "The AI provider configuration needs attention before this can run." :
   238    failureKind === "PROVIDER_FORMAT" ? "The AI returned an unexpected response format." :
   239    "The AI provider could not complete this run.";
   240  metadata.incomplete = true;
   241  metadata.operationalTrace.push({ stage: "failed", status: "failed" });
   242  void persistStructuredFailure({
   243    ...params,
   244    failureKind,
   245    retryable: failureKind !== "CONFIGURATION",
   246    errorCode: code,
   247    errorMessage: message,
   248  }).then((sessionId) => {
   249    emit({
   250      type: "error",
   251      code,
   252      message,
   253      hint: failureKind === "RATE_LIMIT"
   254        ? "Wait a moment and retry the task."
   255        : failureKind === "CONFIGURATION"
   256          ? "Update the AI setup before starting a new task."
   257          : "You can retry this task without sending another prompt.",
   258      retryable: failureKind !== "CONFIGURATION",
   259      failureKind,
   260      outcome: "FAILED",
   261      sessionId,
   262    });
   263    close();
   264  }).catch((persistError) => {
   265    logger.error({ persistError, projectId: params.projectId, task: params.task }, "structured task failure persistence failed");
   266    emit({
   267      type: "error",
   268      code,
   269      message,
   270      hint: "You can retry this task without sending another prompt.",
   271      retryable: failureKind !== "CONFIGURATION",
   272      failureKind,
   273      outcome: "FAILED",
   274    });
   275    close();
   480  return res.json({ ...redactUserFacingValue(result) as Record<string, unknown>, ...auditEnvelope(metadata) });
   481});
