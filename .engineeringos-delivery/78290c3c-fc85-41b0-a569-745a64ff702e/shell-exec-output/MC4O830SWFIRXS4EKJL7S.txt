   100    "Content-Type": "text/event-stream",
   101    "Cache-Control": "no-cache, no-transform",
   102    Connection: "keep-alive",
   103    "X-Accel-Buffering": "no",
   104  });
   105  res.flushHeaders();
   106
   107  let closed = false;
   108  const heartbeat = setInterval(() => {
   109    if (!closed && !res.writableEnded) res.write(": heartbeat\n\n");
   110  }, 15_000);
   111
   112  const emit = (event: StructuredTaskEvent) => {
   113    if (!closed && !res.writableEnded) {
   114      res.write(`data: ${JSON.stringify({ ...event, ...auditEnvelope(metadata) })}\n\n`);
   115    }
   116  };
   117
   118  const close = () => {
   119    if (closed) return;
   120    closed = true;
   121    clearInterval(heartbeat);
   122    if (!res.writableEnded) res.end();
   123  };
   124
   125  res.once("close", () => {
   126    closed = true;
   127    clearInterval(heartbeat);
   128  });
   129
   130  metadata.operationalTrace.push({ stage: "task", status: "started" });
   131  emit({ type: "task_started", task, projectId });
   132  return { emit, close };
   133}

   135function emitTaskFailure(
   136  emit: (event: StructuredTaskEvent) => void,
   137  close: () => void,
   138  err: unknown,
   139  metadata: StructuredAuditMetadata,
   140) {
   141  const candidate = err as { code?: unknown; message?: unknown };
   142  const code = typeof candidate.code === "string" ? candidate.code : "task_failed";
   143  const failureKind =
   144    code === "model_output_invalid" || code === "INVALID_MODEL_OUTPUT" ? "PROVIDER_FORMAT" :
   145    code === "EMPTY_RESPONSE" ? "PROVIDER_FORMAT" :
   146    code === "RATE_LIMITED" ? "RATE_LIMIT" :
   147    code === "INVALID_CONFIG" || code === "AUTH_ERROR" || code === "MODEL_NOT_FOUND" || code === "PLAN_RESTRICTED" ? "CONFIGURATION" :
   148    code === "TIMEOUT" || code === "NETWORK_ERROR" || code === "NON_200" || code === "SERVER_ERROR" ? "PROVIDER_FAILURE" :
   149    "PROVIDER_FAILURE";
   150  const message =
   151    failureKind === "RATE_LIMIT" ? "The AI provider is rate-limited. Please wait before retrying." :
   152    failureKind === "CONFIGURATION" ? "The AI provider configuration needs attention before this can run." :
   153    failureKind === "PROVIDER_FORMAT" ? "The AI returned an unexpected response format." :
   154    "The AI provider could not complete this run.";
   155  metadata.incomplete = true;
   156  metadata.operationalTrace.push({ stage: "failed", status: "failed" });
   157  emit({
   158    type: "error",
   159    code,
   160    message,
   161    hint: failureKind === "RATE_LIMIT"
   162      ? "Wait a moment and retry the analysis."
   163      : failureKind === "CONFIGURATION"
   164        ? "Check the configured provider key or choose another provider."
   165        : "You can retry this task without sending another prompt.",
   166    retryable: failureKind !== "CONFIGURATION",
   167    failureKind,
   168    outcome: "FAILED",
   169  });
   170  close();
   171}

   175router.post("/ai/projects/:projectId/analyze", requireProjectAccess, async (req, res) => {
   176  const projectId = req.params.projectId as string;
   177  const project = req.project;
   178  if (!project) return res.status(500).json({ error: "Project context unavailable" });
   179  const metadata = await createAuditMetadata(projectId, project);
   180  recordTrace(metadata, "analyze", "started");
   181
   182  logger.info({ projectId }, "AI scan analysis requested");
   183
   184  const providerResolved = await requireProvider(req.userId, res, {
   185    qualityProfile: "analysis",
   186  });
   187  if (!providerResolved) return;
   188  const { provider, apiKey } = providerResolved;
   189
   190  const projectContext = await buildProjectContext(projectId, {
   191    sections: ["tasks", "metrics", "graphEntities", "graphRelationships", "events"],
   192  });
   193
   194  const rlAnalyze = await checkProjectRateLimitDb(projectId);
   195  if (!rlAnalyze.allowed) {
   196    return res.status(429).json({
   197      error: `LLM rate limit exceeded — max ${LLM_RATE_LIMIT} calls per minute per project. Retry in ${rlAnalyze.retryAfterSec}s.`,
   198    });
   199  }
   200
   201  let result: Awaited<ReturnType<typeof analyzeScan>>;
   202  let effectiveProvider = provider;
   203  try {
   204    ({ result, effectiveProvider } = await runAgentWithFallback(
   205      req.userId,
   206      { provider, apiKey },
   207      (opts) => analyzeScan(projectContext, opts),
   208      { qualityProfile: "analysis" },
   209    ));
   210  } catch (err) {
   211    metadata.incomplete = true;
   212    recordTrace(metadata, "analyze", "failed", effectiveProvider);
   213    if (handleOrchestratorError(err, res, { projectId, operation: "scan-analysis", provider: effectiveProvider })) return;
   214    throw err;
   215  }
   216
   217  // When the model output failed to parse, we still have the fallback data
   218  // (generated by fallbackScanAnalysis). Serve it with a warning header so the
   219  // UI can still display something useful rather than showing a hard error.
   220  // Only a completely empty/unusable response (no summary at all) should 422.
   221  if (result._parseError) {
   222    metadata.incomplete = true;
   223    recordTrace(metadata, "analyze", "incomplete", effectiveProvider);
   224    logger.warn(
   225      { projectId, parseCode: result._parseError.code, message: result._parseError.message, provider: effectiveProvider },
   226      "scan-analyst: parse error",
   227    );
   228    return res.status(422).json({
   229      ...auditEnvelope(metadata),
   230      error: "model_output_invalid",
   231      code: "model_output_invalid",
   232      hint: "The AI model returned an unexpected response — try again in a moment.",
   233      parseCode: result._parseError.code,
   234    });
   235  }
   236
   237  invalidateContextCache(projectId);
   238
   239  await db.transaction(async (tx) => {
   240    await tx.insert(auditLogsTable).values({
   241      id: randomUUID(),
   242      entityType: "project",
   243      entityId: projectId,
   244      action: "ai_analyzed",
   245      projectId,
   246      actor: req.userId,
   247      stateBefore: {},
   248      stateAfter: { summary: result.summary, overallAssessment: result.overallAssessment },
   249    });
   250    await tx.insert(eventsTable).values({
   251      id: randomUUID(),
   252      type: "AiScanAnalysisCompleted",
   253      projectId,
   254      severity: "info",
   255      message: `AI scan analysis completed: ${result.summary}`,
   256    });
   257  });
   258
   259  recordTrace(metadata, "analyze", "completed", effectiveProvider);
   260  return res.json({ ...redactUserFacingValue(result) as Record<string, unknown>, ...auditEnvelope(metadata) });
   261});
