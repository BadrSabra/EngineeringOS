   337function validateRegistration(adapter: CapabilityAdapter | null | undefined): CapabilityFailure | undefined {
   338  if (!adapter || typeof adapter !== "object") {
   339    return failure("", 0, "CAPABILITY_CONTRACT_INVALID", "A capability registration must be an object.");
   340  }
   341  const parsed = CapabilityRegistrationSchema.safeParse({
   342    contractVersion: adapter.contractVersion,
   343    id: adapter.id,
   344    supportedRecipeVersions: adapter.supportedRecipeVersions,
   345    policy: adapter.policy,
   346  });
   347  if (!parsed.success) {
   348    return failure(
   349      String(adapter.id ?? ""),
   350      Number(adapter.supportedRecipeVersions?.[0] ?? 0),
   351      "CAPABILITY_CONTRACT_INVALID",
   352      "The server capability registration does not satisfy the capability contract.",
   353    );
   354  }
   355  if (adapter.inputSchema === undefined || adapter.outputSchema === undefined || typeof adapter.execute !== "function") {
   356    return failure(
   357      adapter.id,
   358      adapter.supportedRecipeVersions[0],
   359      "CAPABILITY_CONTRACT_INVALID",
   360      "A capability requires input and output schemas plus an implementation.",
   361    );
   362  }
   363  if (typeof adapter.inputSchema.safeParse !== "function" ||
   364      typeof adapter.outputSchema.safeParse !== "function") {
   365    return failure(
   366      adapter.id,
   367      adapter.supportedRecipeVersions[0],
   368      "CAPABILITY_CONTRACT_INVALID",
   369      "A capability requires runtime-valid input and output schemas.",
   370    );
   371  }
   372  return undefined;
   373}

   375function cloneRegistration(adapter: CapabilityAdapter): CapabilityAdapter {
   376  const catalog = adapter.catalog
   377    ? Object.freeze({
   378        ...adapter.catalog,
   379        inputShape: Object.freeze({
   380          ...adapter.catalog.inputShape,
   381          fields: Object.freeze(adapter.catalog.inputShape.fields.map((field) => Object.freeze({ ...field }))),
   382        }),
   383        supportedScopes: Object.freeze([...adapter.catalog.supportedScopes]),
   384        keywords: Object.freeze([...adapter.catalog.keywords]),
   385        allowedPhases: Object.freeze([...adapter.catalog.allowedPhases]),
   386        projectIds: Object.freeze([...adapter.catalog.projectIds]),
   387        expectedEvidence: Object.freeze([...adapter.catalog.expectedEvidence]),
   388      })
   389    : undefined;
   390  return Object.freeze({
   391    ...adapter,
   392    supportedRecipeVersions: Object.freeze([...adapter.supportedRecipeVersions]),
   393    policy: Object.freeze({
   394      ...adapter.policy,
   395      allowedOperations: Object.freeze([...adapter.policy.allowedOperations]),
   396      approvedCommandProfiles: Object.freeze([...adapter.policy.approvedCommandProfiles]),
   397    }),
   398    ...(catalog ? { catalog } : {}),
   399  }) as unknown as CapabilityAdapter;

   402/**
   403 * Deterministic registry of server-approved capability adapters.
   404 *
   405 * Registrations are immutable after insertion. All public execution paths
   406 * validate the opaque ID, recipe version, input, operation, and output before
   407 * returning a success value.
   408 */
   409export class CapabilityRegistry {
   410  private readonly adapters = new Map<CapabilityId, CapabilityAdapter>();
   411
   412  public constructor(adapters: readonly CapabilityAdapter[] = []) {
   413    for (const adapter of adapters) this.register(adapter);
   414  }
   415
   416  public register(adapter: CapabilityAdapter): void {
   417    const invalid = validateRegistration(adapter);
   418    if (invalid) {
   419      throw new CapabilityRegistryError("CAPABILITY_CONTRACT_INVALID", invalid.detail);
   420    }
   421    const id = adapter.id as CapabilityId;
   422    if (this.adapters.has(id)) {
   423      throw new CapabilityRegistryError(
   424        "CAPABILITY_DUPLICATE_ID",
   425        `Capability "${id}" is already registered.`,
   426      );
   427    }
   428    this.adapters.set(id, cloneRegistration(adapter));
   429  }
   430
   431  public has(capabilityId: string): boolean {
   432    return this.adapters.has(capabilityId as CapabilityId);
   433  }
   434
   435  public get(capabilityId: string): CapabilityAdapter | undefined {
   436    const parsed = CapabilityIdSchema.safeParse(capabilityId);
   437    return parsed.success ? this.adapters.get(parsed.data) : undefined;
   438  }
   439
   440  public list(): readonly CapabilityDescriptor[] {
   441    return [...this.adapters.values()]
   442      .sort((left, right) => left.id.localeCompare(right.id))
   443      .map(({ inputSchema: _inputSchema, outputSchema: _outputSchema, execute: _execute, ...descriptor }) => descriptor);
   444  }
   445
   446  public validateInput(
   447    capabilityId: string,
   448    recipeVersion: number,
   449    input: unknown,
   450  ): CapabilityValidation {
   451    const id = CapabilityIdSchema.safeParse(capabilityId);
   452    if (!id.success) return failure(capabilityId, recipeVersion, "CAPABILITY_UNKNOWN_ID", "The capability ID is not registered.");
   453
   454    const adapter = this.adapters.get(id.data);
   455    if (!adapter) return failure(id.data, recipeVersion, "CAPABILITY_UNKNOWN_ID", "The capability ID is not registered.");
   456    if (!RecipeVersionSchema.safeParse(recipeVersion).success ||
   457        !adapter.supportedRecipeVersions.includes(recipeVersion as RecipeVersion)) {
   458      return failure(id.data, recipeVersion, "CAPABILITY_RECIPE_VERSION_UNSUPPORTED", "The recipe version is not supported by this capability.");
   459    }
   460
   461    const inputBytes = jsonByteLength(input);
   462    if (inputBytes === undefined) {
   463      return failure(id.data, recipeVersion, "CAPABILITY_INPUT_INVALID", "Capability input must be JSON-serializable.");
   464    }
   465    if (inputBytes > adapter.policy.maxInputBytes) {
   466      return failure(id.data, recipeVersion, "CAPABILITY_INPUT_TOO_LARGE", "Capability input exceeds the server input limit.");
   467    }
   468    if (hasReservedInputKey(input)) {
   469      return failure(id.data, recipeVersion, "CAPABILITY_INPUT_INVALID", "Capability input contains a server-owned execution control.");
   470    }
   471
   472    let parsedInput: ReturnType<typeof adapter.inputSchema.safeParse>;
   473    try {
   474      parsedInput = adapter.inputSchema.safeParse(input);
   475    } catch {
   476      return failure(id.data, recipeVersion, "CAPABILITY_INPUT_INVALID", "Capability input could not be validated.");
   477    }
   478    if (!parsedInput.success) {
   479      return failure(id.data, recipeVersion, "CAPABILITY_INPUT_INVALID", "Capability input does not match the registered input schema.");
   480    }
   481    return { ok: true, capability: adapter, input: parsedInput.data };
   482  }
