    95async function waitFor(url, label) {
    96  const deadline = Date.now() + timeoutMs;
    97  let lastError = "not attempted";
    98
    99  while (Date.now() < deadline) {
   100    try {
   101      const response = await fetch(url, {
   102        signal: AbortSignal.timeout(5_000),
   103      });
   104      if (response.ok) {
   105        return response;
   106      }
   107      lastError = `HTTP ${response.status}`;
   108    } catch (error) {
   109      lastError = error instanceof Error ? error.message : String(error);
   110    }
   111    await new Promise((resolve) => setTimeout(resolve, 500));
   112  }
   113
   114  throw new Error(`${label} did not become healthy: ${lastError} (${url})`);
   115}
