   491/**
   492 * PR-06: Compact runtime status badge for a provider card.
   493 * Shows circuit state, consecutive failures, last success, and avg latency.
   494 */
   495function ProviderRuntimeBadge({ metric }: { metric: ProviderRuntimeMetric | undefined }) {
   496  if (!metric || metric.requests === 0) return null;
   497
   498  const cooldownSec = metric.cooldownRemainingMs != null
   499    ? Math.ceil(metric.cooldownRemainingMs / 1000)
   500    : null;
   501
   502  const lastSuccessLabel = metric.lastSuccessAt
   503    ? (() => {
   504        const secs = Math.floor((Date.now() - new Date(metric.lastSuccessAt).getTime()) / 1000);
   505        if (secs < 60)   return `${secs}s ago`;
   506        if (secs < 3600) return `${Math.floor(secs / 60)}m ago`;
   507        return `${Math.floor(secs / 3600)}h ago`;
   508      })()
   509    : null;
   510
   511  if (metric.circuitOpen) {
   512    return (
   513      <div className="mt-1.5 flex items-center gap-1.5 text-[10px] text-red-400">
   514        <span className="w-1.5 h-1.5 rounded-full bg-red-400 shrink-0" />
   515        <span>Circuit open{cooldownSec != null ? ` · ${cooldownSec}s cooldown` : ''}</span>
   516      </div>
   517    );
   518  }
   519
   520  const isHealthy = metric.consecutiveFailures === 0 && metric.successRate != null && metric.successRate > 0.8;
   521  const isDegraded = metric.consecutiveFailures > 0 || (metric.successRate != null && metric.successRate < 0.8);
   522
   523  return (
   524    <div className="mt-1.5 flex items-center gap-1.5 text-[10px] text-muted-foreground">
   525      <span className={`w-1.5 h-1.5 rounded-full shrink-0 ${isHealthy ? 'bg-green-400' : isDegraded ? 'bg-yellow-400' : 'bg-muted-foreground'}`} />
   526      <span className="flex-1 truncate">
   527        {metric.consecutiveFailures > 0
   528          ? `${metric.consecutiveFailures} consecutive fail${metric.consecutiveFailures > 1 ? 's' : ''}`
   529          : lastSuccessLabel
   530            ? `OK · ${lastSuccessLabel}`
   531            : 'Active'}
   532        {metric.avgLatencyMs != null ? ` · ${metric.avgLatencyMs}ms` : ''}
   533      </span>
   534    </div>
   535  );
   536}

   570/** Build a provider context suffix for error messages, showing model/status/raw hint. */
   571function providerContextSuffix(err: ApiError): string {
   572  const parts: string[] = [];
   573  const ctx = err.providerContext as Record<string, unknown> | undefined;
   574  if (ctx) {
   575    if (ctx['providerName']) parts.push(`Provider: ${ctx['providerName']}`);
   576    if (ctx['providerModel']) parts.push(`Model: ${fmtModelId(String(ctx['providerModel']))}`);
   577    if (ctx['providerStatus']) parts.push(`Status: ${ctx['providerStatus']}`);
   578  }
   579  return parts.length > 0 ? ` (${parts.join(' · ')})` : '';
   580}

   582function availabilitySuffix(err: unknown): string {
   583  const value = err as { data?: unknown; availabilityState?: unknown; operatorAction?: unknown; correlationId?: unknown };
   584  const data = value.data && typeof value.data === 'object' ? value.data as Record<string, unknown> : {};
   585  const state = data.availabilityState ?? value.availabilityState;
   586  const action = data.operatorAction ?? value.operatorAction;
   587  const correlationId = data.correlationId ?? value.correlationId;
   588  const parts = [
   589    typeof state === 'string' ? `Status: ${state.replace(/_/g, ' ')}` : '',
   590    typeof action === 'string' ? `Next step: ${action}` : '',
   591    typeof correlationId === 'string' ? `Reference: ${correlationId}` : '',
   592  ].filter(Boolean);
   593  return parts.length > 0 ? ` (${parts.join(' · ')})` : '';
   594}

   596function describeAiError(err: unknown): string {
   597  if (err instanceof AiApiError) {
   598    const suffix = providerContextSuffix(err) + availabilitySuffix(err);
   599    switch (err.status) {
   600      case 400: return err.errorMessage + suffix;
   601      case 401: return (err.errorMessage || err.hint || 'AI API key is invalid — delete it and save a valid key from your provider\'s dashboard.') + suffix;
   602      case 403: return 'Access denied — you may not have permission on this project.';
   603      case 429: return (err.errorMessage || err.hint || 'AI rate limit reached — wait 30–60 seconds before retrying.') + suffix;
   604      case 422:
   605        if (err.code === 'model_output_invalid') {
   606          return 'The AI returned an unexpected response format — try rephrasing your message.';
   607        }
   608        if (err.code === 'MODEL_NOT_FOUND') {
   609          return (err.errorMessage || err.hint || 'The selected AI model is unavailable — try again or switch providers.') + suffix;
   610        }
   611        return (err.errorMessage || err.hint || 'AI provider configuration is invalid. Re-save your API key.') + suffix;
   612      case 428: return (err.errorMessage || err.hint || 'No AI key configured — save an OpenRouter, DeepSeek, or Groq API key first.') + suffix;
   613      case 502: return (err.errorMessage || err.hint || 'AI provider returned an error. Check your API key or try again.') + suffix;
   614      case 503: return (err.errorMessage || err.hint || 'AI provider is temporarily unreachable — try again in a moment.') + suffix;
   615      default:  return (err.errorMessage || `Request failed (${err.status}).`) + suffix;
   616    }
   617  }
   618  if (err instanceof Error) return err.message + availabilitySuffix(err);
   619  return String(err);
   620}
