   770/**
   771 * Reads a server-sent-events `body` to completion, parses each `data: {...}\n\n`
   772 * frame, and dispatches the typed event to the matching callback in `callbacks`.
   773 *
   774 * Extracted from `useAiChatStream` so the routing logic can be tested directly
   775 * without a React / fetch runtime.  The hook calls this function internally;
   776 * the behaviour is identical to what it was before extraction.
   777 *
   778 * @internal Exported for testing — not part of the public API contract.
   779 */
   780export async function processAiStream(
   781  body: ReadableStream<Uint8Array>,
   782  callbacks: AiChatStreamCallbacks,
   783): Promise<void> {
   784  const reader = body.getReader();
   785  const decoder = new TextDecoder();
   786  let buffer = '';
   787  let executionStarted = false;
   788  let taskStarted = false;
   789  let terminalEventReceived = false;
   790
   791  // eslint-disable-next-line no-constant-condition
   792  while (true) {
   793    const { done, value } = await reader.read();
   794    if (done) {
   795      // A proxy/provider can close a healthy-looking SSE response without
   796      // sending either `done` or `error`. Once the server has issued an
   797      // execution identity, surface that EOF as a recoverable interruption so
   798      // callers can resume the same operation instead of silently dropping it.
   799      if ((executionStarted || taskStarted) && !terminalEventReceived) {
   800        callbacks.onStreamReset?.();
   801      }
   802      break;
   803    }
   804
   805    buffer += decoder.decode(value, { stream: true });
   806
   807    // SSE events are separated by double newlines: "data: {...}\n\n"
   808    const chunks = buffer.split('\n\n');
   809    // Last element is the incomplete chunk — keep it for the next read
   810    buffer = chunks.pop() ?? '';
   811
   812    for (const chunk of chunks) {
   813      const dataLine = chunk.split('\n').find((l) => l.startsWith('data: '));
   814      if (!dataLine) continue;
   815
   816      let event: AiStreamEvent;
   817      try {
   818        event = JSON.parse(dataLine.slice('data: '.length)) as AiStreamEvent;
   819      } catch {
   820        continue; // malformed event — skip
   821      }
   822
   823      if (event.type === 'execution_started') executionStarted = true;
   824      if (event.type === 'task_started') taskStarted = true;
   825      if (event.type === 'done' || event.type === 'error') terminalEventReceived = true;
   826
   827      switch (event.type) {
   828        case 'execution_started':
   829          callbacks.onExecutionStarted?.(event);
   830          break;
