   283function inferOperationMode(params: {
   284  operationMode?: unknown;
   285  taskResult?: AiTaskResult | null;
   286  proposalId?: string | null;
   287}): OperationMode | undefined {
   288  if (
   289    params.operationMode === 'FORENSIC_AUDIT'
   290    || params.operationMode === 'DELIVERY'
   291    || params.operationMode === 'CHAT'
   292  ) {
   293    return params.operationMode;
   294  }
   295  if (params.proposalId) return 'DELIVERY';
   296  if (params.taskResult?.kind === 'IMPLEMENTATION_PLAN_RESULT') return 'DELIVERY';
   297  if (
   298    params.taskResult?.kind === 'FORENSIC_REPORT_RESULT'
   299    || params.taskResult?.kind === 'WORKSPACE_REVIEW_RESULT'
   300    || params.taskResult?.kind === 'FINDING_RESULT'
   301    || params.taskResult?.kind === 'BEHAVIOR_ANSWER_RESULT'
   302    || params.taskResult?.kind === 'REPAIR_RESULT'
   303  ) {
   304    return 'FORENSIC_AUDIT';
   305  }
   306  return undefined;
   307}
