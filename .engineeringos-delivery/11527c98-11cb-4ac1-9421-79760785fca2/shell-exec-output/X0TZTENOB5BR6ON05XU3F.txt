   223/**
   224 * Verify that the models selected by the provider registry are still exposed
   225 * by Groq. Credential validity and model availability are independent: a key
   226 * can remain valid after Groq retires a model slug.
   227 *
   228 * This intentionally uses the authenticated models endpoint rather than a
   229 * completion request. It is bounded, does not consume tokens, and never
   230 * includes the API key in its result or error message.
   231 */
   232export async function validateGroqDefaultModels(
   233  apiKey: string,
   234  defaults: { fast: string; powerful: string } = {
   235    fast: MODEL_FAST,
   236    powerful: MODEL_POWERFUL,
   237  },
   238  timeoutMs = 10_000,
   239): Promise<GroqDefaultModelValidation> {
   240  const client = getClient(apiKey);
   241  const controller = new AbortController();
   242  const timer = setTimeout(() => controller.abort(), timeoutMs);
   243
   244  try {
   245    const response = await client.models.list({ signal: controller.signal });
   246    const modelIds = new Set(
   247      (response.data ?? [])
   248        .map((model) => model?.id)
   249        .filter((modelId): modelId is string => typeof modelId === "string"),
   250    );
   251    const missing = (["fast", "powerful"] as const).filter(
   252      (role) => !modelIds.has(defaults[role]),
   253    );
   254
   255    if (missing.length > 0) {
   256      const missingDescription = missing
   257        .map((role) => `${role}="${defaults[role]}"`)
   258        .join(", ");
   259      return {
   260        valid: false,
   261        missing,
   262        checkedModels: { ...defaults },
   263        reason:
   264          `Groq model catalog is missing the configured ${missingDescription}. ` +
   265          "Update the Groq default model IDs and restart before accepting AI traffic.",
   266      };
   267    }
   268
   269    return {
   270      valid: true,
   271      missing: [],
   272      checkedModels: { ...defaults },
   273    };
   274  } catch (error) {
   275    throw classifySdkError(error, controller.signal.aborted);
   276  } finally {
   277    clearTimeout(timer);
   278  }
   279}
