   170    return predicate.predicates.some(findContradiction);
   171  }
   172  if (predicate.kind === "any") return predicate.predicates.some(findContradiction);
   173  if (predicate.kind === "not") return findContradiction(predicate.predicate);
   174  return false;
   175}

   177function mapPredicate(
   178  predicate: EvidencePredicate,
   179  sourceNodes: ReadonlyMap<string, CapabilityRecipeNode>,
   180  compiledIds: ReadonlyMap<string, string>,
   181  diagnostics: RecipeCompilationDiagnostic[],
   182  path: string,
   183): EvidencePredicate {
   184  const mapNode = (sourceId: string, childPath: string): string => {
   185    const id = compiledIds.get(sourceId);
   186    if (!sourceNodes.has(sourceId)) {
   187      diagnostics.push(diagnostic("UNKNOWN_TRANSITION_NODE", childPath, `Node "${sourceId}" is not declared.`));
   188    }
   189    return id ?? nodeId("invalid", sourceId);
   190  };
   191  if (predicate.kind === "all" || predicate.kind === "any") {
   192    return {
   193      ...predicate,
   194      predicates: predicate.predicates.map((child, index) =>
   195        mapPredicate(child, sourceNodes, compiledIds, diagnostics, `${path}.predicates[${index}]`)),
   196    };
   197  }
   198  if (predicate.kind === "not") {
   199    return {
   200      kind: "not",
   201      predicate: mapPredicate(predicate.predicate, sourceNodes, compiledIds, diagnostics, `${path}.predicate`),
   202    };
   203  }
   204  const mapped = { ...predicate, nodeId: mapNode(predicate.nodeId, `${path}.nodeId`) };
   205  if (mapped.kind === "output_present" || mapped.kind === "output_equals") {
   206    const source = sourceNodes.get(predicate.nodeId);
   207    if (source && !source.declaredOutputs.includes(predicate.output)) {
   208      diagnostics.push(diagnostic(
   209        "UNDECLARED_OUTPUT",
   210        `${path}.output`,
   211        `Output "${predicate.output}" is not declared by node "${predicate.nodeId}".`,
   212      ));
   213    }
   214  }
   215  return mapped;
   216}

   219  target: RecipeTarget,
   220  compiledIds: ReadonlyMap<string, string>,
   221  diagnostics: RecipeCompilationDiagnostic[],
   222  path: string,
   223): RecipeTarget {

   232function topologicalOrder(
   233  nodes: readonly CapabilityRecipeNode[],
   234  ids: ReadonlySet<string>,
   235  diagnostics: RecipeCompilationDiagnostic[],
   236): CapabilityRecipeNode[] {
   237  const byId = new Map(nodes.map((node) => [node.id, node]));
   238  const indegree = new Map(nodes.map((node) => [node.id, 0]));
   239  const dependents = new Map(nodes.map((node) => [node.id, [] as string[]]));
   240  for (const node of nodes) {
   241    for (const dependency of node.dependsOn) {
   242      if (!ids.has(dependency)) {
   243        diagnostics.push(diagnostic("UNKNOWN_DEPENDENCY", `nodes.${node.id}.dependsOn`, `Node "${dependency}" is not declared.`));
   244        continue;
   245      }
   246      indegree.set(node.id, (indegree.get(node.id) ?? 0) + 1);
   247      dependents.get(dependency)?.push(node.id);
   248    }
   249  }
   250  const queue = nodes.filter((node) => indegree.get(node.id) === 0).map((node) => node.id);
   251  const ordered: CapabilityRecipeNode[] = [];
   252  while (queue.length > 0) {
   253    const current = queue.shift()!;
   254    ordered.push(byId.get(current)!);
   255    for (const dependent of dependents.get(current) ?? []) {
   256      const next = (indegree.get(dependent) ?? 0) - 1;
   257      indegree.set(dependent, next);
   258      if (next === 0) queue.push(dependent);
   259    }
   260  }
   261  if (ordered.length !== nodes.length) {
   262    diagnostics.push(diagnostic("DEPENDENCY_CYCLE", "nodes", "Recipe dependencies must form an acyclic graph."));
   263  }
   264  return ordered;
   265}

   267function graphHasCycle(transitions: readonly CompiledRecipeTransition[]): boolean {
   268  const graph = new Map<string, string[]>();
   269  for (const transition of transitions) {
   270    const targets = [transition.then, transition.otherwise]
   271      .filter((target): target is { kind: "node"; nodeId: string } => target.kind === "node")
   272      .map((target) => target.nodeId);
   273    graph.set(transition.fromNodeId, [...(graph.get(transition.fromNodeId) ?? []), ...targets]);
   274  }
   275  const visiting = new Set<string>();
   276  const visited = new Set<string>();
   277  const visit = (id: string): boolean => {
   278    if (visiting.has(id)) return true;
   279    if (visited.has(id)) return false;
   280    visiting.add(id);
   281    if ((graph.get(id) ?? []).some(visit)) return true;
   282    visiting.delete(id);
   283    visited.add(id);
   284    return false;
   285  };
   286  return [...graph.keys()].some(visit);
   287}

   289function everyNodeCanTerminate(
   290  nodes: readonly ExecutionNode[],
   291  transitions: readonly CompiledRecipeTransition[],
   292): boolean {
   293  const nodeIds = new Set(nodes.map((node) => node.id));
   294  const outgoing = new Map<string, CompiledRecipeTransition[]>();
   295  for (const transition of transitions) {
   296    outgoing.set(transition.fromNodeId, [...(outgoing.get(transition.fromNodeId) ?? []), transition]);
   297  }
   298  const memo = new Map<string, boolean>();
   299  const visiting = new Set<string>();
   300  const canTerminate = (id: string): boolean => {
   301    if (memo.has(id)) return memo.get(id)!;
   302    if (visiting.has(id)) return false;
   303    visiting.add(id);
   304    const result = (outgoing.get(id) ?? []).length > 0
   305      && (outgoing.get(id) ?? []).every((transition) =>
   306        [transition.then, transition.otherwise].every((target) =>
   307          target.kind === "terminal" || (nodeIds.has(target.nodeId) && canTerminate(target.nodeId)),
   308        ));
   309    visiting.delete(id);
   310    memo.set(id, result);
   311    return result;
   312  };
   313  return nodes.every((node) => canTerminate(node.id));
   314}
