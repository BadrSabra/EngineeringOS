    13export const eventsTable = pgTable("events", {
    14  id: text("id").primaryKey(),
    15  type: text("type").notNull(),
    16  projectId: text("project_id")
    17    .notNull()
    18    .references(() => projectsTable.id, { onDelete: "cascade" }),
    19  taskId: text("task_id").references(() => tasksTable.id, { onDelete: "set null" }),
    20  workflowId: text("workflow_id").references(() => workflowsTable.id, { onDelete: "set null" }),
    21  payload: jsonb("payload").$type<Record<string, unknown>>(),

    23  message: text("message").notNull().default(""),
    24  timestamp: timestamp("timestamp").notNull().defaultNow(),
    25  /**
    26   * Soft link to a logical operation (scan, task execute, workflow advance,
    27   * etc.). All records sharing the same correlationId were produced by one
    28   * call. Nullable: older rows and operations that span no secondary tables
    29   * (e.g. a plain project update) carry no correlationId.
    30   */
    31  correlationId: text("correlation_id"),
    32}, (t) => [
    33  index("idx_events_project_id").on(t.projectId),
    34  // Covers: WHERE project_id = ? ORDER BY timestamp DESC LIMIT n  (no type filter)
    35  index("idx_events_project_id_timestamp").on(t.projectId, t.timestamp),
    36  /**
    37   * DB-17: Covers the most common filtered query in events.ts:
    38   *   WHERE project_id = ? AND type = ? ORDER BY timestamp DESC
    39   * PostgreSQL uses a backward index scan on timestamp, so DESC ordering is
    40   * served efficiently without a separate sort step.
    41   */
    42  index("idx_events_project_id_type_timestamp").on(t.projectId, t.type, t.timestamp),
    43  /**
    44   * DB-17: Covers the correlationId scoped lookup in events.ts:
    45   *   WHERE project_id = ? AND correlation_id = ?
    46   * The project_id leading column means this index is scoped — a correlationId
    47   * from a different project cannot accidentally match.
    48   */
    49  index("idx_events_project_id_correlation_id").on(t.projectId, t.correlationId),
    50  // Global timestamp index — kept for admin/dashboard time-range queries
    51  // that do not have a project_id filter (e.g. system-wide event feed).
    52  index("idx_events_timestamp").on(t.timestamp),
    53  // Single-column correlationId index — kept for cross-project audit trail
    54  // lookups (e.g. "show me everything from correlation X regardless of project").
    55  index("idx_events_correlation_id").on(t.correlationId),
    56]);

    13/**
    14 * GET /events
    15 *
    16 * Returns events scoped to a single project, or across all projects owned by
    17 * the authenticated user when projectId is omitted. Ownership is verified
    18 * before returning any rows — a user cannot read events from a project they
    19 * don't own.
    20 *
    21 * Optional filters: type, correlationId, severity, search, page, limit
    22 * (default 50, max 200). Pagination is applied after all filters.
    23 * Results are ordered newest-first.
    24 */
    25router.get("/events", async (req, res) => {
    26  const rawLimit = req.query.limit === undefined ? 50 : Number(req.query.limit);
    27  const rawPage = req.query.page === undefined ? 1 : Number(req.query.page);
    28  if (!Number.isSafeInteger(rawLimit) || rawLimit < 1 || rawLimit > 200) {
    29    return res.status(400).json({ error: "limit must be an integer between 1 and 200" });
    30  }
    31  if (!Number.isSafeInteger(rawPage) || rawPage < 1) {
    32    return res.status(400).json({ error: "page must be at least 1" });
    33  }
    34  const limit = rawLimit;
    35  const page = rawPage;
    36
    37  // Parse optional filters at the boundary so blank values do not become
    38  // restrictive database predicates.
    39  const correlationIdFilter = typeof req.query.correlationId === "string" ? req.query.correlationId.trim() || undefined : undefined;
    40  const typeFilter = typeof req.query.type === "string" ? req.query.type.trim() || undefined : undefined;
    41  const severityFilter = typeof req.query.severity === "string" ? req.query.severity : undefined;
    42  const searchFilter = typeof req.query.search === "string" ? req.query.search.trim() || undefined : undefined;
    43
    44  const projectId =
    45    typeof req.query.projectId === "string" ? req.query.projectId : undefined;
    46  let projectIds: string[];
    47  if (projectId) {
    48    // Explicit project requests retain the existing 404/403 ownership semantics.
    49    const project = await loadProjectByIdForUser(projectId, req.userId, res);
    50    if (!project) return; // response already sent
    51    projectIds = [project.id];
    52  } else {
    53    // Resolve ownership first, then constrain the event query to those IDs.
    54    // This avoids exposing events from projects owned by another user.
    55    const ownedProjects = await db
    56      .select({ id: projectsTable.id })
    57      .from(projectsTable)
    58      .where(eq(projectsTable.ownerId, req.userId));
    59    projectIds = ownedProjects.map(({ id }) => id);
    60     if (projectIds.length === 0) return res.json({ events: [], total: 0 });
    61  }
    62
    63  const conditions: ReturnType<typeof eq>[] = [inArray(eventsTable.projectId, projectIds)];
    64  if (typeFilter) conditions.push(eq(eventsTable.type, typeFilter));
    65  if (correlationIdFilter)
    66    conditions.push(eq(eventsTable.correlationId, correlationIdFilter));
    67  if (severityFilter) conditions.push(eq(eventsTable.severity, severityFilter as "info" | "warning" | "error" | "success"));
    68  if (searchFilter) {
    69    const pattern = `%${searchFilter}%`;
    70    conditions.push(
    71      or(
    72        ilike(eventsTable.message, pattern),
    73        ilike(eventsTable.type, pattern),
    74        ilike(eventsTable.correlationId, pattern),
    75      )!,
    76    );
    77  }
    78
    79  const matchingEvents = db
    80    .select()
    81    .from(eventsTable)
    82    .where(and(...conditions));
    83  const [events, totalResult] = await Promise.all([
    84    matchingEvents
    85      // Keep page boundaries stable when several events share a timestamp.
    86      .orderBy(desc(eventsTable.timestamp), desc(eventsTable.id))
    87      .limit(limit)
    88      .offset((page - 1) * limit),
    89    db
    90      .select({ total: count() })
    91      .from(eventsTable)
    92      .where(and(...conditions)),
    93  ]);
    94
    95  return res.json({ events, total: Number(totalResult[0]?.total ?? 0) });
    96});
