    36// ── Helpers ───────────────────────────────────────────────────────────────────
    37
    38const VALID_PROVIDERS = new Set<string>(Object.keys(PROVIDER_REGISTRY));
    39
    40function isValidProvider(p: string): p is ProviderId {
    41  return VALID_PROVIDERS.has(p);
    42}
    43
    44/** Shared GET handler — return key status (never the key itself). */
    45async function handleGetKey(req: Request, res: Response, provider: ProviderId) {
    46  const [row] = await db
    47    .select({
    48      last4: aiProviderCredentialsTable.last4,
    49      updatedAt: aiProviderCredentialsTable.updatedAt,
    50    })
    51    .from(aiProviderCredentialsTable)
    52    .where(
    53      and(
    54        eq(aiProviderCredentialsTable.ownerId, req.userId),
    55        eq(aiProviderCredentialsTable.provider, provider),
    56      ),
    57    )
    58    .limit(1);
    59
    60  if (!row) return res.json({ configured: false, last4: null, updatedAt: null });
    61  return res.json({ configured: true, last4: row.last4, updatedAt: row.updatedAt });
    62}
    63
    64/** Shared PUT handler — validate and persist the API key. */
    65async function handlePutKey(req: Request, res: Response, provider: ProviderId) {
    66  const { apiKey } = req.body as { apiKey?: string };
    67  if (!apiKey || typeof apiKey !== "string" || apiKey.trim().length < 10) {
    68    return res.status(400).json({ error: "apiKey must be at least 10 characters" });
    69  }
    70  const trimmed = apiKey.trim();
    71
    72  const config = PROVIDER_REGISTRY[provider];
    73  const label = config?.label ?? provider;
    74  const consoleUrl = config?.consoleUrl ?? "your provider's dashboard";
    75
    76  const validation = await validateProviderKey(provider, trimmed);
    77  if (!validation.valid) {
    78    return res.status(422).json({
    79      error: `${label} API key is invalid or unauthorized`,
    80      hint: `Check your key at ${consoleUrl} — it was rejected by the ${label} API.`,
    81      detail: validation.reason,
    82    });
    83  }
    84
    85  const last4 = trimmed.slice(-4);
    86  let encryptedApiKey: string;
    87  try {
    88    encryptedApiKey = encryptApiKey(trimmed);
    89  } catch (err) {
    90    logger.error({ err, provider }, "Key encryption failed");
    91    return res.status(500).json({ error: "Key storage unavailable — encryption not configured" });
    92  }
    93
    94  const now = new Date();
    95  await db
    96    .insert(aiProviderCredentialsTable)
    97    .values({
    98      id: randomUUID(),
    99      ownerId: req.userId,
   100      provider,
   101      encryptedApiKey,
   102      last4,
   103      createdAt: now,
   104      updatedAt: now,
   105    })
   106    .onConflictDoUpdate({
   107      target: [aiProviderCredentialsTable.ownerId, aiProviderCredentialsTable.provider],
   108      set: { encryptedApiKey, last4, updatedAt: now },
   109    });
   110
   111  return res.json({ configured: true, last4, updatedAt: now });
   112}
   113
   114/** Shared DELETE handler — remove the saved key. */
   115async function handleDeleteKey(req: Request, res: Response, provider: ProviderId) {
   116  await db
   117    .delete(aiProviderCredentialsTable)
   118    .where(
   119      and(
   120        eq(aiProviderCredentialsTable.ownerId, req.userId),
   121        eq(aiProviderCredentialsTable.provider, provider),
   122      ),
   123    );
   124  return res.json({ configured: false });
   125}

   127// ── Generic routes (:provider param) ─────────────────────────────────────────

   129router.get("/ai/providers/:provider/key", async (req, res) => {
   130  const { provider } = req.params;
   131  if (!isValidProvider(provider)) {
   132    return res.status(400).json({
   133      error: `Unknown provider "${provider}". Valid values: ${[...VALID_PROVIDERS].join(", ")}`,
   134    });
   135  }
   136  return handleGetKey(req, res, provider);
   137});
   138
   139router.put("/ai/providers/:provider/key", async (req, res) => {
   140  const { provider } = req.params;
   141  if (!isValidProvider(provider)) {
   142    return res.status(400).json({
   143      error: `Unknown provider "${provider}". Valid values: ${[...VALID_PROVIDERS].join(", ")}`,
   144    });
   145  }
   146  return handlePutKey(req, res, provider);
   147});
   148
   149router.delete("/ai/providers/:provider/key", async (req, res) => {
   150  const { provider } = req.params;
   151  if (!isValidProvider(provider)) {
   152    return res.status(400).json({
   153      error: `Unknown provider "${provider}". Valid values: ${[...VALID_PROVIDERS].join(", ")}`,
   154    });
   155  }
   156  return handleDeleteKey(req, res, provider);
   157});

   161/** GET /api/ai/active-provider — which provider will be used for this user */
   162router.get("/ai/active-provider", async (req, res) => {
   163  const resolved = await resolveProvider(req.userId);
   164  if (!resolved) return res.json({ provider: null, configured: false });
   165  return res.json({ provider: resolved.provider, configured: true });
   166});

   168// ── PR-011: Provider metrics ──────────────────────────────────────────────────
   169
   170/**
   171 * GET /api/ai/metrics
   172 *
   173 * PR-05/PR-06: Returns in-memory provider reliability metrics merged with
   174 * circuit-breaker state so the dashboard can show per-provider runtime health.
   175 *
   176 * Shape per entry:
   177 *   requests, failures, fallbackSuccesses, invalidModels, latency percentiles,
   178 *   successRate, lastSuccessAt, lastFailureAt, consecutiveFailures   (PR-05)
   179 *   circuitOpen, cooldownRemainingMs                                 (PR-07)
   180 *
   181 * Resets on process restart — for runtime observability, not persistent analytics.
   182 */
   183router.get("/ai/metrics", async (req, res) => {
   184  const metricsMap = new Map(getProviderMetrics().map((m) => [m.provider, m]));
   185  const configuredRows = await db
   186    .select({ provider: aiProviderCredentialsTable.provider })
   187    .from(aiProviderCredentialsTable)
   188    .where(eq(aiProviderCredentialsTable.ownerId, req.userId));
   189  const configured = new Set(configuredRows.map((row) => row.provider));
   190  // Server-managed keys are intentionally represented only as booleans.
   191  // Never include their values, prefixes, or lengths in the response.
   192  const serverConfigured = new Set(
   193    (["openrouter", "gemini", "deepseek", "groq"] as const).filter((provider) =>
   194      Boolean(process.env[`${provider.toUpperCase()}_API_KEY`]),
   195    ),
   196  );
   197  const catalog = getDynamicCatalogStatus();
   198
   199  // Merge circuit state into each metric snapshot; include all known providers
   200  // even if no requests have been recorded yet (so the UI always has an entry).
   201  const enriched = PROVIDER_PRIORITY.map((provider) => {
   202    const metric = metricsMap.get(provider) ?? {
   203      provider,
   204      requests: 0, failures: 0, fallbackSuccesses: 0, invalidModels: 0,
   205      p50LatencyMs: null, p95LatencyMs: null, avgLatencyMs: null,
   206      successRate: null, lastSuccessAt: null, lastFailureAt: null,
   207      consecutiveFailures: 0,
   208    };
   209    const circuit = getCircuitState(provider);
   210    const isConfigured = configured.has(provider) || serverConfigured.has(provider);
   211    const isCatalogStale =
   212      provider === "openrouter" &&
   213      isConfigured &&
   214      catalog.loaded &&
   215      (!catalog.usable || catalog.lastRefreshStatus === "failed" || catalog.lastRefreshStatus === "empty");
   216    const availabilityState = !isConfigured
   217      ? "missing_credentials"
   218      : circuit.open
   219        ? "circuit_open"
   220        : isCatalogStale
   221          ? "catalog_stale"
   222          : metric.requests === 0
   223            ? "unknown"
   224            : metric.consecutiveFailures > 0 || (metric.successRate !== null && metric.successRate < 0.8)
   225              ? "degraded"
   226              : "healthy";
   227    return {
   228      ...metric,
   229      configured: isConfigured,
   230      availabilityState,
   231      operatorAction: !isConfigured
   232        ? "Save an API key for this provider to enable it."
   233        : circuit.open
   234          ? "Wait for the cooldown to finish, then retry or configure another provider."
   235          : isCatalogStale
   236            ? "Retry shortly so the model catalog can refresh; configure another provider if it persists."
   237            : null,
   238      correlationId: randomUUID(),
   239      circuitOpen:         circuit.open,
   240      circuitHalfOpen:     circuit.halfOpen,
   241      cooldownRemainingMs: circuit.cooldownRemainingMs,
   242    };
   243  });
   244
   245  return res.json({
   246    metrics: enriched,
   247    catalog: {
   248      loaded: catalog.loaded,
   249      usable: catalog.usable,
   250      ageMs: catalog.ageMs,
   251      lastRefreshStatus: catalog.lastRefreshStatus,
   252    },
   253    behavioralScorecards: getBehavioralScorecards(),
   254  });
   255});
