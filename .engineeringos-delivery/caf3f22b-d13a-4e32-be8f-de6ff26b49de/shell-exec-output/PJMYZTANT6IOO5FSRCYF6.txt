   291/**
   292 * Keep the dashboard on the API's public validation contract. The server may
   293 * include compatibility fields on the outer event, but command output,
   294 * failure details, and changed-file lists are never accepted from the nested
   295 * payload.
   296 */
   297export function parseValidationEvent(raw: unknown): AiStreamValidationEvent | null {
   298  if (!raw || typeof raw !== 'object') return null;
   299  const event = raw as Record<string, unknown>;
   300  const validation = event.validation;
   301  if (!validation || typeof validation !== 'object') return null;
   302  const value = validation as Record<string, unknown>;
   303  if (
   304    typeof value.profile !== 'string'
   305    || typeof value.status !== 'string'
   306    || typeof value.scenario !== 'string'
   307    || (typeof value.exitCode !== 'number' && value.exitCode !== null)
   308    || !value.evidence
   309    || typeof value.evidence !== 'object'
   310    || typeof event.repairState !== 'string'
   311    || typeof event.attempt !== 'number'
   312    || typeof event.maxAttempts !== 'number'
   313  ) return null;
   314
   315  const evidence = value.evidence as Record<string, unknown>;
   316  if (
   317    typeof evidence.evidenceId !== 'string'
   318    || typeof evidence.observedAt !== 'string'
   319    || typeof evidence.artifactRef !== 'string'
   320  ) return null;
   321
   322  return {
   323    type: 'validation',
   324    validation: {
   325      profile: value.profile,
   326      status: value.status as PublicValidationResult['status'],
   327      scenario: value.scenario,
   328      exitCode: value.exitCode,
   329      evidence: {
   330        evidenceId: evidence.evidenceId,
   331        observedAt: evidence.observedAt,
   332        artifactRef: evidence.artifactRef,
   333      },
   334      ...(typeof value.detail === 'string' ? { detail: value.detail } : {}),
   335      ...(value.reasonCode === 'ownership' || value.reasonCode === 'invalid_profile' ||
   336        value.reasonCode === 'resource_limit' || value.reasonCode === 'stale_revision'
   337        ? { reasonCode: value.reasonCode as BrowserValidationBlockReason } : {}),
   338    },
   339    repairState: event.repairState as RepairLoopState,
   340    attempt: event.attempt,
   640    }>;
   641    reason?: string;
   642  };
   643  scopeExpansions?: Array<{
   644    kind: 'JUSTIFIED_SCOPE_EXPANSION' | 'UNJUSTIFIED_SCOPE_EXPANSION';
   645    path: string;
   646    matchedPolicyPath?: string;
   647  }>;
   648  unjustifiedReads?: string[];
   649};
