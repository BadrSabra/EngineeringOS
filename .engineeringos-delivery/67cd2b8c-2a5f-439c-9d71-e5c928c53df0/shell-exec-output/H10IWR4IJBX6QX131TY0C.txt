    74describe("provider health failure contract", () => {
    75  const provider = "openrouter" as ProviderId;
    76  const consoleUrl = "https://openrouter.ai/keys";
    77  const statusUrl = "https://status.openrouter.ai";
    78  const cases = [
    79    {
    80      name: "missing credentials",
    81      error: new GroqClientError("INVALID_CONFIG", "missing API key"),
    82      expected: "provider_outage",
    83    },
    84    {
    85      name: "authentication",
    86      error: new GroqClientError("AUTH_ERROR", "Authorization: Bearer sk-or-v1-secret"),
    87      expected: "authentication_failed",
    88    },
    89    {
    90      name: "incompatible model",
    91      error: new GroqClientError("MODEL_NOT_FOUND", "upstream model error: raw upstream detail"),
    92      expected: "incompatible_model",
    93    },
    94    {
    95      name: "stale catalog",
    96      error: new GroqClientError("INVALID_CONFIG", "catalog refresh failed: /srv/catalog.ts", {
    97        context: {
    98          providerCode: "NO_COMPATIBLE_FREE_MODEL",
    99          catalogStatus: "failed",
   100          catalogError: "raw upstream catalog response sk-or-v1-secret",
   101        },
   102      }),
   103      expected: "catalog_stale",
   104    },
   105    {
   106      name: "quota",
   107      error: new GroqClientError("QUOTA", "quota response contains raw upstream detail"),
   108      expected: "quota_exhausted",
   109    },
   110    {
   111      name: "rate limit",
   112      error: new GroqClientError("RATE_LIMITED", "429 raw upstream response"),
   113      expected: "rate_limited",
   114    },
   115    {
   116      name: "circuit open",
   117      error: new GroqClientError("MODEL_NOT_FOUND", "circuit detail raw upstream", {
   118        context: { providerCode: "CIRCUIT_OPEN" },
   119      }),
   120      expected: "circuit_open",
   121    },
   122    {
   123      name: "outage",
   124      error: new GroqClientError("SERVER_ERROR", "provider outage raw upstream"),
   125      expected: "provider_outage",
   126    },
   127  ] as const;
   128
   129  it.each(cases)("classifies $name with a safe operator action", ({ error, expected }) => {
   130    const projection = providerAvailabilityProjection(error, provider, consoleUrl, statusUrl);
   131    expect(projection.availabilityState).toBe(expected);
   132    expect(projection.operatorAction).toMatch(/retry|configure|replace|select|choose|wait|credits/i);
   133    expect(JSON.stringify(projection)).not.toContain("sk-or-v1-secret");
   134    expect(JSON.stringify(projection)).not.toContain("raw upstream");
   135    expect(JSON.stringify(projection)).not.toContain("/srv/catalog.ts");
   136  });
   137
   138  it("keeps circuit-open and stale-catalog failures distinct from incompatible models", () => {
   139    const stale = providerAvailabilityProjection(
   140      new GroqClientError("INVALID_CONFIG", "stale", {
   141        context: { providerCode: "NO_COMPATIBLE_FREE_MODEL", catalogStatus: "empty" },
   142      }),
   143      provider,
   144      consoleUrl,
   145      statusUrl,
   146    );
   147    const circuit = providerAvailabilityProjection(
   148      new GroqClientError("MODEL_NOT_FOUND", "circuit", {
   149        context: { providerCode: "CIRCUIT_OPEN" },
   150      }),
   151      provider,
   152      consoleUrl,
   153      statusUrl,
   154    );
   155    const incompatible = providerAvailabilityProjection(
   156      new GroqClientError("MODEL_NOT_FOUND", "model"),
   157      provider,
   158      consoleUrl,
   159      statusUrl,
   160    );
   161
   162    expect(stale.availabilityState).toBe("catalog_stale");
   163    expect(circuit.availabilityState).toBe("circuit_open");
   164    expect(incompatible.availabilityState).toBe("incompatible_model");
   165    expect(stale.catalogStatus).toBe("empty");
   166  });
   167
   168  it("serializes only safe provider reference details", () => {
   169    const error = new GroqClientError("INVALID_CONFIG", "raw /srv/provider.ts sk-or-v1-secret", {
   170      context: {
   171        providerCode: "NO_COMPATIBLE_FREE_MODEL",
   172        providerMessage: "raw upstream message",
   173        providerModel: "model-free",
   174        catalogStatus: "failed",
   175        catalogError: "raw catalog body",
   176      },
   177    });
   178    const context = error.toProviderContext();
   179    // The SSE route intentionally removes providerMessage before serializing
   180    // this context; keep that user-facing boundary explicit in the fixture.
   181    const publicContext = Object.fromEntries(
   182      Object.entries(context).filter(([key]) => key !== "providerMessage"),
   183    );
   184    const serialized = JSON.stringify(publicContext);
   185
   186    expect(serialized).toContain("NO_COMPATIBLE_FREE_MODEL");
   187    expect(serialized).toContain("failed");
   188    expect(serialized).not.toContain("raw upstream message");
   189    expect(serialized).not.toContain("raw catalog body");
   190    expect(serialized).not.toContain("sk-or-v1-secret");
