    84export function resolveExecutionModel(
    85  providerId: ProviderId,
    86  plan: ExecutionPlan,
    87): ExecutionModelDecision {
    88  const provider = loadProvider(providerId);
    89  // Capability probes are deliberately a fast, tool-capable contract. They
    90  // need exact source quoting, not a long reasoning trace or a large synthesis
    91  // envelope. Keeping this exception in model resolution prevents a probe
    92  // from inheriting task_execution's paid/reasoning-oriented tier merely
    93  // because its read-only tool loop lives inside chat-agent.
    94  const isCapabilityProbe = plan.qualityProfile === "capability_probe";
    95  const wantPowerful = isCapabilityProbe ? false : wantsPowerfulModel(plan);
    96  const capability = selectCapability(plan);
    97
    98  let decision: ExecutionModelDecision;
    99
   100  if (providerId === "openrouter") {
   101    const fallbackChain = resolveFallbackChain({
   102      capability,
   103      quality: wantPowerful ? "powerful" : "fast",
   104      requireTools: plan.strictHints.requireTools ?? false,
   105    }).map((model) => model.id);
   106
   107    // Controlled live checks may opt into a known paid OpenRouter model. Keep
   108    // this override environment-only so ordinary provider-free validation and
   109    // normal free-tier routing retain the catalog-driven fallback chain.
   110    const configuredModel = process.env.OPENROUTER_MODEL?.trim() || undefined;
   111    let liveModel: string | undefined;
   112    if (configuredModel) {
   113      try {
   114        liveModel = resolveFreeModelOverride(configuredModel, {
   115          capability,
   116          requireTools: plan.strictHints.requireTools ?? false,
   117        });
   118      } catch (error) {
   119        // An environment-pinned model can become paid, retired, or lose the
   120        // requested capability between deploys. Treat it as a stale hint and
   121        // use the already-resolved live fallback chain instead of making a
   122        // review fail before the provider is contacted.
   123        console.warn(
   124          JSON.stringify({
   125            scope: "model-resolver",
   126            action: "ignore_unusable_configured_model",
   127            configuredModel,
   128            capability,
   129            reason: error instanceof Error ? error.message : "model is not currently usable",
   130          }),
   131        );
   132      }
   133    }
   134    if (fallbackChain.length === 0 && !liveModel) {
   135      const catalog = getDynamicCatalogStatus();
   136      throw new GroqClientError(
   137        "INVALID_CONFIG",
   138        `No compatible free OpenRouter model is available for capability="${capability}"`,
   139        {
   140          context: {
   141            providerName: "OpenRouter",
   142            providerCode: "NO_COMPATIBLE_FREE_MODEL",
   143            catalogLoaded: catalog.loaded,
   144            catalogUsable: catalog.usable,
   145            catalogStatus: catalog.lastRefreshStatus,
   146            catalogError: catalog.lastRefreshError ?? undefined,
   147          },
   148        },
   149      );
   150    }
   151    const model = liveModel ?? fallbackChain[0]!;
   152    const powerModel = liveModel ?? fallbackChain[1] ?? fallbackChain[0]!;
   153
   154    decision = {
   155      providerId,
   156      model,
   157      powerModel,
   158      fallbackChain: liveModel ? [liveModel] : fallbackChain,
   159      capability,
   160      quality: wantPowerful ? "powerful" : "fast",
   161      source: "openrouter-catalog",
   162    };
   163  } else {
   164    const model = wantPowerful ? provider.defaultModels.powerful : provider.defaultModels.fast;
   165
   166    decision = {
   167      providerId,
   168      model,
   169      powerModel: provider.defaultModels.powerful,
   170      fallbackChain: [provider.defaultModels.fast, provider.defaultModels.powerful].filter((m, i, arr) => arr.indexOf(m) === i),
   171      capability,
   172      quality: wantPowerful ? "powerful" : "fast",
   173      source: "provider-registry",
   174    };
   175  }
