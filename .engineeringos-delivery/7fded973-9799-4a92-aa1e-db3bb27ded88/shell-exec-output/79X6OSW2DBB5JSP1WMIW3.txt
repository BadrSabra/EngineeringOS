   660/**
   661 * Planning must inspect the same source revision represented by the context
   662 * manifest. The bounded filesystem inventory is useful for grounding paths,
   663 * but it is not itself a source revision; the scanner walk is the immutable
   664 * revision contract shared with scan, approval, mutation, and delivery.
   665 *
   666 * A mismatch is represented as an unavailable planning inventory so the
   667 * planner's existing guarded fallback produces an explicitly blocked plan.
   668 * This avoids asking a provider to plan from a stale or partial snapshot.
   669 */
   670async function buildPlanningFilesystemContext(
   671  baseProjectContext: Awaited<ReturnType<typeof buildProjectContext>>,
   672  rootPath: string | undefined,
   673  message: string,
   674): Promise<Awaited<ReturnType<typeof buildProjectContext>>> {
   675  const unavailable = (reason: string) => ({
   676    status: "UNAVAILABLE" as const,
   677    files: [],
   678    directories: [],
   679    packageManifests: [],
   680    configFiles: [],
   681    truncated: false,
   682    reason,
   683  });
   684  const unavailableSources = (reason: string) => ({
   685    status: "UNAVAILABLE" as const,
   686    files: [],
   687    truncated: false,
   688    reason,
   689  });
   690
   691  const manifest = baseProjectContext.contextManifest;
   692  if (!rootPath) {
   693    return {
   694      ...baseProjectContext,
   695      filesystemManifest: unavailable("The project root is unavailable; planning is blocked until it is restored."),
   696      filesystemSources: unavailableSources("The project root is unavailable; no source snapshot can be verified."),
   697    };
   698  }
   699  if (!manifest || manifest.scanCompleteness !== "COMPLETE") {
   700    return {
   701      ...baseProjectContext,
   702      filesystemManifest: unavailable("The project scan is incomplete; complete a fresh scan before creating a plan."),
   703      filesystemSources: unavailableSources("The project scan is incomplete; source excerpts are not planning evidence."),
   704    };
   705  }
   706  // Narrow once after the completeness gate so all subsequent comparisons
   707  // use the exact manifest that authorized this planning attempt.
   708  const completeManifest = manifest;
   709
   710  const rootResult = await establishProjectRoot(rootPath);
   711  if (!rootResult.ok) {
   712    return {
   713      ...baseProjectContext,
   714      filesystemManifest: unavailable(`The project root is unavailable: ${rootResult.reason}.`),
   715      filesystemSources: unavailableSources("The project root could not be canonically established."),
   716    };
   717  }
   718
   719  const liveWalk = await walkProject(rootResult.canonicalPath).catch(() => undefined);
   720  if (!liveWalk) {
   721    return {
   722      ...baseProjectContext,
   723      filesystemManifest: unavailable("The project source tree could not be read; planning is blocked."),
   724      filesystemSources: unavailableSources("The project source tree could not be read."),
   725    };
   726  }
   727  if (liveWalk.truncated) {
   728    return {
   729      ...baseProjectContext,
   730      filesystemManifest: unavailable("The live source walk is incomplete; planning requires a complete snapshot."),
   731      filesystemSources: unavailableSources("The live source walk is incomplete."),
   732    };
   733  }
   734  if (liveWalk.revision !== completeManifest.projectRevision) {
   735    return {
   736      ...baseProjectContext,
   737      filesystemManifest: unavailable(
   738        "The source tree changed after the last scan; complete a fresh scan before creating a plan.",
   739      ),
   740      filesystemSources: unavailableSources(
   741        "The source tree revision is stale relative to the context snapshot.",
   742      ),
   743    };
   744  }
