    53export function resolveToolPolicy(opts: {
    54  provider: ProviderId;
    55  rootPath?: string;
    56  mode?: ToolMode;
    57  allowExecution?: boolean;
    58  allowAnalysis?: boolean;
    59}): ToolPolicy {
    60  const provider = getProvider(opts.provider);
    61  const mode = opts.mode ?? "workspace";
    62  const allowExecution = opts.allowExecution === true && mode === "workspace";
    63
    64  let policy: ToolPolicy;
    65
    66  if (!opts.rootPath) {
    67    policy = {
    68      provider: provider.providerId,
    69      rootPath: opts.rootPath,
    70      mode,
    71      enabled: false,
    72      allowFileRead: false,
    73      allowFileWrite: false,
    74      allowGit: false,
    75      allowExecution: false,
    76      allowAnalysis: false,
    77      reason: "tool policy requires a project root path",
    78    };
    79  } else if (!provider.supportsTools) {
    80    policy = {
    81      provider: provider.providerId,
    82      rootPath: opts.rootPath,
    83      mode,
    84      enabled: false,
    85      allowFileRead: false,
    86      allowFileWrite: false,
    87      allowGit: false,
    88      allowExecution: false,
    89      allowAnalysis: false,
    90      reason: "provider registry marks this endpoint as text-only",
    91    };
    92  } else {
    93    policy = {
    94      provider: provider.providerId,
    95      rootPath: opts.rootPath,
    96      mode,
    97      enabled: true,
    98      allowFileRead: true,
    99      allowFileWrite: mode === "workspace",
   100      allowGit: true,
   101      allowExecution,
   102      allowAnalysis: opts.allowAnalysis === true,
   103    };
   104  }
   105
   106  console.info(
   107    JSON.stringify({
   108      scope: "tool-policy",
   109      action: "resolve_tool_policy",
   110      provider: policy.provider,
   111      rootPath: opts.rootPath ?? null,
   112      mode,
   113      enabled: policy.enabled,
   114      allowFileRead: policy.allowFileRead,
   115      allowFileWrite: policy.allowFileWrite,
   116      allowGit: policy.allowGit,
   117      allowExecution: policy.allowExecution,
   118      allowAnalysis: policy.allowAnalysis,
   119      reason: policy.reason ?? null,
   120      supportsTools: provider.supportsTools,
   121    }),
   122  );
   123
   124  return policy;
   125}

   127export function isToolAllowed(policy: ToolPolicy, toolName: string): boolean {
   128  if (!policy.enabled) return false;
   129  if (FILE_READ_TOOL_NAMES.has(toolName)) return policy.allowFileRead;
   130  if (FILE_WRITE_TOOL_NAMES.has(toolName)) return policy.allowFileWrite;
   131  if (GIT_TOOL_NAMES.has(toolName)) return policy.allowGit;
   132  if (EXECUTION_TOOL_NAMES.has(toolName)) return policy.allowExecution;
   133  if (ANALYSIS_TOOL_NAMES.has(toolName)) return policy.allowAnalysis;
   134  return false;
   135}

   137export function getAllowedToolDefinitions(policy: ToolPolicy): ToolDefinitionLike[] {
   138  if (!policy.enabled) return [];
   139  return [...ALL_TOOL_DEFINITIONS, ...ANALYSIS_TOOL_DEFINITIONS]
   140    .filter((tool) => isToolAllowed(policy, tool.function.name));
   141}

   143/**
   144 * Server-side post-model-turn authorization. Prompt text and repository data
   145 * are never inputs to this decision. An absent manifest is fail-closed.
   146 */
   147export function authorizeToolInvocation(opts: {
   148  toolName: string;
   149  args?: Record<string, unknown>;
   150  allowedTools?: ReadonlySet<string>;
   151  approvedFilePaths?: readonly string[];
   152  approvedValidationProfiles?: readonly string[];
   153  approvalState?: "APPROVED" | "PENDING_APPROVAL" | "REJECTED";
   154}): ToolAuthorization {
   155  const known = new Set([...FILE_READ_TOOL_NAMES, ...FILE_WRITE_TOOL_NAMES, ...GIT_TOOL_NAMES, ...EXECUTION_TOOL_NAMES, ...ANALYSIS_TOOL_NAMES]);
   156  if (!known.has(opts.toolName)) return { allowed: false, reason: "unknown_tool" };
   157  if (opts.allowedTools && !opts.allowedTools.has(opts.toolName)) {
   158    return { allowed: false, reason: "tool_not_in_manifest" };
   159  }
   160  const isWrite = FILE_WRITE_TOOL_NAMES.has(opts.toolName);
   161  if (isWrite && opts.approvalState !== undefined && opts.approvalState !== "APPROVED") {
   162    return { allowed: false, reason: "approval_required" };
   163  }
   164  const requestedPath = typeof opts.args?.path === "string"
   165    ? opts.args.path.replaceAll("\\", "/").replace(/^(\.\/)+/, "")
   166    : undefined;
   167  if (isWrite && opts.approvedFilePaths && (
   168    !requestedPath || !opts.approvedFilePaths.includes(requestedPath)
   169  )) {
   170    return { allowed: false, reason: "path_outside_approved_scope" };
   171  }
   172    if ((opts.toolName === "run_validation" || opts.toolName === "run_command" || opts.toolName === "run_browser_validation") &&
   173      opts.approvedValidationProfiles &&
   174      !opts.approvedValidationProfiles.includes(String(opts.args?.profile ?? "").trim())) {
   175    return { allowed: false, reason: "validation_profile_not_approved" };
   176  }
   177  return { allowed: true, reason: "allowed" };
   178}
