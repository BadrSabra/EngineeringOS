     1  // project root is inaccessible. Such a fallback would let a user authorized
     2  // for a stale/inaccessible project request arbitrary files (e.g. .env) from
     3  // OTHER projects living in the shared workspace — a cross-project data leak.
     4  // Here the project's own configured rootPath is the ONLY authorization
     5  // boundary: if it is unusable, we return `available:false` and serve nothing.
     6  const storedRoot = project.rootPath;
     7  if (!storedRoot) {
     8    return res.status(200).json({ available: false, reason: "project_root_unavailable" });
     9  }
    10
    11  // Canonicalize the root FIRST (realpath resolves its own symlinks) so both
    12  // a real absolute path and a symlinked project root resolve to the same
    13  // canonical directory, and containment can be checked against that truth.
    14  let resolvedRoot: string;
    15  try {
    16    resolvedRoot = await fs.realpath(storedRoot);
    17  } catch {
    18    return res.status(200).json({ available: false, reason: "project_root_unavailable" });
    19  }
    20
    21  const resolved = path.resolve(resolvedRoot, relPath);
    22  if (resolved !== resolvedRoot && !resolved.startsWith(resolvedRoot + path.sep)) {
    23    return res.status(200).json({ available: false, reason: "outside_project_root" });
    24  }
    25
    26  let realResolved: string;
    27  try {
    28    // realpath follows symlinks in the target — catches a symlink pointing
    29    // outside the (canonical) project root.
    30    realResolved = await fs.realpath(resolved);
    31  } catch {
    32    return res.status(200).json({ available: false, reason: "file_not_found" });
    33  }
    34  if (realResolved !== resolvedRoot && !realResolved.startsWith(resolvedRoot + path.sep)) {
    35    return res.status(200).json({ available: false, reason: "outside_project_root" });
    36  }
    37
    38  let rawFile: string;
    39  try {
    40    rawFile = await fs.readFile(realResolved, "utf-8");
    41  } catch {
    42    return res.status(200).json({ available: false, reason: "file_unreadable" });
    43  }
    44
    45  const allLines = rawFile.split(/\r?\n/);
    46  // A file typically ends with a trailing newline, which split() turns into a
    47  // spurious empty line; drop a single trailing empty line so line counts line
    48  // up with an editor's visible line numbers.
    49  if (allLines.length > 0 && allLines[allLines.length - 1] === "") allLines.pop();
    50  // Bound the requested window so a single huge file never floods the response.
    51  const MAX_WINDOW = 200;
    52  const to = Math.min(allLines.length, endLine, startLine + MAX_WINDOW - 1);
    53
    54  return res.status(200).json({
    55    available: true,
    56    path: relPath,
    57    startLine,
    58    endLine: to,
    59    fileLines: allLines.length,
    60    truncated: to < endLine || to < allLines.length,
    61    lines: allLines.slice(startLine - 1, to).map((text, i) => ({
    62      line: startLine + i,
    63      text,
    64    })),
    65  });

   533                  }),
   534                }
   535              : undefined,
   536          });

   538        }

   540        rebasedChanges.push({
   541          ...change,
   542          absolutePath: realResolved,
   543          originalContent: currentContent,
   544          baseHash: actualHash,
   545          newContent: rebased.content,
   546          hunks: buildPatchHunks(currentContent, rebased.content, change.reason, {
   547            risk: change.risk,
   548            evidence: change.evidence,
   549          }),
   550        });
   551      } catch (error) {
   552        conflicts.push({
   553          path: change.path,
   554          ok: false,
   555          code: "PATCH_REBASE_UNAVAILABLE",
   556          error: error instanceof Error ? error.message : String(error),
   557        });
   558      }
   559    }
   560
   561    if (conflicts.length > 0 || rebasedChanges.length !== approvedChanges.length) {
   562      return res.status(409).json({
   563        error: "Patch could not be safely rebased. No proposal data was changed.",
   564        code: "PATCH_REBASE_CONFLICT",
   565        results: [
   566          ...conflicts,
   567          ...rebasedChanges.map((change) => ({
   568            path: change.path,
   569            ok: false as const,
   570            code: "PATCH_REBASE_ABORTED" as const,
   571            error: "Batch rebase aborted because another file conflicted.",
   572          })),
   573        ],
   574      });
   575    }
   576
   577    await db.transaction(async (tx) => {
   578      const nextRevision = proposal.revision + 1;
   579      const updated = await tx
   580        .update(aiChangeProposalsTable)
   581        .set({
   582          changes: serializeServerPendingChanges(rebasedChanges),
   583          revision: nextRevision,
   584          approvalRequired: true,
   585        })
   586        .where(and(
   587          eq(aiChangeProposalsTable.id, proposalId),
   588          eq(aiChangeProposalsTable.projectId, projectId),
   589          eq(aiChangeProposalsTable.status, "pending"),
   590        ))
   591        .returning({ id: aiChangeProposalsTable.id });
   592      if (!updated[0]) throw new Error("Change proposal was consumed during rebase");
   593      await tx.insert(eventsTable).values({
   594        id: randomUUID(),
   595        type: "AiPatchRebased",
   596        projectId,
   597        severity: "info",
   598        message: `AI patch rebased for ${rebasedChanges.length} file${rebasedChanges.length === 1 ? "" : "s"}; user review is required again.`,
   599        correlationId: proposalId,
   600        payload: {
   601          proposalId,
   602          rebasedFiles: rebasedChanges.map((change) => change.path),
   603          approvalRequired: true,
   604        },
   605      });
   606    });
   607
   608    return res.status(200).json({
   609      proposalId,
   610      changes: rebasedChanges,
   611      rebasedFiles: rebasedChanges.map((change) => change.path),
   612      approvalRequired: true,
   613      revision: proposal.revision + 1,
   614    });
   615  } catch (error) {
   616    if (error instanceof Error && error.message === "Change proposal was consumed during rebase") {
   617      return res.status(409).json({
   618        error: "Change proposal has already been consumed",
   619        code: "PROPOSAL_ALREADY_CONSUMED",
   620      });
   621    }
   622    throw error;
   623  } finally {
   624    await applyLock.release();
   625  }
   626});

   628// ── POST /api/ai/chat/proposals/:proposalId/approve ──────────────────────────
   629//
   630// Rebased proposals remain pending but are explicitly blocked from apply until
   631// the user approves the new revision. The compare-and-set prevents an approval
   632// from racing with another rebase of the same proposal.
   633router.post("/ai/chat/proposals/:proposalId/approve", async (req, res) => {
   634  const body = z.object({
   635    projectId: z.string().min(1),
   636    revision: z.number().int().min(0),
   637  }).strict().safeParse(req.body);
   638  if (!body.success) {
   639    return res.status(400).json({
   640      error: body.error.issues[0]?.message ?? "Invalid proposal approval request",
   641      code: "PROPOSAL_APPROVAL_INVALID",
   642    });
   643  }
   644
   645  const { projectId, revision } = body.data;
   646  const project = await loadProjectByIdForUser(projectId, req.userId, res);
   647  if (!project) return;
   648
   649  const [proposal] = await db
   650    .select()
   651    .from(aiChangeProposalsTable)
   652    .where(and(
   653      eq(aiChangeProposalsTable.id, req.params.proposalId),
   654      eq(aiChangeProposalsTable.projectId, projectId),
   655    ))
   656    .limit(1);
   657  if (!proposal) {
   658    return res.status(404).json({ error: "Change proposal not found", code: "PROPOSAL_NOT_FOUND" });
   659  }
   660  if (proposal.status !== "pending") {
   661    return res.status(409).json({
   662      error: "Change proposal has already been consumed",
   663      code: "PROPOSAL_ALREADY_CONSUMED",
   664    });
   665  }
   666  if (!proposal.approvalRequired) {
   667    return res.status(409).json({
   668      error: "This proposal does not require re-approval",
   669      code: "PROPOSAL_REAPPROVAL_NOT_REQUIRED",
   670      approvalRequired: false,
   671      revision: proposal.revision,
   672    });
   673  }
   674
   675  const updated = await db.transaction(async (tx) => {
   676    const rows = await tx
   677      .update(aiChangeProposalsTable)
   678      .set({ approvalRequired: false })
   679      .where(and(
   680        eq(aiChangeProposalsTable.id, proposal.id),
   681        eq(aiChangeProposalsTable.projectId, projectId),
   682        eq(aiChangeProposalsTable.status, "pending"),
   683        eq(aiChangeProposalsTable.approvalRequired, true),
   684        eq(aiChangeProposalsTable.revision, revision),
   685      ))
   686      .returning({ id: aiChangeProposalsTable.id });
   687    if (!rows[0]) return false;
   688
   689    await tx.insert(eventsTable).values({
   690      id: randomUUID(),
   691      type: "AiPatchReapproved",
   692      projectId,
   693      severity: "success",
   694      message: "AI rebased patch approved for apply",
   695      correlationId: proposal.messageId,
   696      payload: {
   697        proposalId: proposal.id,
   698        revision,
   699        approvalRequired: false,
   700      },
   701    });
   702    return true;
   703  });
   704
   705  if (!updated) {
   706    return res.status(409).json({
   707      error: "The proposal changed before it could be approved; reload and review it again",
   708      code: "PROPOSAL_APPROVAL_CONFLICT",
   709    });
   710  }
   711
   712  return res.status(200).json({
   713    proposalId: proposal.id,
   714    approvalRequired: false,
   715    revision,
   716  });
   717});
