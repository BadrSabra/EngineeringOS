   125      content: "",
   126      outcome: "FAILED",
   127      errorCode: params.errorCode,
   128      errorMessage: redactUserFacingText(params.errorMessage).slice(0, 500),
   129      toolTrace: trace,
   130      createdAt: now,
   131    });
   132    await tx.update(aiChatSessionsTable)
   133      .set({ updatedAt: now })
   134      .where(eq(aiChatSessionsTable.id, sessionId!));
   135  });
   136  return sessionId!;

   211function beginTaskStream(
   212  res: import("express").Response,
   213  task: StructuredTask,
   214  projectId: string,
   215  metadata: StructuredAuditMetadata,
   216) {
   217  res.status(200).set({
   218    "Content-Type": "text/event-stream",
   219    "Cache-Control": "no-cache, no-transform",
   220    Connection: "keep-alive",
   221    "X-Accel-Buffering": "no",
   222  });
   223  res.flushHeaders();
   224
   225  let closed = false;
   226  const heartbeat = setInterval(() => {
   227    if (!closed && !res.writableEnded) res.write(": heartbeat\n\n");
   228  }, 15_000);
   229
   230  const emit = (event: StructuredTaskEvent) => {
   231    if (!closed && !res.writableEnded) {
   232      res.write(`data: ${JSON.stringify({ ...event, ...auditEnvelope(metadata) })}\n\n`);
   233    }
   234  };
   235
   236  const close = () => {
   237    if (closed) return;
   238    closed = true;
   239    clearInterval(heartbeat);
   240    if (!res.writableEnded) res.end();
   241  };
   242
   243  res.once("close", () => {
   244    closed = true;
   245    clearInterval(heartbeat);
   246  });
   247
   248  metadata.operationalTrace.push({ stage: "task", status: "started" });
   249  emit({ type: "task_started", task, projectId });
   250  return { emit, close };
   251}
