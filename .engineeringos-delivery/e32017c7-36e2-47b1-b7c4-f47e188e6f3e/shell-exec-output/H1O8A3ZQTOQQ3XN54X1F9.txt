   516export async function requireProvider(

   518  res: import("express").Response,

   526    res.status(428).json({
   527      error: "AI provider not configured",
   528      hint,

   531      correlationId: randomUUID(),
   532    });
   533    return null;
   534  }
   535  return resolved;

   409router.post("/ai/projects/:projectId/review", requireProjectAccess, async (req, res) => {
   410  const projectId = req.params.projectId as string;
   411  const project = req.project;
   412  if (!project) return res.status(500).json({ error: "Project context unavailable" });
   413  const metadata = await createAuditMetadata(projectId, project);
   414  recordTrace(metadata, "review", "started");
   415  const { fileContents } = req.body as { fileContents?: Record<string, string> };
   416
   417  logger.info({ projectId }, "AI code review requested");
   418
   419  const MAX_FILE_CONTENTS_BYTES = 50_000;
   420  if (fileContents) {
   421    const invalidKey = Object.keys(fileContents).find(
   422      (k) => path.isAbsolute(k) || k.includes(".."),
   423    );
   424    if (invalidKey) {
   425      return res.status(400).json({
   426        error: `fileContents key "${invalidKey}" must be a relative path without traversal (no ".." segments)`,
   427      });
   428    }
   429    const totalSize = Object.values(fileContents).reduce((sum, v) => sum + v.length, 0);
   430    if (totalSize > MAX_FILE_CONTENTS_BYTES) {
   431      return res.status(413).json({
   432        error: `fileContents total size (${Math.round(totalSize / 1_000)} KB) exceeds the ${MAX_FILE_CONTENTS_BYTES / 1_000} KB limit — send fewer or smaller files`,
   433      });
   434    }
   435  }
   436
   437  const rlReview = await checkProjectRateLimitDb(projectId);
   438  if (!rlReview.allowed) {
   439    return res.status(429).json({
   440      error: "The AI provider is temporarily rate-limited.",
   441      code: "RATE_LIMITED",
   442      hint: `Wait ${rlReview.retryAfterSec}s, then retry the code review.`,
   443      retryable: true,
   444      failureKind: "RATE_LIMIT",
   445    });
   446  }
   447
   448  const providerResolved = await requireProvider(req.userId, res, {
   449    qualityProfile: "code_review",
   450  });
   451  if (!providerResolved) return;
   452  const { provider, apiKey } = providerResolved;
   453
   454  const projectContext = await buildProjectContext(projectId, {
   455    sections: ["tasks", "metrics", "graphEntities", "graphRelationships", "events"],
   456  });
   457  let result: Awaited<ReturnType<typeof reviewCode>>;
   458  let effectiveProvider = provider;
   459  try {
   460    ({ result, effectiveProvider } = await runAgentWithFallback(
   461      req.userId,
   462      { provider, apiKey },
   463      (opts) => reviewCode(projectContext, fileContents, opts),
   464      { qualityProfile: "code_review" },
   465    ));
   466  } catch (err) {
   467    metadata.incomplete = true;
   468    recordTrace(metadata, "review", "failed", effectiveProvider);
   469    if (handleOrchestratorError(err, res, { projectId, operation: "code-review", provider: effectiveProvider })) return;
   470    throw err;
   471  }
   472
   473  // When the model output failed to parse, we still have the fallback data.
   474  // Serve it with a warning header so the UI can display something useful.
   475  if (result._parseError) {
   476    metadata.incomplete = true;
   477    recordTrace(metadata, "review", "incomplete", effectiveProvider);
   478    logger.warn(
   479      { projectId, parseCode: result._parseError.code, message: result._parseError.message, provider: effectiveProvider },
   480      "code-reviewer: parse error",
   481    );
   482    return res.status(422).json({
   483      ...auditEnvelope(metadata),
   484      error: "model_output_invalid",
   485      code: "model_output_invalid",
   486      hint: "The AI model returned an unexpected response — try again in a moment.",
   487      parseCode: result._parseError.code,
   488    });
   489  }
   490
   491  invalidateContextCache(projectId);
   492
   493  await db.transaction(async (tx) => {
   494    await tx.insert(auditLogsTable).values({
   495      id: randomUUID(),
   496      entityType: "project",
   497      entityId: projectId,
   498      action: "ai_reviewed",
   499      projectId,
   500      actor: req.userId,
   501      stateBefore: {},
   502      stateAfter: { verdict: result.verdict, overallScore: result.overallScore },
   503    });
   504    await tx.insert(eventsTable).values({
   505      id: randomUUID(),
   506      type: "AiCodeReviewCompleted",
   507      projectId,
   508      severity: result.verdict === "approved" ? "success" : "warning",
   509      message: `AI code review: ${result.verdict} (score: ${result.overallScore}/100)`,
   510    });
   511  });
   512
   513  recordTrace(metadata, "review", "completed", effectiveProvider);
   514  return res.json({ ...redactUserFacingValue(result) as Record<string, unknown>, ...auditEnvelope(metadata) });
   515});
