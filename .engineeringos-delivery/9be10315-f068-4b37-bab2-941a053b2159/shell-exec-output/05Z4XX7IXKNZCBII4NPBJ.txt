     1/**
     2 * Provider-free release coverage for structured code-review fallback.
     3 *
     4 * These fixtures deliberately cross the OpenRouter HTTP boundary instead of
     5 * mocking agentComplete. That keeps the check useful when model selection,
     6 * catalog refresh, transport classification, or reviewer parsing changes.
     7 */
     8import { afterEach, beforeEach, describe, expect, it, vi } from "vitest";
     9import { reviewCode } from "../agents/code-reviewer.js";
    10import { _resetForTest, getDynamicCatalogStatus } from "../openrouter/dynamic-catalog.js";
    11import { FREE_MODELS } from "../openrouter/model-catalog.js";

    13const selectedFile = "src/example.ts";
    14const reviewContext = {
    15  project: "Small TypeScript utility project",
    16  recentTasks: "No recent tasks",
    17  latestMetrics: "No metrics available",
    18  graphSummary: "No graph entities",
    19  recentEvents: "No recent events",
    20  workflows: "No workflows",
    21  metricsVerified: false,
    22};

    24const reviewWithFinding = JSON.stringify({
    25  summary: "The selected file contains a small maintainability issue.",
    26  overallScore: 70,
    27  strengths: [
    28    "The function has a clear name.",
    29    "The selected module has a narrow responsibility.",
    30    "The current behavior is easy to reproduce.",
    31  ],
    32  issues: [{
    33    type: "style",
    34    severity: "low",
    35    file: selectedFile,
    36    title: "Use a named constant",
    37    description: "The value is declared inline.",
    38    suggestion: "Give the value a descriptive name.",
    39  }],
    40  refactoringOpportunities: ["Extract the inline value into a named constant."],
    41  securityConcerns: ["No security concern was found in the selected file."],
    42  verdict: "needs_changes",
    43});

    45const reviewModels = FREE_MODELS
    46  .filter((model) => model.capabilities.includes("json"))
    47  .slice(-3);
    48
    49if (reviewModels.length < 3) {
    50  throw new Error("The structured-review release fixture requires three JSON-capable catalog models.");
    51}
    52
    53type FixtureResponse = {
    54  ok: boolean;
    55  status: number;
    56  headers?: Headers;
    57  json: () => Promise<unknown>;
    58  text: () => Promise<string>;
    59};
    60
    61function jsonResponse(body: unknown, status = 200, headers?: Headers): FixtureResponse {
    62  return {
    63    ok: status >= 200 && status < 300,
    64    status,
    65    headers,
    66    json: async () => body,
    67    text: async () => JSON.stringify(body),
    68  };
    69}
    70
    71function catalogResponse(): FixtureResponse {
    72  return jsonResponse({
    73    data: reviewModels.map((model) => ({
    74      id: model.id,
    75      pricing: { prompt: "0", completion: "0" },
    76    })),
    77  });
    78}
    79
    80function installOpenRouterFixture(
    81  completion: (model: string, callNumber: number) => FixtureResponse,
    82  options: { catalog?: () => Promise<FixtureResponse> } = {},
    83) {
    84  const completionCalls: string[] = [];
    85  const fetchMock = vi.fn(async (url: string | URL, init?: RequestInit) => {
    86    if (String(url).endsWith("/models")) {
    87      return options.catalog ? await options.catalog() : catalogResponse();
    88    }
    89
    90    const body = JSON.parse(String(init?.body ?? "{}")) as { model?: string };
    91    const model = String(body.model ?? "");
    92    completionCalls.push(model);
    93    return completion(model, completionCalls.length);
    94  });
    95  vi.stubGlobal("fetch", fetchMock);
    96  return { completionCalls, fetchMock };
    97}

    99async function runReview() {
   100  return reviewCode(
   101    reviewContext,
   102    { [selectedFile]: "export const value = 1;" },
   103    {
   104      provider: "openrouter",
   105      apiKey: "provider-free-fixture-key",
   106      qualityProfile: "code_review",
   107    },
   108  );
   109}
   110
   111describe("structured review provider fallback release check", () => {
   112  beforeEach(() => {
   113    _resetForTest();
   114  });
   115
   116  afterEach(() => {
   117    vi.unstubAllGlobals();
   118    vi.restoreAllMocks();
   119  });
   120
   121  const transportFailureCases: Array<{
   122    name: string;
   123    firstResponse: () => FixtureResponse;
   124    failureResponseCount?: number;
   125  }> = [
   126    {
   127      name: "reasoning-only response",
   128      firstResponse: () => jsonResponse({
   129        choices: [{
   130          message: {
   131            content: null,
   132            reasoning_content: "The model reasoned but did not produce a final answer.",
   133          },
   134        }],
   135      }),
   136    },
   137    {
   138      name: "agent-harness response",
   139      firstResponse: () => jsonResponse({
   140        error: {
   141          code: "agent_only",
   142          message: "This model is only available through an agent harness.",
   143        },
   144      }, 400),
   145      // JSON-mode incompatibility is retried once without response_format
   146      // before the fallback chain advances.
   147      failureResponseCount: 2,
   148    },
   149    {
   150      name: "rate-limit response",
   151      firstResponse: () => jsonResponse({
   152        error: { message: "Too many requests; try again later." },
   153      }, 429, new Headers({ "Retry-After": "1" })),
   154    },
   155    {
   156      name: "empty response",
   157      firstResponse: () => jsonResponse({
   158        choices: [{ message: { content: "" } }],
   159      }),
   160    },
   161  ];
   162
   163  it.each(transportFailureCases)(
   164    "falls back after a $name and returns a cited finding",
   165    async ({ firstResponse, failureResponseCount = 1 }) => {
   166      const { completionCalls } = installOpenRouterFixture((model, callNumber) =>
   167        callNumber <= failureResponseCount ? firstResponse() : jsonResponse({
   168          choices: [{ message: { content: reviewWithFinding } }],
   169          model,
   170          usage: {},
   171        }),
   172      );
   173
   174      const result = await runReview();
   175
   176      expect(result._parseError).toBeUndefined();
   177      expect(result.issues).toEqual(expect.arrayContaining([
   178        expect.objectContaining({ file: selectedFile }),
   179      ]));
   180      // The reviewer may spend its remaining quality-retry budget on the
   181      // successful fallback response. The transport fallback itself must
   182      // still be the first transition to a different model.
   183      expect(completionCalls.length).toBeGreaterThanOrEqual(failureResponseCount + 1);
   184      expect(completionCalls.some((model, index) => index > 0 && model !== completionCalls[0])).toBe(true);
   185    },
   186  );
   187
   188  it("marks malformed structured output terminally incomplete instead of completing a review", async () => {
   189    const { completionCalls } = installOpenRouterFixture(() =>
   190      jsonResponse({
   191        choices: [{ message: { content: "{ malformed review output" } }],
   192      }),
   193    );
   194
   195    const result = await runReview();
   196
   197    expect(result._parseError).toMatchObject({ code: "MALFORMED_JSON" });
   198    expect(result._parseError?.message).toBeTruthy();
   199    expect(result.issues).toEqual([]);
   200    expect(completionCalls.length).toBeGreaterThanOrEqual(2);
   201  });
   202});
   203
   204describe("structured review catalog refresh release check", () => {
   205  beforeEach(() => {
   206    _resetForTest();
   207  });
   208
   209  afterEach(() => {
   210    vi.unstubAllGlobals();
   211    vi.restoreAllMocks();
   212  });
   213
   214  const catalogFailureCases: Array<{
   215    name: string;
   216    catalog: () => Promise<FixtureResponse>;
   217    expectedStatus: "failed" | "empty";
   218  }> = [
   219    {
   220      name: "network failure",
   221      catalog: async () => {
   222        throw new Error("fixture catalog unavailable");
   223      },
   224      expectedStatus: "failed",
   225    },
   226    {
   227      name: "non-200 response",
   228      catalog: async () => jsonResponse({ error: "catalog unavailable" }, 503),
   229      expectedStatus: "failed",
   230    },
   231    {
   232      name: "empty response",
   233      catalog: async () => jsonResponse({ data: [] }),
   234      expectedStatus: "empty",
   235    },
   236    {
   237      name: "malformed response",
   238      catalog: async () => ({
   239        ok: true,
   240        status: 200,
   241        json: async () => {
   242          throw new Error("fixture catalog JSON is malformed");
   243        },
   244        text: async () => "{ malformed",
   245      }),
   246      expectedStatus: "failed",
   247    },
   248  ];
   249
   250  it.each(catalogFailureCases)(
   251    "keeps static compatibility routing after a $name",
   252    async ({ catalog, expectedStatus }) => {
   253      const { completionCalls } = installOpenRouterFixture(
   254        (model) => jsonResponse({
   255          choices: [{ message: { content: reviewWithFinding } }],
   256          model,
   257          usage: {},
   258        }),
   259        { catalog },
   260      );
   261
   262      const result = await runReview();
   263      const catalogStatus = getDynamicCatalogStatus();
   264
   265      expect(result._parseError).toBeUndefined();
   266      expect(result.issues[0]?.file).toBe(selectedFile);
   267      // A valid structured review can receive bounded quality retries; the
   268      // catalog failure must not add another model or prevent completion.
   269      expect(completionCalls.length).toBeGreaterThanOrEqual(1);
   270      expect(new Set(completionCalls)).toHaveProperty("size", 1);
   271      expect(catalogStatus.lastRefreshStatus).toBe(expectedStatus);
   272      expect(catalogStatus.usable).toBe(false);
   273    },
   274  );
   275});
