  1090/**
  1091 * Non-streaming completion via OpenRouter.
  1092 *
  1093 * OR-003: trims context to `OPENROUTER_MAX_MESSAGES` non-system turns and
  1094 *         defaults `maxTokens` to `OPENROUTER_DEFAULT_MAX_TOKENS` (2 048).
  1095 * OR-005: retries once (after 1.5 s back-off) on transient errors (429, 5xx,
  1096 *         timeout, network) before propagating — avoids needless provider
  1097 *         fallback for brief free-tier blips.
  1098 */
  1099export async function openrouterCompleteRaw(
  1100  messages: RawMessage[],
  1101  opts: Omit<OpenAICompatibleOptions, "baseUrl" | "providerName" | "extraHeaders">,
  1102): Promise<RawGroqResponse> {
  1103  const trimmed = trimMessagesForOpenRouter(messages);
  1104  const fullOpts: OpenAICompatibleOptions = {
  1105    maxTokens: OPENROUTER_DEFAULT_MAX_TOKENS,
  1106    ...opts,
  1107    baseUrl: OPENROUTER_BASE_URL,
  1108    providerName: "OpenRouter",
  1109    extraHeaders: OPENROUTER_EXTRA_HEADERS,
  1110  };
  1111
  1112  try {
  1113    return await oacCompleteRaw(trimmed, fullOpts);
  1114  } catch (err) {
  1115    // OpenRouter-compatible models do not all implement JSON mode consistently.
  1116    // A 400 caused by response_format must not consume the whole Recovery chain:
  1117    // retry the same model once with the prompt-only contract.
  1118    if (
  1119      fullOpts.responseFormat &&
  1120      err instanceof GroqClientError &&
  1121      (err.code === "NON_200" || err.code === "MODEL_UNAVAILABLE") &&
  1122      err.providerStatus === 400
  1123    ) {
  1124      console.warn(
  1125        JSON.stringify({
  1126          scope: "openrouter-client",
  1127          code: "JSON_MODE_UNSUPPORTED",
  1128          model: fullOpts.model,
  1129          action: "retry_without_response_format",
  1130        }),
  1131      );
  1132      return oacCompleteRaw(trimmed, { ...fullOpts, responseFormat: undefined });
  1133    }
  1134    if (
  1135      err instanceof GroqClientError &&
  1136      err.code === "RATE_LIMITED" &&
  1137      err.retryAfterMs !== undefined
  1138    ) {
  1139      throw err;
  1140    }
  1141    if (!isTransientError(err) || opts.retryTransient === false) throw err;
  1142    console.warn(
  1143      JSON.stringify({
  1144        scope: "openrouter-client",
  1145        code: "TRANSIENT_RETRY",
  1146        errorCode: err.code,
  1147        backoffMs: 1500,
  1148      }),
  1149    );
  1150    await sleep(1500);
  1151    return oacCompleteRaw(trimmed, fullOpts);
  1152  }
  1153}

  1175/**
  1176 * Non-streaming completion via OpenRouter with automatic free-model fallback
  1177 * (STORY-03).
  1178 *
  1179 * PR-003: MODEL_UNAVAILABLE (410/422) is now also fallback-worthy in addition
  1180 * to MODEL_NOT_FOUND (404/402/400). EMPTY_RESPONSE is also fallback-worthy
  1181 * because a silent completion identifies an unusable model candidate, not the
  1182 * whole OpenRouter provider.
  1183 *
  1184 * RC-04: when `model` is not provided, resolves the best available free model
  1185 * from the live catalog at call time using the `quality` + `capability` hints.
  1186 */
  1187export async function openrouterCompleteWithFallback(
  1188  messages: RawMessage[],
  1189  opts: OpenRouterFallbackOptions,
  1190): Promise<RawGroqResponse> {
  1191  const initialModel = opts.model;
  1192  const capability = opts.capability ?? "chat";
  1193  const requireTools = opts.requireTools ?? !!(opts.tools?.length);
  1194  const knownInitialModel = initialModel
  1195    ? FREE_MODELS.some((candidate) => candidate.id === initialModel)
  1196    : true;
  1197  if (initialModel && !knownInitialModel) {
  1198    throw new GroqClientError(
  1199      "INVALID_CONFIG",
  1200      `OpenRouter configured model is not a known free catalog model: ${initialModel}`,
  1201      { context: { providerName: "OpenRouter", providerModel: initialModel, providerCode: "STALE_CONFIGURED_MODEL" } },
  1202    );
  1203  }
  1204
  1205  // RC-04 / PR-01: when no model is specified, use the resolver to build a full
  1206  // fallback chain from the live free-tier catalog at call time.
  1207  // `quality` and `capability` let callers express intent (e.g. "powerful" for
  1208  // code-review) without hardcoding a specific model ID that may become stale.
  1209  const resolvedChain = initialModel
  1210    ? buildFallbackChainFromId(initialModel, opts.taskType, { capability, requireTools })
  1211    : resolveFallbackChain({
  1212        capability,
  1213        quality:    opts.quality    ?? "fast",
  1214        requireTools,
  1215         taskType: opts.taskType,
  1216      }).map((m) => m.id);
  1217  if (initialModel && resolvedChain.length === 0) {
  1218    const catalog = getDynamicCatalogStatus();
  1219    throw new GroqClientError(
  1220      "INVALID_CONFIG",
  1221      `OpenRouter model ${initialModel} does not satisfy capability="${capability}" in the current catalog`,
  1222      {
  1223        context: {
  1224          providerName: "OpenRouter",
  1225          providerModel: initialModel,
  1226          providerCode: "MODEL_CAPABILITY_MISMATCH",
  1227          catalogLoaded: catalog.loaded,
  1228          catalogUsable: catalog.usable,
  1229          catalogStatus: catalog.lastRefreshStatus,
  1230          catalogError: catalog.lastRefreshError ?? undefined,
  1231        },
  1232      },
  1233    );
  1234  }
  1235  const maxFallbackModels =
  1236    Number.isInteger(opts.maxFallbackModels) && opts.maxFallbackModels! > 0
  1237      ? opts.maxFallbackModels
  1238      : undefined;
  1239  const chain = maxFallbackModels
  1240    ? resolvedChain.slice(0, maxFallbackModels)
  1241    : resolvedChain;
  1242  if (maxFallbackModels && resolvedChain.length > chain.length) {
  1243    console.info(
  1244      JSON.stringify({
  1245        scope: "openrouter-fallback",
  1246        code: "FALLBACK_CHAIN_BOUNDED",
  1247        requestedLength: resolvedChain.length,
  1248        appliedLength: chain.length,
  1249        maxFallbackModels,
  1250      }),
  1251    );
  1252  }
  1253  let lastError: GroqClientError | undefined;
  1254  const attemptedModels: string[] = [];
  1255
  1256  for (let i = 0; i < chain.length; i++) {
  1257    const model = chain[i] as string;
  1258    attemptedModels.push(model);
  1259    try {
  1260      return await openrouterCompleteRaw(messages, { ...opts, model });
  1261    } catch (err) {
  1262      // PR-003: treat MODEL_UNAVAILABLE (422/410) the same as MODEL_NOT_FOUND.
  1263      // Bounded Recovery also owns transient fallback: it disables the
  1264      // per-model retry above and advances immediately to the next live
  1265      // reasoning candidate instead of letting chat-agent replay a tool-loop
  1266      // chain on top of this one.
  1267      const fallbackOnTransient =
  1268        opts.retryTransient === false && isTransientError(err);
  1269      if ((isModelUnavailableError(err) || fallbackOnTransient) && i < chain.length - 1) {
  1270        const nextModel = chain[i + 1] as string;
  1271        console.warn(
  1272          JSON.stringify({
  1273            scope: "openrouter-fallback",
  1274            code: "MODEL_FALLBACK",
  1275            reason: err.code,
  1276            failedModel: model,
  1277            nextModel,
  1278            providerStatus: err.providerStatus,
  1279            attempt: i + 1,
  1280            remaining: chain.length - i - 1,
  1281          }),
  1282        );
  1283        lastError = err;
  1284        continue;
  1285      }
  1286      if (err instanceof GroqClientError) {
  1287        throw new GroqClientError(err.code, err.message, {
  1288          cause: err,
  1289          context: {
  1290            ...err.toProviderContext(),
  1291            providerModel: err.providerModel ?? model,
  1292            providerAttemptedModels: [...attemptedModels],
  1293          },
  1294        });
  1295      }
  1296      throw err;
  1297    }
  1298  }
  1299
  1300  if (lastError) {
  1301    throw new GroqClientError(lastError.code, lastError.message, {
  1302      cause: lastError,
  1303      context: {
  1304        ...lastError.toProviderContext(),
  1305        providerModel: lastError.providerModel ?? attemptedModels.at(-1),
  1306        providerAttemptedModels: [...attemptedModels],
  1307      },
  1308    });
  1309  }
  1310  const catalog = getDynamicCatalogStatus();
  1311  throw new GroqClientError("MODEL_NOT_FOUND", "All OpenRouter free-tier fallback models exhausted", {
  1312    context: {
  1313      providerAttemptedModels: [...attemptedModels],
  1314      catalogLoaded: catalog.loaded,
  1315      catalogUsable: catalog.usable,
  1316      catalogStatus: catalog.lastRefreshStatus,
  1317      catalogError: catalog.lastRefreshError ?? undefined,
  1318    },
  1319  });
  1320}

   123/**
   124 * Validates an API key by sending a minimal 1-token probe to the provider.
   125 *
   126 * Returns `{ valid: true }` on success.
   127 * Returns `{ valid: false, reason }` only on AUTH_ERROR — the key is definitively
   128 * rejected by the provider.
   129 * On any transient error (NETWORK_ERROR, TIMEOUT, SERVER_ERROR, NON_200) returns
   130 * `{ valid: true }` so a connectivity blip never prevents the user from saving a
   131 * key that might be perfectly valid.
   132 */
   133export async function validateProviderKey(
   134  provider: ProviderId,
   135  apiKey: string,
   136): Promise<{ valid: boolean; reason?: string }> {
   137  const config = loadProvider(provider);
   138  try {
   139    await probeProviderKey(config, apiKey);
   140    return { valid: true };
   141  } catch (err) {
   142    if (err instanceof GroqClientError && err.code === "AUTH_ERROR") {
   143      return { valid: false, reason: err.message };
   144    }
   145    // Transient error — allow the save; the next real request has provider fallback.
   146    return { valid: true };
   147  }
   148}
