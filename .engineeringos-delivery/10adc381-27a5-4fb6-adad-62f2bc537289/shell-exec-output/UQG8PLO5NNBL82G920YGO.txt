    87function unavailableResult(
    88  options: ProviderHealthProbeOptions,
    89  startedAt: number,
    90  fields: Pick<ProviderHealthProbeResult, "model" | "failureCode" | "failureReason">,
    91): ProviderHealthProbeResult {
    92  const model = fields.model ?? options.model ?? null;
    93  const attemptedModels = model ? [redactProviderErrorText(model).slice(0, 200)] : [];
    94  const failureCategory = failureCategoryFor(fields.failureCode);
    95  const recoveryAction = recoveryActionFor(fields.failureCode);
    96  return {
    97    provider: options.provider,
    98    status: "unavailable",
    99    providerUnavailable: true,
   100    toolCalling: false,
   101    structuredArguments: false,
   102    latencyMs: Math.max(0, Date.now() - startedAt),
   103    ...fields,
   104    model,
   105    report: {
   106      kind: "provider-health-report",
   107      version: 1,
   108      provider: options.provider,
   109      model: model ? redactProviderErrorText(model).slice(0, 200) : null,
   110      status: "unavailable",
   111      evidenceStatus: "incomplete",
   112      failureCategory,
   113      recoveryAction,
   114      attemptCount: 1,
   115      attemptedModels,
   116    },
   117  };
   118}

   120function failureCategoryFor(code?: ProviderHealthFailureCode): ProviderHealthFailureCategory {
   121  switch (code) {
   122    case "AUTH_ERROR":
   123      return "authentication";
   124    case "QUOTA":
   125      return "quota";
   126    case "RATE_LIMITED":
   127      return "rate-limit";
   128    case "MODEL_NOT_FOUND":
   129    case "MODEL_UNAVAILABLE":
   130    case "PLAN_RESTRICTED":
   131      return "catalog";
   132    case "EMPTY_RESPONSE":
   133      return "empty-response";
   134    case "NETWORK_ERROR":
   135    case "TIMEOUT":
   136      return "network";
   137    case "SERVER_ERROR":
   138      return "server";
   139    case "NON_200":
   140    case "INVALID_CONFIG":
   141      return "request";
   142    case "TOOL_CALL_UNSUPPORTED":
   143    case "MALFORMED_TOOL_ARGUMENTS":
   144    case "UNEXPECTED_TOOL_CALL":
   145      return "capability";
   146    default:
   147      return "unknown";
   148  }
   149}

   151function recoveryActionFor(code?: ProviderHealthFailureCode): OpenRouterFailureAction {
   152  if (
   153    code === "AUTH_ERROR" ||
   154    code === "QUOTA" ||
   155    code === "INVALID_CONFIG"
   156  ) return "stop-safely";
   157  if (code === "RATE_LIMITED") return "wait";
   158  if (code === "NETWORK_ERROR" || code === "TIMEOUT" || code === "SERVER_ERROR") return "retry";
   159  if (code === "MODEL_NOT_FOUND" || code === "MODEL_UNAVAILABLE" || code === "PLAN_RESTRICTED" || code === "EMPTY_RESPONSE") {
   160    return "choose-alternative";
   161  }
   162  if (code === "NON_200") return "narrow-request";
   163  return "stop-safely";
   164}

   166function parseProbeArguments(
   167  response: RawGroqResponse,
   168  options: ProviderHealthProbeOptions,
   169  startedAt: number,
   170): ProviderHealthProbeResult {
   171  const toolCalls = response.toolCalls ?? [];
   172  if (toolCalls.length === 0) {
   173    return unavailableResult(options, startedAt, {
   174      model: response.model,
   175      failureCode: "TOOL_CALL_UNSUPPORTED",
   176      failureReason: "The provider returned no tool call for a required probe.",
   177    });
   178  }
   179  if (toolCalls.length !== 1 || toolCalls[0]?.function.name !== PROBE_TOOL_NAME) {
   180    return unavailableResult(options, startedAt, {
   181      model: response.model,
   182      failureCode: "UNEXPECTED_TOOL_CALL",
   183      failureReason: "The provider returned an unexpected tool call for the probe.",
   184    });
   185  }
   186
   187  try {
   188    const parsed: unknown = JSON.parse(toolCalls[0].function.arguments);
   189    if (
   190      !parsed ||
   191      typeof parsed !== "object" ||
   192      Array.isArray(parsed) ||
   193      (parsed as { probe?: unknown }).probe !== "ok"
   194    ) {
   195      return unavailableResult(options, startedAt, {
   196        model: response.model,
   197        failureCode: "MALFORMED_TOOL_ARGUMENTS",
   198        failureReason: "The provider returned tool arguments without the required probe marker.",
   199      });
   200    }
   201  } catch {
   202    return unavailableResult(options, startedAt, {
   203      model: response.model,
   204      failureCode: "MALFORMED_TOOL_ARGUMENTS",
   205      failureReason: "The provider returned malformed JSON tool arguments.",
   206    });
   207  }
   208
   209  return {
   210    provider: options.provider,
   211    model: response.model ?? options.model ?? null,
   212    status: "usable",
   213    providerUnavailable: false,
   214    toolCalling: true,
   215    structuredArguments: true,
   216    latencyMs: Math.max(0, Date.now() - startedAt),
   217    report: {
   218      kind: "provider-health-report",
   219      version: 1,
   220      provider: options.provider,
   221      model: response.model ?? options.model
   222        ? redactProviderErrorText(response.model ?? options.model!).slice(0, 200)
   223        : null,
   224      status: "usable",
   225      evidenceStatus: "complete",
   226      failureCategory: null,
   227      recoveryAction: null,
   228      attemptCount: 1,
   229      attemptedModels: response.model ?? options.model ? [redactProviderErrorText(response.model ?? options.model!).slice(0, 200)] : [],
   230    },
