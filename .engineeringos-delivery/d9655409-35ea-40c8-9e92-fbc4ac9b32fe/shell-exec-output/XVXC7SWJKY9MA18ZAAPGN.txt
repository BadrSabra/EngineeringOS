   432  projectId: string,
   433  changes: Array<{

   435    absolutePath: string;
   436    newContent: string;
   437    originalContent?: string | null;

   462    id: messageId,

   464    role: "assistant",
   465    content: "Test repair proposal",
   466    createdAt: now,
   467  });
   468  await db.insert(aiChangeProposalsTable).values({
   469    id: proposalId,
   470    projectId,

   472    messageId,

   486  return proposalId;
