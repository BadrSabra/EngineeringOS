   650/** Build a provider context suffix for error messages, showing model/status/raw hint. */
   651function providerContextSuffix(err: ApiError): string {
   652  const parts: string[] = [];
   653  const ctx = err.providerContext as Record<string, unknown> | undefined;
   654  if (ctx) {
   655    if (ctx['providerName']) parts.push(`Provider: ${ctx['providerName']}`);
   656    if (ctx['providerModel']) parts.push(`Model: ${fmtModelId(String(ctx['providerModel']))}`);
   657    if (ctx['providerStatus']) parts.push(`Status: ${ctx['providerStatus']}`);
   658  }
   659  return parts.length > 0 ? ` (${parts.join(' · ')})` : '';
   660}

   676function describeAiError(err: unknown): string {
   677  if (err instanceof AiApiError) {
   678    const suffix = providerContextSuffix(err) + availabilitySuffix(err);
   679    switch (err.status) {
   680      case 400: return err.errorMessage + suffix;
   681      case 401: return (err.errorMessage || err.hint || 'AI API key is invalid — delete it and save a valid key from your provider\'s dashboard.') + suffix;
   682      case 403: return 'Access denied — you may not have permission on this project.';
   683      case 429: return (err.errorMessage || err.hint || 'AI rate limit reached — wait 30–60 seconds before retrying.') + suffix;
   684      case 422:
   685        if (err.code === 'model_output_invalid') {
   686          return 'The AI returned an unexpected response format — try rephrasing your message.';
   687        }
   688        if (err.code === 'MODEL_NOT_FOUND') {
   689          return (err.errorMessage || err.hint || 'The selected AI model is unavailable — try again or switch providers.') + suffix;
   690        }
   691        return (err.errorMessage || err.hint || 'AI provider configuration is invalid. Re-save your API key.') + suffix;
   692      case 428: return (err.errorMessage || err.hint || 'No AI key configured — save an OpenRouter, DeepSeek, or Groq API key first.') + suffix;
   693      case 502: return (err.errorMessage || err.hint || 'AI provider returned an error. Check your API key or try again.') + suffix;
   694      case 503: return (err.errorMessage || err.hint || 'AI provider is temporarily unreachable — try again in a moment.') + suffix;
   695      default:  return (err.errorMessage || `Request failed (${err.status}).`) + suffix;
   696    }
   697  }
   698  if (err instanceof Error) return err.message + availabilitySuffix(err);
   699  return String(err);
   700}
