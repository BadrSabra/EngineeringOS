   268  if (!project) return res.status(500).json({ error: "Project context unavailable" });

   272
   273  logger.info({ projectId }, "AI code review requested");
   274
   275  const MAX_FILE_CONTENTS_BYTES = 50_000;
   276  if (fileContents) {
   277    const invalidKey = Object.keys(fileContents).find(
   278      (k) => path.isAbsolute(k) || k.includes(".."),
   279    );
   280    if (invalidKey) {
   281      return res.status(400).json({
   282        error: `fileContents key "${invalidKey}" must be a relative path without traversal (no ".." segments)`,
   283      });
   284    }
   285    const totalSize = Object.values(fileContents).reduce((sum, v) => sum + v.length, 0);
   286    if (totalSize > MAX_FILE_CONTENTS_BYTES) {
   287      return res.status(413).json({
   288        error: `fileContents total size (${Math.round(totalSize / 1_000)} KB) exceeds the ${MAX_FILE_CONTENTS_BYTES / 1_000} KB limit — send fewer or smaller files`,
   289      });
   290    }
   291  }
   292
   293  const rlReview = await checkProjectRateLimitDb(projectId);
   294  if (!rlReview.allowed) {
   295    return res.status(429).json({
   296      error: `LLM rate limit exceeded — max ${LLM_RATE_LIMIT} calls per minute per project. Retry in ${rlReview.retryAfterSec}s.`,
   297    });
   298  }
   299
   300  const providerResolved = await requireProvider(req.userId, res, {
   301    qualityProfile: "code_review",
   302  });
   303  if (!providerResolved) return;
   304  const { provider, apiKey } = providerResolved;
   305
   306  const projectContext = await buildProjectContext(projectId, {
   307    sections: ["tasks", "metrics", "graphEntities", "graphRelationships", "events"],
   308  });
   309  let result: Awaited<ReturnType<typeof reviewCode>>;
   310  let effectiveProvider = provider;
   311  try {
   312    ({ result, effectiveProvider } = await runAgentWithFallback(
   313      req.userId,
   314      { provider, apiKey },
   315      (opts) => reviewCode(projectContext, fileContents, opts),
   316      { qualityProfile: "code_review" },
   317    ));
   318  } catch (err) {
   319    metadata.incomplete = true;
   320    recordTrace(metadata, "review", "failed", effectiveProvider);
   321    if (handleOrchestratorError(err, res, { projectId, operation: "code-review", provider: effectiveProvider })) return;
   322    throw err;
   323  }
   324
   325  // When the model output failed to parse, we still have the fallback data.
   326  // Serve it with a warning header so the UI can display something useful.
   327  if (result._parseError) {
   328    metadata.incomplete = true;
   329    recordTrace(metadata, "review", "incomplete", effectiveProvider);
   330    logger.warn(
   331      { projectId, parseCode: result._parseError.code, message: result._parseError.message, provider: effectiveProvider },

   337      code: "model_output_invalid",
   338      hint: "The AI model returned an unexpected response — try again in a moment.",
   339      parseCode: result._parseError.code,
   340    });
   341  }
   342
   343  invalidateContextCache(projectId);
   344
   345  await db.transaction(async (tx) => {
   346    await tx.insert(auditLogsTable).values({
   347      id: randomUUID(),
   348      entityType: "project",
   349      entityId: projectId,
   350      action: "ai_reviewed",
   351      projectId,
   352      actor: req.userId,
   353      stateBefore: {},
   354      stateAfter: { verdict: result.verdict, overallScore: result.overallScore },
   355    });
   356    await tx.insert(eventsTable).values({
   357      id: randomUUID(),
   358      type: "AiCodeReviewCompleted",
   359      projectId,
   360      severity: result.verdict === "approved" ? "success" : "warning",
   361      message: `AI code review: ${result.verdict} (score: ${result.overallScore}/100)`,
   362    });
   363  });
   364
   365  recordTrace(metadata, "review", "completed", effectiveProvider);
