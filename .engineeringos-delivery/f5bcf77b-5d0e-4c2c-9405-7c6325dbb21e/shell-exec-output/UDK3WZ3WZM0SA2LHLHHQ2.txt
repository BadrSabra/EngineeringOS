     1import { z } from "zod";
     2
     3/**
     4 * The capability contract is intentionally independent from provider/tool
     5 * prompts. A model may name a capability and provide business input, but it
     6 * cannot provide an executable command, profile, path, or process settings.
     7 */
     8export const CAPABILITY_CONTRACT_VERSION = 1 as const;
     9export const SUPPORTED_RECIPE_VERSIONS = [1] as const;
    10
    11export const CapabilityIdSchema = z
    12  .string()
    13  .min(1)
    14  .max(80)
    15  .regex(/^[A-Za-z0-9][A-Za-z0-9._:-]*$/, "capability IDs must be opaque safe identifiers");
    16export type CapabilityId = z.infer<typeof CapabilityIdSchema>;
    17
    18export const RecipeVersionSchema = z.number().int().min(1).max(100);
    19export type RecipeVersion = z.infer<typeof RecipeVersionSchema>;
    20
    21export const CapabilityRiskSchema = z.enum(["low", "medium", "high", "critical"]);
    22export type CapabilityRisk = z.infer<typeof CapabilityRiskSchema>;
    23
    24/**
    25 * Catalog metadata is deliberately a small, descriptive projection. It is
    26 * not an execution recipe: command names, argv, cwd, environment, profiles,
    27 * and process limits are not representable here.
    28 */
    29const CATALOG_RESERVED_INPUT_NAMES = new Set([
    30  "args",
    31  "argv",
    32  "command",
    33  "commandline",
    34  "commandtext",
    35  "cwd",
    36  "env",
    37  "environment",
    38  "executable",
    39  "executablepath",
    40  "profile",
    41  "shell",
    42  "shellcommand",
    43  "workdir",
    44  "workingdir",
    45  "workingdirectory",
    46]);
    47
    48export const CapabilityScopeKindSchema = z.enum([
    49  "none",
    50  "project",
    51  "paths",
    52  "file",
    53  "workspace",
    54]);
    55export type CapabilityScopeKind = z.infer<typeof CapabilityScopeKindSchema>;
    56
    57export const CapabilityCostSchema = z.enum(["low", "medium", "high"]);
    58export type CapabilityCost = z.infer<typeof CapabilityCostSchema>;
    59
    60export const CapabilityInputFieldSchema = z
    61  .object({
    62    name: z.string().min(1).max(64).regex(/^[A-Za-z][A-Za-z0-9_-]*$/),
    63    type: z.enum(["string", "number", "boolean", "object", "array"]),
    64    required: z.boolean(),
    65    description: z.string().min(1).max(180),
    66  })
    67  .strict()
    68  .superRefine((field, ctx) => {
    69    if (CATALOG_RESERVED_INPUT_NAMES.has(field.name.replace(/[_-]/g, "").toLowerCase())) {
    70      ctx.addIssue({
    71        code: z.ZodIssueCode.custom,
    72        path: ["name"],
    73        message: "catalog input fields cannot contain server-owned execution controls",
    74      });
    75    }
    76  });
    77export type CapabilityInputField = z.infer<typeof CapabilityInputFieldSchema>;
    78
    79export const CapabilityInputShapeSchema = z
    80  .object({
    81    type: z.literal("object"),
    82    fields: z.array(CapabilityInputFieldSchema).max(24),
    83  })
    84  .strict();
    85export type CapabilityInputShape = z.infer<typeof CapabilityInputShapeSchema>;
    86
    87export const CapabilityCatalogMetadataSchema = z
    88  .object({
    89    purpose: z.string().min(1).max(240),
    90    inputShape: CapabilityInputShapeSchema,
    91    defaultScope: CapabilityScopeKindSchema,
    92    supportedScopes: z.array(CapabilityScopeKindSchema).min(1).max(5),
    93    estimatedCost: CapabilityCostSchema,
    94    mutatesProject: z.boolean(),
    95    keywords: z.array(z.string().min(1).max(40)).max(16),
    96    allowedPhases: z.array(z.string().min(1).max(40)).max(8),
    97    /** Empty means all projects; values are never emitted in catalog output. */
    98    projectIds: z.array(z.string().min(1).max(160)).max(64),
    99    requiresAuthorization: z.boolean(),
   100    expectedEvidence: z.array(z.string().min(1).max(160)).max(5),
   101  })
   102  .strict()
   103  .superRefine((metadata, ctx) => {
   104    if (new Set(metadata.supportedScopes).size !== metadata.supportedScopes.length) {
   105      ctx.addIssue({
   106        code: z.ZodIssueCode.custom,
   107        path: ["supportedScopes"],
   108        message: "supported scopes must be unique",
   109      });
   110    }
   111    if (!metadata.supportedScopes.includes(metadata.defaultScope)) {
   112      ctx.addIssue({
   113        code: z.ZodIssueCode.custom,
   114        path: ["defaultScope"],
   115        message: "the default scope must be supported",
   116      });
   117    }
   118    if (new Set(metadata.allowedPhases).size !== metadata.allowedPhases.length) {
   119      ctx.addIssue({
   120        code: z.ZodIssueCode.custom,
   121        path: ["allowedPhases"],
   122        message: "allowed phases must be unique",
   123      });
   124    }
   125    if (new Set(metadata.projectIds).size !== metadata.projectIds.length) {
   126      ctx.addIssue({
   127        code: z.ZodIssueCode.custom,
   128        path: ["projectIds"],
   129        message: "project IDs must be unique",
   130      });
   131    }
   132    if (new Set(metadata.inputShape.fields.map((field) => field.name)).size !== metadata.inputShape.fields.length) {
   133      ctx.addIssue({
   134        code: z.ZodIssueCode.custom,
   135        path: ["inputShape", "fields"],
   136        message: "input field names must be unique",
   137      });
   138    }
   139  });
   140export type CapabilityCatalogMetadata = z.infer<typeof CapabilityCatalogMetadataSchema>;
   141
   142export const CapabilityPolicySchema = z
   143  .object({
   144    risk: CapabilityRiskSchema,
   145    requiresApproval: z.boolean(),
   146    allowedOperations: z.array(z.string().min(1).max(80)).max(32),
   147    /**
   148     * These profiles are part of the server-owned registration. They are not
   149     * accepted from capability input and are checked against invocation
   150     * context before an adapter can run.
   151     */
   152    approvedCommandProfiles: z.array(z.string().min(1).max(120)).max(32),
   153    maxInputBytes: z.number().int().min(1).max(1024 * 1024),
   154    maxOutputBytes: z.number().int().min(1).max(8 * 1024 * 1024),
   155  })
   156  .strict()
   157  .superRefine((policy, ctx) => {
   158    if (new Set(policy.allowedOperations).size !== policy.allowedOperations.length) {
   159      ctx.addIssue({
   160        code: z.ZodIssueCode.custom,
   161        path: ["allowedOperations"],
   162        message: "allowed operations must be unique",
   163      });
   164    }
   165    if (new Set(policy.approvedCommandProfiles).size !== policy.approvedCommandProfiles.length) {
   166      ctx.addIssue({
   167        code: z.ZodIssueCode.custom,
   168        path: ["approvedCommandProfiles"],
   169        message: "approved command profiles must be unique",
   170      });
   171    }
   172  });
   173export type CapabilityPolicy = z.infer<typeof CapabilityPolicySchema>;
   174
   175export const CapabilityDescriptorSchema = z
   176  .object({
   177    contractVersion: z.literal(CAPABILITY_CONTRACT_VERSION),
   178    id: CapabilityIdSchema,
   179    supportedRecipeVersions: z.array(RecipeVersionSchema).min(1).max(32),
   180    policy: CapabilityPolicySchema,
   181    catalog: CapabilityCatalogMetadataSchema.optional(),
   182  })
   183  .strict()
   184  .superRefine((descriptor, ctx) => {
   185    if (new Set(descriptor.supportedRecipeVersions).size !== descriptor.supportedRecipeVersions.length) {
   186      ctx.addIssue({
   187        code: z.ZodIssueCode.custom,
   188        path: ["supportedRecipeVersions"],
   189        message: "supported recipe versions must be unique",
   190      });
   191    }
   192    if (descriptor.supportedRecipeVersions.some(
   193      (version) => !(SUPPORTED_RECIPE_VERSIONS as readonly number[]).includes(version),
   194    )) {
   195      ctx.addIssue({
   196        code: z.ZodIssueCode.custom,
   197        path: ["supportedRecipeVersions"],
   198        message: "registration includes a recipe version unsupported by this contract",
   199      });
   200    }
   201  });
   202export type CapabilityDescriptor = z.infer<typeof CapabilityDescriptorSchema>;
   203
   204/**
   205 * This is the serializable portion of a registration. The implementation is
   206 * deliberately kept out of the public contract schema.
   207 */
   208export const CapabilityRegistrationSchema = CapabilityDescriptorSchema;
   209export type CapabilityRegistration = CapabilityDescriptor;

   241export type CapabilityExecutionContext = {
   242  /** Server-established project root; it is never read from capability input. */
   243  rootPath: string;
   244  /** Server-established operation name used for policy authorization. */
   245  operation: string;
   246  /** Server-owned approved profile set, if the capability uses profiles. */
   247  approvedCommandProfiles?: ReadonlySet<string>;
   248  /** Server-owned approval state; model output cannot satisfy this gate. */
   249  approvalState?: "APPROVED" | "PENDING_APPROVAL" | "REJECTED";
   250  signal?: AbortSignal;
   251};
   252
   253export type CapabilityAdapter<Input = unknown, Output = unknown> = CapabilityRegistration & {
   254  inputSchema: z.ZodType<Input>;
   255  outputSchema: z.ZodType<Output>;
   256  execute: (input: Input, context: CapabilityExecutionContext) => Promise<Output> | Output;
   257};
