   650function providerContextSuffix(err: ApiError): string {
   651  const parts: string[] = [];
   652  const ctx = err.providerContext as Record<string, unknown> | undefined;
   653  if (ctx) {
   654    if (ctx['providerName']) parts.push(`Provider: ${ctx['providerName']}`);
   655    if (ctx['providerModel']) parts.push(`Model: ${fmtModelId(String(ctx['providerModel']))}`);
   656    if (ctx['providerStatus']) parts.push(`Status: ${ctx['providerStatus']}`);
   657  }
   658  return parts.length > 0 ? ` (${parts.join(' · ')})` : '';
   659}

   661function availabilitySuffix(err: unknown): string {
   662  const value = err as { data?: unknown; availabilityState?: unknown; operatorAction?: unknown; correlationId?: unknown };
   663  const data = value.data && typeof value.data === 'object' ? value.data as Record<string, unknown> : {};
   664  const state = data.availabilityState ?? value.availabilityState;
   665  const action = data.operatorAction ?? value.operatorAction;
   666  const correlationId = data.correlationId ?? value.correlationId;
   667  const parts = [
   668    typeof state === 'string' ? `Status: ${state.replace(/_/g, ' ')}` : '',
   669    typeof action === 'string' ? `Next step: ${action}` : '',
   670    typeof correlationId === 'string' ? `Reference: ${correlationId}` : '',
   671  ].filter(Boolean);
   672  return parts.length > 0 ? ` (${parts.join(' · ')})` : '';
   673}

   675function describeAiError(err: unknown): string {
   676  if (err instanceof AiApiError) {
   677    const suffix = providerContextSuffix(err) + availabilitySuffix(err);
   678    switch (err.status) {
   679      case 400: return err.errorMessage + suffix;
   680      case 401: return (err.errorMessage || err.hint || 'AI API key is invalid — delete it and save a valid key from your provider\'s dashboard.') + suffix;
   681      case 403: return 'Access denied — you may not have permission on this project.';
   682      case 429: return (err.errorMessage || err.hint || 'AI rate limit reached — wait 30–60 seconds before retrying.') + suffix;
   683      case 422:
   684        if (err.code === 'model_output_invalid') {
   685          return 'The AI returned an unexpected response format — try rephrasing your message.';
   686        }
   687        if (err.code === 'MODEL_NOT_FOUND') {
   688          return (err.errorMessage || err.hint || 'The selected AI model is unavailable — try again or switch providers.') + suffix;
   689        }
   690        return (err.errorMessage || err.hint || 'AI provider configuration is invalid. Re-save your API key.') + suffix;
   691      case 428: return (err.errorMessage || err.hint || 'No AI key configured — save an OpenRouter, DeepSeek, or Groq API key first.') + suffix;
   692      case 502: return (err.errorMessage || err.hint || 'AI provider returned an error. Check your API key or try again.') + suffix;
   693      case 503: return (err.errorMessage || err.hint || 'AI provider is temporarily unreachable — try again in a moment.') + suffix;
   694      default:  return (err.errorMessage || `Request failed (${err.status}).`) + suffix;
   695    }
   696  }
   697  if (err instanceof Error) return err.message + availabilitySuffix(err);
   698  return String(err);
   699}

   715    .map((value) => value?.match(/\bsupport reference:\s*([a-z0-9][a-z0-9_-]{2,80})\b/i)?.[1])
   716    .find(Boolean);
   717
   718  let guidance = 'The AI provider could not complete this request. Try again in a moment.';
   719  if (message.failureKind === 'CONFIGURATION' || /authentication|unauthori[sz]|invalid.*key|credential/.test(raw)) {
   720    guidance = 'Authentication failed. Check the provider configuration or choose another provider, then retry.';
   721  } else if (message.failureKind === 'RATE_LIMIT' || /rate.?limit|quota|429/.test(raw)) {
   722    guidance = 'The provider quota or rate limit was reached. Wait a moment, then retry or choose another provider.';

   725  }
