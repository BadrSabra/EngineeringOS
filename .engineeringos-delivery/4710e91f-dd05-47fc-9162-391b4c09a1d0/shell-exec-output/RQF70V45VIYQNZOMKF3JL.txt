  5063function parseStructuredFailureMetadata(toolTrace: string | null | undefined): {
  5064  task: 'analyze' | 'review';
  5065  failureKind?: AiStreamErrorEvent['failureKind'];
  5066  retryable?: boolean;
  5067} | undefined {
  5068  if (!toolTrace) return undefined;

  5073      entry
  5074      && typeof entry === 'object'
  5075      && (entry as Record<string, unknown>).kind === 'structured_task_failure'
  5076    )) as Record<string, unknown> | undefined;
  5077    if (marker?.task !== 'analyze' && marker?.task !== 'review') return undefined;
  5078    const failureKinds: AiStreamErrorEvent['failureKind'][] = [

  5080    ];

  5082      task: marker.task,
  5083      failureKind: failureKinds.includes(marker.failureKind as AiStreamErrorEvent['failureKind'])
  5084        ? marker.failureKind as AiStreamErrorEvent['failureKind']
  5085        : undefined,
  5086      retryable: typeof marker.retryable === 'boolean' ? marker.retryable : undefined,
